factorial <- function(n) {
  if (n <= 1) return(1)
  return(n * factorial(n - 1))
}

# Function to get sum of factorials of digits
sum_of_digit_factorials <- function(n) {
  digits <- as.numeric(strsplit(as.character(n), "")[[1]])
  return(sum(sapply(digits, factorial)))
}

# Function to check if a number is equal to sum of factorials of its digits
is_equal_to_digit_factorial_sum <- function(n) {
  return(n == sum_of_digit_factorials(n))

  }
