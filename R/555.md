  if (n < 0) return(FALSE)
  sqrt_n <- sqrt(n)
  return(sqrt_n == floor(sqrt_n))
}

# Function to calculate sum of divisors (sigma function)
sigma <- function(n) {
  if (n <= 0) return(0)
  
  divisors <- c()
  for (i in 1:sqrt(n)) {
    if (n %% i == 0) {
      divisors <- c(divisors, i)
      if (i != n/i) {
        divisors <- c(divisors, n/i)
      }
    }
  }
  return(sum(divisors))
}
