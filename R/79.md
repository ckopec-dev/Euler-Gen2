## Problem Statement
A common security method used for online banking is to ask the user for three random characters from a passcode. For example, if the passcode was 531278, they may ask for the 2nd, 3rd, and 5th characters; the expected reply would be: 317.

The text file, `keylog.txt`, contains fifty successful login attempts. Given that the three characters are always asked for in order, analyse the file to determine the shortest possible secret passcode of unknown length.

## Solution Approach
1. Parse the keylog file to extract all login attempts
2. Analyze the ordering relationships between digits
3. Use topological sorting to determine the shortest passcode
4. Handle duplicate attempts and overlapping patterns

# Function to build dependency graph
build_graph <- function(attempts) {
  # Initialize adjacency list
  graph <- list()
  
  # Process each attempt
  for (attempt in attempts) {
    # For each consecutive pair of digits, add dependency
    for (i in 1:(length(attempt) - 1)) {
      from_digit <- attempt[i]
      to_digit <- attempt[i + 1]
      
      # Add to adjacency list
      if (!(from_digit %in% names(graph))) {
        graph[[from_digit]] <- c()
      }
      if (!(to_digit %in% graph[[from_digit]])) {
        graph[[from_digit]] <- c(graph[[from_digit]], to_digit)
      }
    }
  }
  
  return(graph)
}

# Function to perform topological sort (Kahn's algorithm)
topological_sort <- function(graph) {
  # Calculate in-degrees
  in_degree <- rep(0, length(graph))
  names(in_degree) <- names(graph)
  
  for (node in names(graph)) {
    for (neighbor in graph[[node]]) {
      if (neighbor %in% names(in_degree)) {
        in_degree[neighbor] <- in_degree[neighbor] + 1
      }
    }
  }
  
  # Initialize queue with nodes having in-degree 0
  queue <- names(in_degree)[in_degree == 0]
  result <- c()
  
  while (length(queue) > 0) {
    # Remove node from queue
    current <- queue[1]
    queue <- queue[-1]
    result <- c(result, current)
    
    # Process neighbors
    if (current %in% names(graph)) {
      for (neighbor in graph[[current]]) {
        if (neighbor %in% names(in_degree)) {
          in_degree[neighbor] <- in_degree[neighbor] - 1
          if (in_degree[neighbor] == 0) {
            queue <- c(queue, neighbor)
          }
        }
      }
    }
  }
  
  return(result)
}
