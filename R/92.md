# Euler Problem 92 Solution in R

## Problem Statement
A number chain is created by continuously adding the square of the digits of a number to form a new number until it has been seen before. For example:
- 44 → 32 → 13 → 10 → 1 → 1
- 85 → 89 → 145 → 42 → 20 → 4 → 16 → 37 → 58 → 89

The chain that starts with 85 ends in 89, and the chain that starts with 44 ends in 1.

Any chain that arrives at 1 or 89 will become stuck in an endless loop. What is most amazing is that EVERY starting number will eventually arrive at 1 or 89.

How many starting numbers below ten million will arrive at 89?

## Solution Approach
1. Create a function to compute the sum of squares of digits
2. Use memoization to avoid recalculating the same chains
3. For each number, follow the chain until reaching 1 or 89
4. Count how many reach 89

## R Code Implementation

```r
# Function to calculate sum of squares of digits
sum_of_squares <- function(n) {
  sum((as.numeric(strsplit(as.character(n), "")[[1]]))^2)
}

# Function to determine if a chain ends in 89
# Using memoization to store results
chain_ends_in_89 <- function(n, memo) {
  # Base cases
  if (n == 1) return(FALSE)
  if (n == 89) return(TRUE)
  
  # Check if already computed
  if (!is.na(memo[n])) {
    return(memo[n])
  }
  
  # Compute next number in chain
  next_n <- sum_of_squares(n)
  
  # Recursively check the next number
  result <- chain_ends_in_89(next_n, memo)
  
  # Store result in memo
  memo[n] <- result
  
  return(result)
}

# Main solution
solve_euler_92 <- function(limit) {
  # Initialize memoization array
  memo <- rep(NA, limit + 1)
  
  count <- 0
  
  # Check each number from 1 to limit - 1
  for (i in 1:(limit - 1)) {
    if (chain_ends_in_89(i, memo)) {
      count <- count + 1
    }
  }
  
  return(count)
}

# Execute solution
result <- solve_euler_92(10000000)
print(paste("Answer:", result))
```

## Alternative Implementation (More Efficient)

```r
# More efficient approach using direct chain tracking
solve_euler_92_efficient <- function(limit) {
  # Precompute all results up to limit
  results <- rep(0, limit)  # 0 = unknown, 1 = ends in 1, 2 = ends in 89
  
  # Helper function to compute chain
  compute_chain <- function(n) {
    chain <- c()
    current <- n
    
    while (current != 1 && current != 89) {
      # Check if we've seen this number before
      if (current <= limit && results[current] != 0) {
        # Propagate the result back
        for (i in seq_along(chain)) {
          if (chain[i] <= limit) {
            results[chain[i]] <- results[current]
          }
        }
        return(results[current] == 2)  # TRUE if ends in 89
      }
      
      chain <- c(chain, current)
      current <- sum((as.numeric(strsplit(as.character(current), "")[[1]]))^2)
    }
    
    # Mark the result
    end_value <- if (current == 1) 1 else 2
    
    # Store results for all numbers in chain
    for (i in seq_along(chain)) {
      if (chain[i] <= limit) {
        results[chain[i]] <- end_value
      }
    }
    
    return(end_value == 2)
  }
  
  count <- 0
  for (i in 1:(limit - 1)) {
    if (results[i] == 0) {
      if (compute_chain(i)) {
        count <- count + 1
      }
    } else if (results[i] == 2) {
      count <- count + 1
    }
  }
  
  return(count)
}

# Execute the efficient solution
result_efficient <- solve_euler_92_efficient(10000000)
print(paste("Answer:", result_efficient))
```

## Final Solution

```r
# Simple and clean solution
sum_of_squares <- function(n) {
  sum((as.numeric(strsplit(as.character(n), "")[[1]]))^2)
}

# Memoization approach
solve_euler_92 <- function(limit) {
  # Memoization vector
  memo <- rep(0, limit + 1)
  memo[1] <- 1  # 1 ends in 1
  memo[89] <- 89  # 89 ends in 89
  
  count <- 0
  
  # Process each number
  for (i in 1:(limit - 1)) {
    if (memo[i] == 0) {
      # Build the chain
      chain <- c()
      current <- i
      
      while (current != 1 && current != 89 && memo[current] == 0) {
        chain <- c(chain, current)
        current <- sum_of_squares(current)
      }
      
      # Determine the end of the chain
      if (current == 1) {
        end_value <- 1
      } else if (current == 89) {
        end_value <- 89
      } else {
        end_value <- memo[current]
      }
      
      # Store results for all numbers in chain
      for (j in seq_along(chain)) {
        memo[chain[j]] <- end_value
      }
      
      # Count if it ends in 89
      if (end_value == 89) {
        count <- count + 1
      }
    } else if (memo[i] == 89) {
      count <- count + 1
    }
  }
  
  return(count)
}

# Run the solution
answer <- solve_euler_92(10000000)
print(answer)
```

## Answer
The number of starting numbers below ten million that arrive at 89 is **8581146**.
