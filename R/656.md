count_divisors <- function(n) {
  if (n == 1) return(1)
  
  # Get prime factorization
  factors <- prime_factorization(n)
  
  # For n = p1^a1 * p2^a2 * ... * pk^ak
  # Number of divisors = (a1+1)(a2+1)...(ak+1)
  divisors <- 1
  for (factor in factors) {
    divisors <- divisors * (factor + 1)
  }
  return(divisors)
}

  factors <- c()
  d <- 2
  
  while (d * d <= n) {
    while (n %% d == 0) {
      factors <- c(factors, d)
      n <- n / d
    }
    d <- d + 1
  }
  
  if (n > 1) {
    factors <- c(factors, n)
  }
