# Function to count partitions of n into distinct parts, each ≤ k
count_distinct_partitions <- function(n, k) {
  # Memoization table
  memo <- matrix(0, nrow = n + 1, ncol = k + 1)
  
  # Base case: 0 can be partitioned in 1 way (empty partition)
  memo[1, ] <- 1
  
  # Fill the memoization table
  for (i in 1:n) {
    for (j in 1:min(i, k)) {
      if (i == j) {
        # If we use exactly j, we need to partition (i-j) with parts ≤ (j-1)
        memo[i + 1, j + 1] <- memo[i - j + 1, j] + memo[i + 1, j]
      } else {
        # Either use j or don't use j
        memo[i + 1, j + 1] <- memo[i + 1, j] + memo[i - j + 1, j]
      }
    }
  }
  
  return(memo[n + 1, k + 1])
}

count_distinct_partitions_simple <- function(n, k) {
  # dp[i][j] = number of ways to partition i using distinct parts ≤ j
  dp <- matrix(0, nrow = n + 1, ncol = k + 1)
  
  # Base case: 0 can be partitioned in 1 way (empty set)
  for (j in 1:(k + 1)) {
    dp[1, j] <- 1
  }
  
  # Fill the DP table
  for (i in 1:n) {
    for (j in 1:min(i, k)) {
      # Don't use j: dp[i][j] = dp[i][j-1]
      # Use j: dp[i][j] += dp[i-j][j-1] (if i >= j)
      dp[i + 1, j + 1] <- dp[i + 1, j] + if (i >= j) dp[i - j + 1, j] else 0
    }
  }
  
  return(dp[n + 1, k + 1])
}

distinct_partitions_recursive <- function(n, k, memo = NULL) {
  if (is.null(memo)) {
    memo <- matrix(-1, nrow = n + 1, ncol = k + 1)
  }
  
  # Base cases
  if (n == 0) return(1)
  if (n < 0 || k == 0) return(0)
  
  # Check memo
  if (memo[n + 1, k + 1] != -1) {
    return(memo[n + 1, k + 1])
  }
  
  # Either include k or exclude k
  result <- distinct_partitions_recursive(n, k - 1, memo) + 
            distinct_partitions_recursive(n - k, k - 1, memo)
  
  memo[n + 1, k + 1] <- result
  return(result)
}
