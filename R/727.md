# Function to get prime factorization
prime_factors <- function(n) {
  factors <- c()
  d <- 2
  while (d * d <= n) {
    while (n %% d == 0) {
      factors <- c(factors, d)
      n <- n %/% d
    }
    d <- d + 1
  }
  if (n > 1) factors <- c(factors, n)
  return(factors)
}

# Function to get all divisors of a number
divisors <- function(n) {
  divs <- c()
  for (i in 1:sqrt(n)) {
    if (n %% i == 0) {
      divs <- c(divs, i)
      if (i != n %/% i) divs <- c(divs, n %/% i)
    }
  }
  return(sort(divs))
}
