# Clean solution for Euler Problem 554
euler554 <- function() {
  # The number of ways to tile a 3×n rectangle with 1×2 dominoes
  # follows the recurrence relation:
  # f(n) = f(n-1) + 5*f(n-2) + f(n-3) - f(n-4)
  # with base cases f(0) = 1, f(1) = 1, f(2) = 5, f(3) = 11
  
  n <- 1000
  MOD <- 1000000007
  
  # Initialize array
  f <- numeric(n + 1)
  f[1] <- 1  # f(0) = 1
  f[2] <- 1  # f(1) = 1
  f[3] <- 5  # f(2) = 5
  f[4] <- 11 # f(3) = 11
  
  # Fill using recurrence relation
  for (i in 5:(n + 1)) {
    f[i] <- (f[i-1] + 5 * f[i-2] + f[i-3] - f[i-4]) %% MOD
  }
  
  return(f[n + 1])
}
