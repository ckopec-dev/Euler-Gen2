count_partitions_at_least_2 <- function(n) {
  if (n < 2) return(0)
  if (n == 2) return(1)
  
  # dp[i] will store number of ways to partition i
  dp <- numeric(n + 1)
  dp[0] <- 1  # One way to partition 0 (empty partition)

    # Update dp array for all values from part to n

      dp[i] <- dp[i] + dp[i - part]

  }
