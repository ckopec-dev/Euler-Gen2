# Function to compute continued fraction coefficients for sqrt(n)
contfrac_coeffs <- function(n) {
  if (sqrt(n) == floor(sqrt(n))) return(NULL)  # Perfect square
  
  a0 <- floor(sqrt(n))
  coeffs <- c(a0)
  
  # Use the continued fraction algorithm
  m <- 0
  d <- 1
  a <- a0
  
  repeat {
    m <- d * a - m
    d <- (n - m * m) / d
    a <- floor((a0 + m) / d)
    coeffs <- c(coeffs, a)
    
    if (a == 2 * a0) break  # Periodicity condition
  }
  
  return(coeffs)
}

# Function to get convergents from continued fraction coefficients
get_convergents <- function(coeffs) {
  if (is.null(coeffs)) return(NULL)
  
  # Initialize first two convergents
  h <- c(coeffs[1], coeffs[1] * coeffs[2] + 1)
  k <- c(1, coeffs[2])
  
  convergents <- cbind(h, k)
  
  # Generate remaining convergents
  for (i in 3:length(coeffs)) {
    h_new <- coeffs[i] * h[length(h)] + h[length(h)-1]
    k_new <- coeffs[i] * k[length(k)] + k[length(k)-1]
    h <- c(h, h_new)
    k <- c(k, k_new)
  }
  
  return(cbind(h, k))
}

best_approximation <- function(n) {
  coeffs <- contfrac_coeffs(n)
  if (is.null(coeffs)) return(NULL)
