# Euler Problem 201 Solution in R

## Problem Statement
Find the sum of all the unique subsequence sums of the set {1, 2, 3, ..., 70} that are perfect squares.

## Solution Approach
1. Generate all possible subsequences of the set {1, 2, 3, ..., 70}
2. Calculate the sum of each subsequence
3. Identify which sums are perfect squares
4. Sum all unique perfect square sums

## Implementation

```r
# Function to check if a number is a perfect square
is.perfect.square <- function(n) {
  if (n < 0) return(FALSE)
  sqrt_n <- sqrt(n)
  return(sqrt_n == as.integer(sqrt_n))
}

# Generate all possible subsequences and their sums
# For efficiency, we'll use a more direct approach

# Create the set {1, 2, 3, ..., 70}
numbers <- 1:70

# Generate all possible subset sums using bit manipulation
# This is more efficient than generating all subsets explicitly
subset_sums <- c()

# Generate all possible sums using bit manipulation
n <- length(numbers)
for (i in 0:(2^n - 1)) {
  sum <- 0
  for (j in 1:n) {
    if (i %% (2^j) >= 2^(j-1)) {
      sum <- sum + numbers[j]
    }
  }
  subset_sums <- c(subset_sums, sum)
}

# Find unique perfect square sums
unique_sums <- unique(subset_sums)
perfect_squares <- unique_sums[is.perfect.square(unique_sums)]

# Calculate the final answer
result <- sum(perfect_squares)
result
```

## Most Efficient Solution

```r
# Efficient solution using dynamic programming approach
# We'll build up sums incrementally

# Function to solve the problem efficiently
solve_euler201 <- function() {
  # Set of numbers from 1 to 70
  n <- 70
  numbers <- 1:n
  
  # Use a set to track possible sums (more efficient than vector)
  possible_sums <- numeric(1)  # Start with 0
  possible_sums[1] <- 0
  
  # For each number, update all possible sums
  for (num in numbers) {
    # Create new sums by adding current number to existing sums
    new_sums <- possible_sums + num
    possible_sums <- unique(c(possible_sums, new_sums))
  }
  
  # Find perfect squares among the sums
  perfect_squares <- possible_sums[is.perfect.square(possible_sums)]
  
  return(sum(perfect_squares))
}

# Helper function for perfect square check
is.perfect.square <- function(n) {
  if (n < 0) return(FALSE)
  sqrt_n <- sqrt(n)
  return(sqrt_n == as.integer(sqrt_n))
}

# Execute the solution
result <- solve_euler201()
result
```
