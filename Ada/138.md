# Euler Problem 138 Solution in Ada

This problem asks us to find the sum of the first ten values of n where the triangle with sides n, n+1, and n+2 has an integer area.

Let me break down the approach:

1. For a triangle with sides a, b, c, we can use Heron's formula to calculate area
2. We need to check when the area is an integer
3. We're looking for triangles where the sides form consecutive integers (n, n+1, n+2)

```ada

   type Integer_64 is range -2**63 .. 2**63-1;
   
   function Is_Perfect_Square(n : Integer_64) return Boolean is
      Root : Integer_64 := Integer_64(Sqrt(Float(n)));
   begin
      return Root * Root = n;
   end Is_Perfect_Square;
   
   function Area_Is_Integer(n : Integer_64) return Boolean is
      -- For triangle with sides n, n+1, n+2
      -- Using Heron's formula: area = sqrt(s*(s-a)*(s-b)*(s-c))
      -- where s = (a+b+c)/2
      
      s : Integer_64 := 3 * n + 3;  -- semi-perimeter
      a : Integer_64 := n;
      b : Integer_64 := n + 1;
      c : Integer_64 := n + 2;
      
      -- Heron's formula: area = sqrt(s*(s-a)*(s-b)*(s-c))
      -- s*(s-a)*(s-b)*(s-c) = (3n+3)*n*1*(3n+1) = (3n+3)*n*(3n+1)
      -- = 3(n+1)*n*(3n+1) = 3*n*(3n^2 + n + 3n + 1) = 3*n*(3n^2 + 4n + 1)
      
      term1 : Integer_64 := s / 2;        -- (3n+3)/2
      term2 : Integer_64 := s / 2 - a;    -- n
      term3 : Integer_64 := s / 2 - b;    -- 1  
      term4 : Integer_64 := s / 2 - c;    -- (3n+1)/2
      
      discriminant : Integer_64;
   begin
      -- Simplified calculation:
      -- For sides n, n+1, n+2: area = sqrt(3*n*(3n^2 + 4n + 1))
      -- We want this to be an integer, so 3*n*(3n^2 + 4n + 1) must be a perfect square
      
      discriminant := 3 * n * (3 * n * n + 4 * n + 1);
      return Is_Perfect_Square(discriminant);
   end Area_Is_Integer;
   
   function Get_Triangle_Area(n : Integer_64) return Integer_64 is
      s : Integer_64 := 3 * n + 3;  -- semi-perimeter  
      discriminant : Integer_64 := 3 * n * (3 * n * n + 4 * n + 1);
      sqrt_discriminant : Float;
   begin
      sqrt_discriminant := Sqrt(Float(discriminant));
      return Integer_64(sqrt_discriminant);
   end Get_Triangle_Area;
   
   Count : Integer := 0;
   Total_Sum : Integer_64 := 0;
   N : Integer_64 := 1;
begin

         Count := Count + 1;
         Total_Sum := Total_Sum + N;

         Put(" with area = ");
         Put(Get_Triangle_Area(N), 0);
         New_Line;
      end if;
      N := N + 1;
   end loop;
   
   Put("Sum of first 10 values: ");
   Put(Total_Sum, 0);
   New_Line;
end Euler_138;
```

```ada
procedure Euler_138 is
   type Integer_64 is range -2**63 .. 2**63-1;
   
   function Is_Perfect_Square(n : Integer_64) return Boolean is
      Root : Integer_64;
   begin
      if n < 0 then
         return False;
      elsif n = 0 then
         return True;
      else
         Root := Integer_64(Sqrt(Float(n)));
         -- Check a few values around the sqrt to handle floating point errors
         for I in -1..1 loop
            if (Root + I) * (Root + I) = n then
               return True;
            end if;
         end loop;
         return False;
      end if;
   end Is_Perfect_Square;
   
   function Area_Is_Integer(n : Integer_64) return Boolean is
      -- For triangle with sides n, n+1, n+2
      -- Using Heron's formula: area = sqrt(s*(s-a)*(s-b)*(s-c))
      -- s = (3n + 3)/2 = 3(n+1)/2
      -- Area² = [3(n+1)/2] * [3(n+1)/2 - n] * [3(n+1)/2 - (n+1)] * [3(n+1)/2 - (n+2)]
      --       = [3(n+1)/2] * [(3n+3-2n)/2] * [(3n+3-2n-2)/2] * [(3n+3-2n-4)/2]
      --       = [3(n+1)/2] * [(n+3)/2] * [(n+1)/2] * [(3n-1)/2]
      --       = (3(n+1)(n+3)(n+1)(3n-1))/16
      --       = 3(n+1)²(n+3)(3n-1)/16
      
      discriminant : Integer_64;
   begin
      discriminant := 3 * (n + 1) * (n + 1) * (n + 3) * (3 * n - 1);
      
      -- For area to be integer, 16 * discriminant must be a perfect square
      -- But we want the actual area to be integer, so discriminant must be 
      -- a perfect square divided by 16
      
      return Is_Perfect_Square(discriminant);
   end Area_Is_Integer;
   
   Count : Integer := 0;
   Total_Sum : Integer_64 := 0;
   N : Integer_64 := 1;
```
