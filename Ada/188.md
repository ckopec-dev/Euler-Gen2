   function Modular_Exponentiation(Base, Exp, Modulus : Long_Long_Integer) return Long_Long_Integer is
      Result : Long_Long_Integer := 1;

         if Exp_Mod mod 2 = 1 then
            Result := (Result * Base_Mod) mod Modulus;
         end if;

         Exp_Mod := Exp_Mod / 2;
      end loop;
      return Result;
   end Modular_Exponentiation;

            Result := (Result * B) mod Mod;
         end if;
         B := (B * B) mod Mod;
         E := E / 2;
      end loop;
      return Result;
   end Power_Mod;

   -- The answer to Euler problem 188 is the last 8 digits of 3↑↑1855
   -- This is computed iteratively:
   -- 3↑↑1 = 3
   -- 3↑↑2 = 3^3 = 27  
   -- 3↑↑3 = 3^(3^3) = 3^27 = ...
   
   Result : Long_Long_Integer := 3;
   Last_8 : constant Long_Long_Integer := 100_000_000;  -- 10^8

   for i in 2..1855 loop
