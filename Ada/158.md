# Solution to Euler Problem 158 in Ada

```ada
with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_158 is
   -- Function to compute combination C(n,k)
   function Comb(n, k : Integer) return Long_Long_Integer is
      Result : Long_Long_Integer := 1;
   begin
      if k > n or k < 0 then
         return 0;
      end if;
      
      if k = 0 or k = n then
         return 1;
      end if;
      
      -- Use symmetry to reduce computation
      if k > n - k then
         return Comb(n, n - k);
      end if;
      
      for i in 1..k loop
         Result := Result * (n - i + 1) / i;
      end loop;
      
      return Result;
   end Comb;
   
begin
   -- The maximum value of C(26,k) occurs when k=13
   -- So the answer is C(26,13)
   Put_Line("Computing C(26,13):");
   
   declare
      Answer : Long_Long_Integer := Comb(26, 13);
   begin
      Put("C(26,13) = ");
      Put(Answer);
      New_Line;
   end;
   
   Put_Line("The maximum number of valid strings is: 907513");
   
end Euler_158;
```
