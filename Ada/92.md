# Euler Problem 92 Solution in Ada

## Problem Statement
A number chain is created by continuously adding the square of the digits in a number to form a new number until it has been seen before. For example:
- 44 → 32 → 13 → 10 → 1 → 1
- 85 → 89 → 145 → 42 → 20 → 4 → 16 → 37 → 58 → 89

Any chain that arrives at 1 or 89 will become stuck in an endless loop. What is most amazing is that EVERY starting number will eventually arrive at 1 or 89.

How many starting numbers below ten million will arrive at 89?

## Solution Approach
1. Create a function to compute the sum of squares of digits
2. Use memoization to avoid recalculating chains
3. For each number, follow its chain until reaching 1, 89, or a previously computed result
4. Count how many start with 89

## Ada Implementation

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_92 is
   
   MAX_NUMBER : constant := 10_000_000;
   
   -- Array to store the final destination of each number (1 or 89)
   Destination : array (1 .. MAX_NUMBER) of Integer;
   
   -- Initialize the array
   procedure Initialize_Destination is
   begin
      for I in 1 .. MAX_NUMBER loop
         Destination(I) := 0;  -- 0 means not computed yet
      end loop;
      
      -- Base cases
      Destination(1) := 1;
      Destination(89) := 89;
   end Initialize_Destination;
   
   -- Compute sum of squares of digits
   function Sum_Of_Squares_Of_Digits(N : Integer) return Integer is
      Result : Integer := 0;
      Temp : Integer := N;
   begin
      while Temp > 0 loop
         declare
            Digit : Integer := Temp mod 10;
         begin
            Result := Result + Digit * Digit;
            Temp := Temp / 10;
         end;
      end loop;
      return Result;
   end Sum_Of_Squares_Of_Digits;
   
   -- Find the destination of a number chain
   function Get_Destination(N : Integer) return Integer is
      Visited : array (1 .. MAX_NUMBER) of Boolean := (others => False);
      Current : Integer := N;
      Temp : Integer;
   begin
      loop
         -- If we've already computed this result, return it
         if Destination(Current) /= 0 then
            -- Propagate the result back to all numbers in chain
            Temp := N;
            while Temp /= Current loop
               if Destination(Temp) = 0 then
                  Destination(Temp) := Destination(Current);
               end if;
               Temp := Sum_Of_Squares_Of_Digits(Temp);
            end loop;
            return Destination(Current);
         end if;
         
         -- If we've seen this number before in current chain, we found a loop
         if Visited(Current) then
            -- This means we're at the start of a cycle
            -- We need to determine which cycle we're in
            Temp := Current;
            while Temp /= 1 and Temp /= 89 loop
               Temp := Sum_Of_Squares_Of_Digits(Temp);
            end loop;
            
            -- The result is either 1 or 89
            declare
               Final : Integer := Temp;
            begin
               -- Store the result for all numbers in this chain
               Temp := N;
               while Temp /= Current loop
                  if Destination(Temp) = 0 then
                     Destination(Temp) := Final;
                  end if;
                  Temp := Sum_Of_Squares_Of_Digits(Temp);
               end loop;
               return Final;
            end;
         end if;
         
         Visited(Current) := True;
         Current := Sum_Of_Squares_Of_Digits(Current);
      end loop;
   end Get_Destination;
   
   -- Simplified approach: process numbers one by one
   function Get_Destination_Simple(N : Integer) return Integer is
      Current : Integer := N;
      Visited : array (1 .. MAX_NUMBER) of Boolean := (others => False);
   begin
      loop
         if Current = 1 or Current = 89 then
            return Current;
         end if;
         
         if Destination(Current) /= 0 then
            return Destination(Current);
         end if;
         
         if Visited(Current) then
            -- This means we've hit a cycle, and the chain will continue forever
            -- but we know that all chains eventually reach either 1 or 89
            -- So this shouldn't happen in a correct implementation
            return Current;  -- Should never be reached
         end if;
         
         Visited(Current) := True;
         Current := Sum_Of_Squares_Of_Digits(Current);
      end loop;
   end Get_Destination_Simple;
   
   -- Even simpler approach with proper memoization
   function Compute_Destination(N : Integer) return Integer is
      Temp : Integer := N;
      Chain : array (1 .. 100) of Integer;  -- Maximum chain length
      Chain_Length : Integer := 0;
      Found : Boolean := False;
      Result : Integer;
   begin
      loop
         if Temp = 1 or Temp = 89 then
            Result := Temp;
            exit;
         end if;
         
         -- Check if we've already computed this value
         if Destination(Temp) /= 0 then
            Result := Destination(Temp);
            exit;
         end if;
         
         Chain_Length := Chain_Length + 1;
         Chain(Chain_Length) := Temp;
         
         -- Check for cycles (if we see a number we've seen before)
         for I in 1 .. Chain_Length - 1 loop
            if Chain(I) = Temp then
               Found := True;
               exit;
            end if;
         end loop;
         
         if Found then
            -- We found a cycle, but since we know it must end in 1 or 89,
            -- we can determine the result from where we started in the cycle
            Temp := Chain(Chain_Length);
            while Temp /= 1 and Temp /= 89 loop
               Temp := Sum_Of_Squares_Of_Digits(Temp);
            end loop;
            Result := Temp;
            exit;
         end if;
         
         Temp := Sum_Of_Squares_Of_Digits(Temp);
      end loop;
      
      -- Store the computed result for all numbers in the chain
      for I in 1 .. Chain_Length loop
         if Destination(Chain(I)) = 0 then
            Destination(Chain(I)) := Result;
         end if;
      end loop;
      
      return Result;
   end Compute_Destination;

begin
   Initialize_Destination;
   
   declare
      Count : Integer := 0;
   begin
      for I in 2 .. MAX_NUMBER - 1 loop
         if Compute_Destination(I) = 89 then
            Count := Count + 1;
         end if;
      end loop;
      
      Put("Answer: ");
      Put(Count);
      New_Line;
   end;
   
end Euler_92;
```

```ada
      end loop;
      return Result;
   end Sum_Of_Squares;

   function Get_Dest(N : Integer) return Integer is
      Current : Integer := N;
      Temp : Integer;
      Visited : array (0 .. MAX_NUMBER) of Boolean := (others => False);
   begin
      while Current /= 1 and Current /= 89 loop
         if Destination(Current) /= 0 then
            -- Propagate result back to all visited numbers in this chain
            Temp := N;
            while Temp /= Current loop
               if Destination(Temp) = 0 then
                  Destination(Temp) := Destination(Current);
               end if;
               Temp := Sum_Of_Squares(Temp);
            end loop;
            return Destination(Current);
         end if;
         
         if Visited(Current) then
            -- Cycle detected, but we know it must be 1 or 89
            -- In this case, just compute it normally
            exit;
         end if;
         
         Visited(Current) := True;
         Current := Sum_Of_Squares(Current);
      end loop;
      
      -- If we reached 1 or 89 directly
      if Current = 1 or Current = 89 then
         Temp := N;
         while Temp /= Current loop
            if Destination(Temp) = 0 then
               Destination(Temp) := Current;
            end if;
            Temp := Sum_Of_Squares(Temp);
         end loop;
         return Current;
      else
         -- This case shouldn't happen but for safety:
         return Current;
      end if;
   end Get_Dest;
   
begin
   -- Initialize base cases
   Destination(1) := 1;
   Destination(89) := 89;
   
   declare
      Count : Integer := 0;
   begin
      -- Process all numbers from 2 to MAX_NUMBER-1 (since 1 ends at 1)
      for I in 2 .. MAX_NUMBER - 1 loop
         if Get_Dest(I) = 89 then
            Count := Count + 1;
         end if;
      end loop;
      
      Put_Line("Answer: " & Integer'Image(Count));
   end;
   
end Euler_92;
```

## Final Optimized Solution

```ada
with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_92 is
   
   MAX_NUMBER : constant := 10_000_000;
   
   -- Memoization array: 0 = not computed, 1 = ends at 1, 89 = ends at 89
   Memo : array (0 .. MAX_NUMBER) of Integer := (others => 0);
   
   -- Compute sum of squares of digits
   function Sum_Of_Squares(N : Integer) return Integer is
      Result : Integer := 0;
   begin
      while N > 0 loop
         declare
            Digit : Integer := N mod 10;
         begin
            Result := Result + Digit * Digit;
            N := N / 10;
         end;
      end loop;
      return Result;
   end Sum_Of_Squares;
   
   -- Find the destination of a chain
   function Get_Destination(N : Integer) return Integer is
      Current : Integer := N;
      Path : array (1 .. 100) of Integer := (others => 0);
      Path_Length : Integer := 0;
   begin
      while Current /= 1 and Current /= 89 loop
         if Memo(Current) /= 0 then
            -- Return the computed result
            return Memo(Current);
         end if;
         
         Path_Length := Path_Length + 1;
         Path(Path_Length) := Current;
         
         Current := Sum_Of_Squares(Current);
      end loop;
      
      -- We've reached either 1 or 89
      declare
         Final_Result : Integer := Current;
      begin
         -- Memoize all numbers in the path
         for I in 1 .. Path_Length loop
            if Memo(Path(I)) = 0 then
               Memo(Path(I)) := Final_Result;
            end if;
         end loop;
         
         return Final_Result;
      end;
   end Get_Destination;
   
begin
   -- Base cases
   Memo(1) := 1;
   Memo(89) := 89;
   
   declare
      Count : Integer := 0;
   begin
      for I in 2 .. MAX_NUMBER - 1 loop
         if Get_Destination(I) = 89 then
            Count := Count + 1;
         end if;
      end loop;
      
      Put_Line("Answer: " & Integer'Image(Count));
   end;
   
end Euler_92;
```
