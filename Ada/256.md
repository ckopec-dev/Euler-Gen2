      A, B, C : Integer;
   end record;

   function Is_Integer_Area(T : Triangle) return Boolean is
      Perimeter : constant Integer := T.A + T.B + T.C;
      S         : constant Float := Float(Perimeter) / 2.0;
      Area_Sq   : constant Float := S * (S - Float(T.A)) * (S - Float(T.B)) * (S - Float(T.C));
      
      -- Check if area is a positive integer
   begin
      return Area_Sq > 0.0 and then

      return Integer(Sqrt(Area_Sq));

                  end if;
