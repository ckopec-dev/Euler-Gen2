with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_32 is
   
   -- Function to check if a string contains digits 1-9 exactly once (no zeros)
   function Is_Pandigital(S : String) return Boolean is
      Digit_Count : array (1..9) of Natural := (others => 0);
      I : Integer;
   begin
      -- Must be exactly 9 characters
      if S'Length /= 9 then
         return False;
      end if;
      
      -- Count each digit
      for I in S'Range loop
         declare
            Digit : Natural := Character'Pos(S(I)) - Character'Pos('0');
         begin
            if Digit = 0 or else Digit > 9 then
               return False;  -- Invalid digit
            end if;
            Digit_Count(Digit) := Digit_Count(Digit) + 1;
         end;
      end loop;
      
      -- Check that each digit 1-9 appears exactly once
      for I in 1..9 loop
         if Digit_Count(I) /= 1 then
            return False;
         end if;
      end loop;
      
      return True;
   end Is_Pandigital;
   
   -- Convert integer to string (simple version)
   function Int_To_String(N : Integer) return String is
      Temp : Integer := N;
      Result : String(1..10);  -- Enough space for any reasonable integer
      Len : Integer := 0;
   begin
      if N = 0 then
         return "0";
      end if;
      
      while Temp > 0 loop
         Len := Len + 1;
         Result(Len) := Character'Val(Temp mod 10 + Character'Pos('0'));
         Temp := Temp / 10;
      end loop;
      
      -- Reverse the string (since we built it backwards)
      declare
         Reversed : String(1..Len);
      begin
         for I in 1..Len loop
            Reversed(I) := Result(Len - I + 1);
         end loop;
         return Reversed;
      end;
   end Int_To_String;
   
   -- Check if A * B = C forms a pandigital equation (concatenated)
   function Is_Valid_Equation(A, B : Integer) return Boolean is
      Product : Integer := A * B;
      Concat : String(1..9);
      Pos : Integer := 0;
   begin
      -- Concatenate A, B, and Product as strings
      declare
         A_Str : String := Int_To_String(A);
         B_Str : String := Int_To_String(B);
         P_Str : String := Int_To_String(Product);
      begin
         -- Put them together
         for I in A_Str'Range loop
            Pos := Pos + 1;
            Concat(Pos) := A_Str(I);
         end loop;
         
         for I in B_Str'Range loop
            Pos := Pos + 1;
            Concat(Pos) := B_Str(I);
         end loop;
         
         for I in P_Str'Range loop
            Pos := Pos + 1;
            Concat(Pos) := P_Str(I);
         end loop;
         
         return Is_Pandigital(Concat);
      end;
   end Is_Valid_Equation;
   
   -- Main solution
   type Product_Set is array (Integer range 1..99999) of Boolean;
   Products : Product_Set := (others => False);
   Total_Sum : Integer := 0;
   
begin
   -- We need to find all combinations where A * B = C and the concatenation 
   -- of A, B, C uses each digit 1-9 exactly once
   
   -- The key insight: if we have 3 numbers with a total of 9 digits,
   -- and A * B = C, then:
   -- - A can be at most 4 digits (since 9999 * 9999 = 99980001, 8 digits)
   -- - B can be at most 4 digits
   -- - C can be at most 4 digits
   
   for A in 1..9999 loop
      exit when A > 999;
      for B in A..9999 loop
         declare
            Product : Integer := A * B;
         begin
            if Product > 9999 then
               exit;  -- No point continuing, product too large
            end if;
            
            if Is_Valid_Equation(A, B) then
               Products(Product) := True;
            end if;
         end;
      end loop;
   end loop;
   
   -- Sum all valid products
   for I in 1..99999 loop
      if Products(I) then
         Total_Sum := Total_Sum + I;
      end if;
   end loop;
   
   Put_Line("Sum of all pandigital products: " & Integer'Image(Total_Sum));
   
end Euler_32;
