# Euler Problem 279 Solution in Ada

This problem asks us to find the number of integer-sided triangles with perimeter 1000000 that are "almost right triangles" - meaning they satisfy the Pythagorean theorem with a small error.

Let me break down the approach:

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_279 is
   type Triangle is record
      A, B, C : Integer;
   end record;
   
   function Is_Right_Triangle(T : Triangle) return Boolean is
   begin
      return T.A * T.A + T.B * T.B = T.C * T.C;
   end Is_Right_Triangle;
   
   function Is_Almost_Right_Triangle(T : Triangle) return Boolean is
   begin
      -- Check if it's a Pythagorean triple with small error
      return abs (T.A * T.A + T.B * T.B - T.C * T.C) <= 1;
   end Is_Almost_Right_Triangle;
   
   function Count_Almost_Right_Triangles(P : Integer) return Integer is
      Count : Integer := 0;
      A, B, C : Integer;
   begin
      -- For a triangle with perimeter P, we have A + B + C = P
      -- Also, for an almost right triangle, |A² + B² - C²| <= 1
      for A in 1 .. P/3 loop
         for B in A .. (P - A)/2 loop
            C := P - A - B;
            
            -- Check if this forms a valid triangle
            if A + B > C and then A + C > B and then B + C > A then
               -- Check if it's an almost right triangle
               if abs (A * A + B * B - C * C) <= 1 then
                  Count := Count + 1;
               end if;
            end if;
         end loop;
      end loop;
      
      return Count;
   end Count_Almost_Right_Triangles;
   
begin
   -- For the actual problem, we need to be more efficient
   -- Let's use a mathematical approach
   
   declare
      Perimeter : constant Integer := 1000000;
      Result    : Integer := 0;
      A, B, C   : Integer;
   begin
      -- Mathematical optimization: for perimeter P and triangle with sides a,b,c
      -- We know a + b + c = P and |a² + b² - c²| <= 1
      -- This means c² = a² + b² ± k where k ∈ {0,1}
      
      for A in 1 .. Perimeter/3 loop
         for B in A .. (Perimeter - A)/2 loop
            C := Perimeter - A - B;
            
            if C > 0 and then A + B > C and then A + C > B and then B + C > A then
               declare
                  Diff : constant Integer := abs(A * A + B * B - C * C);
               begin
                  if Diff <= 1 then
                     Result := Result + 1;
                  end if;
               end;
            end if;
         end loop;
      end loop;
```
