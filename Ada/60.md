   function Is_Prime(N : Natural) return Boolean is
   begin
      if N < 2 then
         return False;
      elsif N = 2 then
         return True;
      elsif N mod 2 = 0 then
         return False;
      else
         for I in 3..Natural'Sqrt(N) loop
            if N mod I = 0 then
               return False;
            end if;
         end loop;
         return True;
      end if;
   end Is_Prime;

            for J in I*I..Max_Prime loop
               if J mod I = 0 then

               end if;
            end loop;
         end if;
      end loop;
      
      Prime_Count := 0;
      for I in 2..Max_Prime loop
         if Primes(I) then
            Prime_Count := Prime_Count + 1;
            Prime_List(Prime_Count) := I;
         end if;
      end loop;
   end Generate_Primes;

   function Concatenate_Strings(A, B : Natural) return Natural is
      Temp : String(1..20);
      Length : Integer;
   begin

      declare
         S_A : constant String := Integer'Image(A);
         S_B : constant String := Integer'Image(B);
      begin
         if S_A'Length > 1 then
            Temp(1..S_A'Length-1) := S_A(2..S_A'Length);
            Length := S_A'Length - 1;
         else
            Temp(1) := ' ';
            Length := 0;
         end if;
         Temp(Length+1..Length+S_B'Length-1) := S_B(2..S_B'Length);
         Length := Length + S_B'Length - 1;
         return Natural'Value(Temp(1..Length));
      end;
   end Concatenate_Strings;
   
   -- Check if two primes form a prime pair

      C1 : constant Natural := Concatenate_Strings(P1, P2);
      C2 : constant Natural := Concatenate_Strings(P2, P1);
   begin
      return Is_Prime(C1) and Is_Prime(C2);
   end Is_Prime_Pair;
   
   -- Check if a list of primes forms a valid set (all pairs are prime)

                        end;
                     end if;
                  end loop;
               end loop;
            end loop;
         end loop;
      end loop;
      
      return Min_Sum;
   end Find_Min_Sum;
