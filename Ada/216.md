# Euler Problem 216 Solution in Ada

This problem asks us to find the sum of all primes of the form 2n² - 1 where n > 0 and 2n² - 1 is prime, up to a certain limit.

Let me break down the approach:
1. Generate numbers of the form 2n² - 1
2. Check if each number is prime
3. Sum all such primes that are less than or equal to 10⁸

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Long_Long_Integer_Text_IO; use Ada.Long_Long_Integer_Text_IO;

procedure Euler_216 is
   
   type Prime_Array is array (Positive range <>) of Boolean;
   
   -- Function to check if a number is prime using trial division
   function Is_Prime(N : Long_Long_Integer) return Boolean is
      Root : Long_Long_Integer;
   begin
      if N < 2 then
         return False;
      elsif N = 2 then
         return True;
      elsif N mod 2 = 0 then
         return False;
      else
         Root := Long_Long_Integer(Sqrt(Float(N)));
         for I in 3 .. Root loop
            if N mod I = 0 then
               return False;
            end if;
         end loop;
         return True;
      end if;
   end Is_Prime;
   
   -- Function to calculate 2*n^2 - 1
   function F(n : Positive) return Long_Long_Integer is
   begin
      return 2 * Long_Long_Integer(n) * Long_Long_Integer(n) - 1;
   end F;
   
   -- Main computation
   Sum : Long_Long_Integer := 0;
   N : Positive := 1;
   Current_Value : Long_Long_Integer;
   Limit : constant Long_Long_Integer := 10_000_000_000; -- 10^10
   
begin
   loop
      Current_Value := F(N);
      
      -- Stop if we exceed our limit
      exit when Current_Value > Limit;
      
      -- Check if this number is prime
      if Is_Prime(Current_Value) then
         Sum := Sum + Current_Value;
         Put_Line("Found prime: " & Current_Value'Image);
      end if;
      
      N := N + 1;
   end loop;
   
   Put_Line("Sum of all primes of the form 2n² - 1: ");
   Put(Sum);
   New_Line;
   
end Euler_216;
```
