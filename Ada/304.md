First, let me understand what Euler Problem 304 asks:
- Find the sum of all Fibonacci numbers F(n) where n is a prime number
- The result should be taken modulo 10^9 + 7
- We need to find the first 10^4 such Fibonacci numbers

## Analysis
1. Generate Fibonacci numbers
2. Check if the index is prime
3. Sum those Fibonacci numbers at prime indices
4. Apply modulo operation

with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_304 is
   MODULUS : constant := 1_000_000_007;
   MAX_N   : constant := 100_000;
   
   type Prime_Array is array (1..MAX_N) of Boolean;
   type Fibonacci_Array is array (0..MAX_N) of Long_Long_Integer;
   
   -- Function to check if a number is prime
   function Is_Prime(N : Long_Long_Integer) return Boolean is
      use Long_Long_Integer;
   begin
      if N < 2 then
         return False;
      elsif N = 2 then
         return True;
      elsif N mod 2 = 0 then
         return False;
      else
         for I in 3..Long_Long_Integer(Sqrt(Float(N))) by 2 loop
            if N mod I = 0 then
               return False;
            end if;
         end loop;
         return True;
      end if;
   end Is_Prime;
   
   -- Sieve of Eratosthenes to find primes up to MAX_N
   function Get_Primes(Max : Long_Long_Integer) return Prime_Array is
      P : Prime_Array := (others => True);
   begin
      P(0) := False;
      P(1) := False;
      
      for I in 2..Long_Long_Integer(Sqrt(Float(Max))) loop
         if P(I) then
            for J in I*I..Max by I loop
               P(J) := False;
            end loop;
         end if;
      end loop;
      
      return P;
   end Get_Primes;
   
   -- Generate Fibonacci numbers up to index MAX_N
   function Generate_Fibonacci(Max : Long_Long_Integer) return Fibonacci_Array is
      F : Fibonacci_Array;
   begin
      F(0) := 0;
      F(1) := 1;
      
      for I in 2..Max loop
         F(I) := (F(I-1) + F(I-2)) mod MODULUS;
      end loop;
      
      return F;
   end Generate_Fibonacci;
   
   -- Main computation
   Primes : Prime_Array := Get_Primes(MAX_N);
   Fib    : Fibonacci_Array := Generate_Fibonacci(MAX_N);
   Sum    : Long_Long_Integer := 0;
   Count  : Natural := 0;
   
begin
   -- Find sum of F(n) where n is prime
   for I in 2..MAX_N loop
      if Primes(I) then
         Sum := (Sum + Fib(I)) mod MODULUS;
         Count := Count + 1;
         
         -- Stop after we have enough terms
         if Count >= 10_000 then
            exit;
         end if;
      end if;
   end loop;
   
   Put("Answer: ");
   Put(Sum, 0);
   New_Line;
   
end Euler_304;
