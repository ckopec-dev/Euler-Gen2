# Euler Problem 175 Solution in Ada

This problem asks us to find the number of ways to write a given fraction as a sum of unit fractions with distinct denominators.

with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_175 is
   type Fraction is record
      Num : Integer;
      Den : Integer;
   end record;
   
   function "+"(F1, F2 : Fraction) return Fraction is
   begin
      return (Num => F1.Num * F2.Den + F2.Num * F1.Den,
              Den => F1.Den * F2.Den);
   end "+";
   
   function "-"(F1, F2 : Fraction) return Fraction is
   begin
      return (Num => F1.Num * F2.Den - F2.Num * F1.Den,
              Den => F1.Den * F2.Den);
   end "-";
   
   function "*" (F1, F2 : Fraction) return Fraction is
   begin
      return (Num => F1.Num * F2.Num,
              Den => F1.Den * F2.Den);
   end "*";
   
   function "/" (F1, F2 : Fraction) return Fraction is
   begin
      return (Num => F1.Num * F2.Den,
              Den => F1.Den * F2.Num);
   end "/";
   
   function "="(F1, F2 : Fraction) return Boolean is
   begin
      return F1.Num * F2.Den = F2.Num * F1.Den;
   end "=";
   
   function "<"(F1, F2 : Fraction) return Boolean is
   begin
      return F1.Num * F2.Den < F2.Num * F1.Den;
   end "<";
   
   procedure Reduce(F : in out Fraction) is
      function GCD(A, B : Integer) return Integer is
      begin
         if B = 0 then
            return A;
         else
            return GCD(B, A mod B);
         end if;
      end GCD;
      G : constant Integer := GCD(Abs(F.Num), Abs(F.Den));
   begin
      if G /= 0 then
         F.Num := F.Num / G;
         F.Den := F.Den / G;
      end if;
      
      -- Ensure denominator is positive
      if F.Den < 0 then
         F.Num := -F.Num;
         F.Den := -F.Den;
      end if;
   end Reduce;
   
   function Is_Unit_Fraction(F : Fraction) return Boolean is
   begin
      return F.Num = 1;
   end Is_Unit_Fraction;
   
   function Find_Smallest_Unit_Fraction(F : Fraction) return Fraction is
      -- Find the smallest unit fraction that can be subtracted from F
      -- This is 1 / ceil(1/F)
   begin
      if F.Den mod F.Num = 0 then
         -- If F is already a unit fraction
         return F;
      else
         -- Find ceiling of F.Den / F.Num
         declare
            Ceiling : constant Integer := (F.Den + F.Num - 1) / F.Num;
         begin
            return (Num => 1, Den => Ceiling);
         end;
      end if;
   end Find_Smallest_Unit_Fraction;
   
   function Egyptian_Representation(F : Fraction) return String is
      Temp_F : Fraction := F;
      Result : String(1..1000) := (others => ' ');
      Pos : Integer := 1;
      First : Boolean := True;
   begin
      loop
         if Is_Unit_Fraction(Temp_F) then
            if not First then
               Result(Pos) := '+';
               Pos := Pos + 1;
            end if;
            
            Result(Pos..Pos+1) := "1/";
            Pos := Pos + 2;
            Put_Integer(Result(Pos..Pos+5), Temp_F.Den);
            Pos := Pos + Integer'Image(Temp_F.Den).Length - 1;
            exit;
         else
            if not First then
               Result(Pos) := '+';
               Pos := Pos + 1;
            end if;
            
            declare
               Unit_Frac : constant Fraction := Find_Smallest_Unit_Fraction(Temp_F);
            begin
               Result(Pos..Pos+1) := "1/";
               Pos := Pos + 2;
               Put_Integer(Result(Pos..Pos+5), Unit_Frac.Den);
               Pos := Pos + Integer'Image(Unit_Frac.Den).Length - 1;
               
               Temp_F := Temp_F - Unit_Frac;
               Reduce(Temp_F);
               First := False;
            end;
         end if;
      end loop;
      
      return Result(1..Pos-1);
   end Egyptian_Representation;
   
   function Count_Ways(F : Fraction) return Integer is
      -- This is a recursive approach to count the number of ways
      -- to represent F as sum of unit fractions
      function Count_Recursive(Frac : Fraction; Max_Den : Integer) return Integer is
      begin
         if Frac.Num = 0 then
            return 1;
         elsif Frac.Num > 0 and Frac.Den < 0 then
            -- If numerator is positive but denominator is negative
            return 0;
         elsif Frac.Num < 0 then
            -- Negative fraction - this should not happen in our case
            return 0;
         elsif Frac.Num = 1 and Frac.Den <= Max_Den then
            -- Simple case: unit fraction with acceptable denominator
            return 1;
         else
            declare
               Count : Integer := 0;
               Den : Integer := Max_Den + 1;
            begin
               loop
                  if Den > Frac.Den then
                     exit;
                  end if;
                  
                  -- Check if we can subtract 1/Den from Frac
                  declare
                     New_Frac : Fraction := Frac - (Num => 1, Den => Den);
                  begin
                     Reduce(New_Frac);
                     if New_Frac.Num >= 0 then
                        Count := Count + Count_Recursive(New_Frac, Den);
                     end if;
                  end;
                  
                  Den := Den + 1;
               end loop;
               
               return Count;
            end;
         end if;
      end Count_Recursive;
   begin
      return Count_Recursive(F, F.Den);
   end Count_Ways;
   
   function Solve_Problem(N : Integer) return Integer is
      -- For the specific case mentioned in the problem (2/3)
      F : constant Fraction := (Num => 2, Den => 3);
   begin
      return Count_Ways(F);
   end Solve_Problem;
