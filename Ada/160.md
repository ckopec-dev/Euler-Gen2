# Euler Problem 160 Solution in Ada

This problem asks for the last 5 digits of 1000! (factorial of 1000).

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_160 is
   type Long_Integer is range 0..2**63-1;
   
   function Mod_Mult(A, B, M : Long_Integer) return Long_Integer is
      -- Computes (A * B) mod M efficiently to avoid overflow
   begin
      if A = 0 or B = 0 then
         return 0;
      end if;
      
      if A > M then
         A := A rem M;
      end if;
      
      if B > M then
         B := B rem M;
      end if;
      
      return (A * B) mod M;
   end Mod_Mult;
   
   function Factorial_Last_Digits(N : Long_Integer; Modulus : Long_Integer) 
      return Long_Integer is
      Result : Long_Integer := 1;
      I : Long_Integer := 2;
   begin
      while I <= N loop
         -- Skip multiples of 2 and 5 (they contribute trailing zeros)
         if I mod 2 /= 0 and I mod 5 /= 0 then
            Result := Mod_Mult(Result, I, Modulus);
         end if;
         I := I + 1;
      end loop;
      return Result;
   end Factorial_Last_Digits;
   
   function Count_Factors(N : Long_Integer; P : Integer) return Long_Integer is
      -- Counts how many times prime P divides N!
      Count : Long_Integer := 0;
      Power : Long_Integer := P;
   begin
      while Power <= N loop
         Count := Count + N / Power;
         Power := Power * P;
      end loop;
      return Count;
   end Count_Factors;
   
   function Solve(N : Long_Integer) return Long_Integer is
      -- Compute last 5 digits of N!
      -- We need to compute N! mod 100000 = N! mod (2^5 * 5^5)
      
      -- Number of factors of 2 and 5 in N!
      F2 : Long_Integer := Count_Factors(N, 2);
      F5 : Long_Integer := Count_Factors(N, 5);
      
      -- We need to compute the result mod 100000
      Modulus : constant Long_Integer := 100000;
      
      -- The number of trailing zeros is min(F2, F5)
      Trailing_Zeros : Long_Integer := F5; -- Since there are more factors of 2 than 5
      
      -- Compute N! without the trailing zeros (divided by 10^Trailing_Zeros)
      Result : Long_Integer := 1;
      I : Long_Integer := 2;
      
   begin
      while I <= N loop
         -- Count how many times 2 and 5 divide I
         local_2 : Long_Integer := 0;
         local_5 : Long_Integer := 0;
         J : Long_Integer := I;
         
         while J rem 2 = 0 loop
            local_2 := local_2 + 1;
            J := J / 2;
         end loop;
         
         while J rem 5 = 0 loop
            local_5 := local_5 + 1;
            J := J / 5;
         end loop;
         
         -- Only multiply by I if it doesn't contribute trailing zeros
         if local_2 < Trailing_Zeros or local_5 < Trailing_Zeros then
            Result := Mod_Mult(Result, I, Modulus);
         end if;
         
         I := I + 1;
      end loop;
      
      return Result;
   end Solve;
   
   function Fast_Factorial(N : Long_Integer) return Long_Integer is
      -- A more efficient approach using Legendre's formula and modular arithmetic
      Modulus : constant Long_Integer := 100000;
      Result : Long_Integer := 1;
      I : Long_Integer := 2;
      
   begin
      while I <= N loop
         -- For each number, remove factors of 2 and 5 and multiply by remaining part
         local_2 : Long_Integer := 0;
         local_5 : Long_Integer := 0;
         temp : Long_Integer := I;
         
         while temp rem 2 = 0 loop
            local_2 := local_2 + 1;
            temp := temp / 2;
         end loop;
         
         while temp rem 5 = 0 loop
            local_5 := local_5 + 1;
            temp := temp / 5;
         end loop;
         
         -- Multiply by the remaining part (mod 100000)
         Result := Mod_Mult(Result, temp, Modulus);
         I := I + 1;
      end loop;
      
      return Result;
   end Fast_Factorial;
   
begin
   -- The direct approach for small numbers works fine
   -- But we'll compute it more carefully
```

```ada
with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_160 is
   type Large_Integer is range 0..2**63-1;
   
   function Mod_Multiply(A, B, M : Large_Integer) return Large_Integer is
   begin
      if A = 0 or B = 0 then
         return 0;
      end if;
      
      -- Compute (A * B) mod M using modular arithmetic to avoid overflow
      return (A * (B rem M)) rem M;
   end Mod_Multiply;
   
   function Solve_160(N : Large_Integer) return Large_Integer is
      Result : Large_Integer := 1;
      I : Large_Integer := 2;
      Modulus : constant Large_Integer := 100000; -- Last 5 digits
   begin
      while I <= N loop
         -- Remove all factors of 2 and 5 from I, then multiply result by remaining value
         declare
            Temp : Large_Integer := I;
            Reduced : Large_Integer := 1;
         begin
            -- Remove factors of 2
            while Temp rem 2 = 0 loop
               Temp := Temp / 2;
            end loop;
            
            -- Remove factors of 5  
            while Temp rem 5 = 0 loop
               Temp := Temp / 5;
            end loop;
            
            -- Multiply result by the reduced value
            Result := Mod_Multiply(Result, Temp, Modulus);
         end;
         
         I := I + 1;
      end loop;
      
      return Result;
   end Solve_160;
   
begin
   Put_Line("The last 5 digits of 1000! are: " & 
            Integer'Image(Solve_160(1000)));
end Euler_160;
```
