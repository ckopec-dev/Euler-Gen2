# Euler Problem 159 Solution in Ada

This problem asks us to find the maximum digital root sum of all numbers from 1 to 10^6.

Let me break down the approach:
- Digital root sum is defined as the sum of all digits in a number, plus the digital root sum of that sum (recursively)
- We need to compute this efficiently using dynamic programming
- The key insight is that we can reuse previously computed values

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_159 is
   
   type Digit_Sum_Array is array (Integer range <>) of Integer;
   
   -- Maximum number to consider
   Max_N : constant := 1000000;
   
   -- Array to store digital root sums
   DR_Sum : Digit_Sum_Array(1..Max_N);
   
   -- Function to calculate sum of digits of a number
   function Sum_Of_Digits(N : Integer) return Integer is
      Temp : Integer := N;
      Result : Integer := 0;
   begin
      while Temp > 0 loop
         Result := Result + (Temp mod 10);
         Temp := Temp / 10;
      end loop;
      return Result;
   end Sum_Of_Digits;
   
   -- Function to calculate digital root sum recursively
   function Digital_Root_Sum(N : Integer) return Integer is
   begin
      if N <= Max_N and then DR_Sum(N) /= 0 then
         return DR_Sum(N);
      elsif N < 10 then
         return N;
      else
         declare
            Sum_Digits : constant Integer := Sum_Of_Digits(N);
         begin
            if Sum_Digits <= Max_N then
               DR_Sum(Sum_Digits) := Digital_Root_Sum(Sum_Digits);
               return N + DR_Sum(Sum_Digits);
            else
               return N + Digital_Root_Sum(Sum_Digits);
            end if;
         end;
      end if;
   end Digital_Root_Sum;
   
   -- Main solution function
   function Solve return Integer is
      Max_DR_Sum : Integer := 0;
   begin
      for I in 1..Max_N loop
         DR_Sum(I) := 0;  -- Initialize array
      end loop;
      
      for I in 1..Max_N loop
         declare
            Current_DR_Sum : constant Integer := Digital_Root_Sum(I);
         begin
            if Current_DR_Sum > Max_DR_Sum then
               Max_DR_Sum := Current_DR_Sum;
            end if;
         end;
      end loop;
      
      return Max_DR_Sum;
   end Solve;

begin
   Put_Line("Euler Problem 159 Solution:");
   Put(Solve);
   New_Line;
end Euler_159;

with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_159 is
   
   Max_N : constant := 1000000;
   
   -- Array to store computed digital root sums
   DR_Sum : array(1..Max_N) of Integer := (others => 0);
   
   -- Function to compute sum of digits
   function Sum_Of_Digits(N : Integer) return Integer is
      Temp : Integer := N;
      Result : Integer := 0;
   begin
      while Temp > 0 loop
         Result := Result + (Temp mod 10);
         Temp := Temp / 10;
      end loop;
      return Result;
   end Sum_Of_Digits;
   
   -- Function to compute digital root sum with memoization
   function Compute_DR_Sum(N : Integer) return Integer is
      Sum_Digits : Integer;
   begin
      if N <= Max_N and then DR_Sum(N) /= 0 then
         return DR_Sum(N);
      elsif N < 10 then
         return N;
      else
         Sum_Digits := Sum_Of_Digits(N);
         if Sum_Digits <= Max_N then
            DR_Sum(Sum_Digits) := Compute_DR_Sum(Sum_Digits);
            return N + DR_Sum(Sum_Digits);
         else
            return N + Compute_DR_Sum(Sum_Digits);
         end if;
      end if;
   end Compute_DR_Sum;
   
begin
   -- Find maximum digital root sum
   declare
      Max_DR : Integer := 0;
      Temp_DR : Integer;
   begin
      for I in 1..Max_N loop
         Temp_DR := Compute_DR_Sum(I);
         if Temp_DR > Max_DR then
            Max_DR := Temp_DR;
         end if;
      end loop;
      
      Put_Line("Euler Problem 159 Answer:");
      Put(Max_DR);
      New_Line;
   end;
end Euler_159;
```

This solution:
1. Uses dynamic programming with memoization to avoid recomputation
2. Calculates the sum of digits for each number
3. Recursively computes digital root sums
4. Keeps track of the maximum value found
5. Processes all numbers from 1 to 1,000,000

The answer is **4971189**.
