with Ada.Text_IO;
with Ada.Integer_Text_IO;

procedure Euler_252 is

   -- Calculate combination C(n,k) = n! / (k!(n-k)!)
   function C(n, k : Integer) return Long_Long_Integer is
   begin
      if k > n or k < 0 then
         return 0;
      elsif k = 0 or k = n then
         return 1;
      else
         declare
            result : Long_Long_Integer := 1;
         begin
            for i in 1..k loop
               result := result * (n - i + 1) / i;
            end loop;
            return result;
         end;
      end if;
   end C;

   -- Calculate the number of triangles in a triangular grid with n rows
   function Triangle_Count(n : Integer) return Long_Long_Integer is
      -- Total points in triangular grid with n rows:
      -- 1 + 2 + 3 + ... + n = n*(n+1)/2
      total_points : constant Long_Long_Integer := (Long_Long_Integer(n) * Long_Long_Integer(n + 1)) / 2;
      
      -- Total ways to choose 3 points from all points
      total_triangles : constant Long_Long_Integer := C(Integer(total_points), 3);
      
      -- Subtract collinear triplets
      collinear_count : Long_Long_Integer := 0;
      
   begin
      -- For a triangular grid, we need to count:
      -- 1. Horizontal lines (each line i has i points)
      for i in 1..n loop
         if i >= 3 then
            collinear_count := collinear_count + C(i, 3);
         end if;
      end loop;
      
      -- 2. Left diagonals (similar to horizontal)
      for i in 1..n loop
         if i >= 3 then
            collinear_count := collinear_count + C(i, 3);
         end if;
      end loop;
      
      -- 3. Right diagonals (different count for each diagonal)
      for i in 2..n loop
         if i >= 3 then
            collinear_count := collinear_count + C(i, 3);
         end if;
      end loop;
      
      return total_triangles - collinear_count;
   end Triangle_Count;

begin
   Ada.Text_IO.Put_Line("Project Euler Problem 252");
   Ada.Text_IO.Put_Line("=========================");
   
   declare
      result : constant Long_Long_Integer := Triangle_Count(20);
   begin
      Ada.Text_IO.Put("Number of triangles in triangular grid: ");
      Ada.Integer_Text_IO.Put(result, Width => 15);
      Ada.Text_IO.New_Line;
   end;
   
end Euler_252;
