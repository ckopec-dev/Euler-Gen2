# Euler Problem 82 Solution in Ada

This solution finds the minimal path sum from the left column to the right column in a matrix, where movement is allowed up, down, and right (but not left).

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;
with Ada.Containers.Vectors;

procedure Euler_82 is
   
   -- Matrix dimensions
   Width : constant := 80;
   Height : constant := 80;
   
   -- Matrix type definition
   type Matrix_Row is array (1..Width) of Integer;
   type Matrix is array (1..Height) of Matrix_Row;
   
   -- Read matrix from file
   procedure Read_Matrix(M : out Matrix) is
      File : File_Type;
      Line : String(1..1000);
      Last : Natural;
      I : Integer := 1;
   begin
      Open(File, In_File, "matrix.txt");
      while not End_Of_File(File) loop
         Get_Line(File, Line, Last);
         declare
            J : Integer := 1;
            Start : Integer := 1;
         begin
            for K in 1..Width loop
               while Line(Start) = ' ' loop
                  Start := Start + 1;
               end loop;
               declare
                  Num : Integer := 0;
                  Pos : Integer := Start;
               begin
                  while Pos <= Last and then Line(Pos) /= ',' loop
                     Num := Num * 10 + (Integer'Value(Line(Pos..Pos)) - Integer'Value("0"));
                     Pos := Pos + 1;
                  end loop;
                  M(I)(J) := Num;
                  J := J + 1;
                  Start := Pos + 1;
               end;
            end loop;
         end;
         I := I + 1;
      end loop;
      Close(File);
   end Read_Matrix;
   
   -- Dynamic programming approach to find minimal path sum
   procedure Solve(M : Matrix; Min_Sum : out Integer) is
      type DP_Row is array (1..Width) of Integer;
      type DP_Matrix is array (1..Height) of DP_Row;
      
      -- Initialize DP matrix with infinity
      DP : DP_Matrix;
      Infinity : constant := 2**30;
      
      -- Initialize all values to infinity
      procedure Init_DP is
      begin
         for I in 1..Height loop
            for J in 1..Width loop
               DP(I)(J) := Infinity;
            end loop;
         end loop;
      end Init_DP;
      
   begin
      Init_DP;
      
      -- Initialize first column of DP matrix
      for I in 1..Height loop
         DP(I)(1) := M(I)(1);
      end loop;
      
      -- Fill the DP matrix column by column
      for J in 2..Width loop
         -- First, fill from top to bottom (upward movement)
         for I in 1..Height loop
            DP(I)(J) := Min(DP(I)(J), DP(I)(J-1) + M(I)(J));
            
            -- Check upward movement
            for K in 1..I-1 loop
               DP(K)(J) := Min(DP(K)(J), DP(I)(J-1) + M(I)(J) + 
                               (if K < I then 
                                   (for all L in K+1..I-1 => M(L)(J)) 
                                else 0));
            end loop;
         end loop;
         
         -- Then fill from bottom to top (downward movement)
         for I in reverse 1..Height loop
            -- Check downward movement
            for K in I+1..Height loop
               DP(K)(J) := Min(DP(K)(J), DP(I)(J-1) + M(I)(J) + 
                               (if K > I then 
                                   (for all L in I+1..K-1 => M(L)(J)) 
                                else 0));
            end loop;
         end loop;
      end loop;
      
      -- Find minimum value in last column
      Min_Sum := DP(1)(Width);
      for I in 2..Height loop
         Min_Sum := Min(Min_Sum, DP(I)(Width));
      end loop;
   end Solve;

      for J in 2..Width loop
         -- Process all rows from top to bottom (left to right)
         for I in 1..Height loop
            -- From the same row in previous column
            DP(I)(J) := Min(DP(I)(J), DP(I)(J-1) + M(I)(J));
            
            -- Check upward paths
            for K in 1..I-1 loop
               DP(K)(J) := Min(DP(K)(J), DP(I)(J-1) + 
                               (for all L in K+1..I => M(L)(J)));
            end loop;
         end loop;
         
         -- Process all rows from bottom to top (reverse order)
         for I in reverse 1..Height loop
            -- Check downward paths
            for K in I+1..Height loop
               DP(K)(J) := Min(DP(K)(J), DP(I)(J-1) +

            end loop;
         end loop;
      end loop;
      
      -- Find minimum value in last column
      Min_Sum := DP(1)(Width);
      for I in 2..Height loop
         Min_Sum := Min(Min_Sum, DP(I)(Width));
      end loop;
   end Solve_Final;
   
   M : Matrix;
   Result : Integer;
begin
   Read_Matrix(M);
   Solve_Final(M, Result);
   Put("The minimal path sum is: ");
   Put(Result);
   New_Line;
end Euler_82;
```

This Ada solution uses dynamic programming to solve the problem efficiently:

1. **Matrix Reading**: Reads the 80×80 matrix from a file named "matrix.txt"
2. **Dynamic Programming Approach**: 
   - Initialize a DP table where `DP[i][j]` represents the minimum sum path to reach cell `(i,j)`
   - Fill the table column by column, considering movement from left (same row) or upward/downward paths
3. **Optimization**: Processes both forward and backward directions to ensure all possible paths are considered
4. **Result**: Finds the minimum value in the last column

The key insight is that we can move up, down, or right but not left, so we must consider all valid paths from previous columns to each cell.

**Note**: This solution assumes the matrix file "matrix.txt" exists in the working directory with the proper format containing comma-separated integers.
