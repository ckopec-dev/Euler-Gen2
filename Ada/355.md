      P.Clear;

         if Is_Prime (I) then

         if N mod (D * D) = 0 then
            return False;
         end if;
         D := D + 1;
      end loop;

begin

procedure Euler_355 is
   
   Limit : constant := 10**8;
   
   -- Check if a number is squarefree (not divisible by any perfect square > 1)
   function Is_Squarefree (N : Natural) return Boolean is

   begin
      while I * I <= N loop
         if N mod (I * I) = 0 then
            return False;
         end if;
         I := I + 1;
      end loop;
      return True;
   end Is_Squarefree;
   
   -- Main computation
   Sum : Natural := 0;
   Current : Natural := 3;  -- Start with 3 (first multiple of 3)
   
begin
   Put_Line ("Computing sum of squarefree numbers divisible by 3 and less than " &
             Natural'Image (Limit));
   
   while Current < Limit loop
      if Is_Squarefree (Current) then
         Sum := Sum + Current;
      end if;
      Current := Current + 3;  -- Move to next multiple of 3
   end loop;
   
   Put ("Sum = ");
   Put (Sum);

end Euler_355;

This Ada program computes the sum of all squarefree numbers that are divisible by 3 and less than 10^8. The approach:

1. **Squarefree Check**: For each number, we check if it's divisible by any perfect square greater than 1
2. **Multiple Generation**: We generate multiples of 3 (3, 6, 9, 12, ...) 
3. **Filtering**: Only include those that are squarefree in our sum

The algorithm works by:
- Starting with 3 (the first multiple of 3)
- Checking if each multiple is squarefree
- Adding to the sum if it is
- Incrementing by 3 to get the next multiple
- Continuing until we exceed the limit

The time complexity is reasonable for this problem size, as we're only checking multiples of 3 and using an efficient squarefree test.
