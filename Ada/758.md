## Problem Understanding

The problem asks for the number of ways to represent a given number as a sum of distinct Fibonacci numbers (where each Fibonacci number can be used at most once).

## Solution Approach

I'll use dynamic programming where:
- `dp[i]` = number of ways to represent number `i` as sum of distinct Fibonacci numbers
- I'll generate Fibonacci numbers up to the required limit
- For each Fibonacci number, I'll update the dp array backwards to avoid using the same Fibonacci number multiple times

with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_758 is
   type Long_Integer is range -2**63 .. 2**63-1;
   
   -- Maximum number we need to consider
   MAX_N : constant Long_Integer := 10**12;
   
   -- Fibonacci numbers up to MAX_N
   type Fibonacci_Array is array (1..100) of Long_Integer;
   Fib : Fibonacci_Array;
   Fib_Count : Natural := 0;
   
   -- DP array for counting representations
   type DP_Array is array (0..1000000) of Long_Integer;
   DP : DP_Array := (others => 0);
   
   -- Generate Fibonacci numbers up to MAX_N
   procedure Generate_Fibonacci is
      F1, F2 : Long_Integer := 1;
      Index : Natural := 1;
   begin
      Fib(Index) := F1;
      Index := Index + 1;
      Fib(Index) := F2;
      Index := Index + 1;
      
      while True loop
         declare
            Next : Long_Integer := F1 + F2;
         begin
            if Next > MAX_N then
               exit;
            end if;
            
            Fib(Index) := Next;
            Index := Index + 1;
            F1 := F2;
            F2 := Next;
         end;
      end loop;
      
      Fib_Count := Index - 1;
   end Generate_Fibonacci;
   
   -- Count representations using dynamic programming
   function Count_Representations(N : Long_Integer) return Long_Integer is
      Result : Long_Integer := 0;
   begin
      -- Initialize DP array
      for I in 0..1000000 loop
         DP(I) := 0;
      end loop;
      
      DP(0) := 1;  -- One way to make 0 (use nothing)
      
      -- For each Fibonacci number
      for I in 1..Fib_Count loop
         -- Update DP array from right to left to avoid using same Fibonacci multiple times
         for J in reverse Fib(I)..1000000 loop
            if J >= Fib(I) then
               DP(J) := DP(J) + DP(J - Fib(I));
            end if;
         end loop;
      end loop;
      
      return DP(N);
   end Count_Representations;
   
   -- Main computation
   function Solve return Long_Integer is
      N : constant Long_Integer := 10**12;
   begin
      Generate_Fibonacci;
      return Count_Representations(N);
   end Solve;
   
begin
   Put_Line("Euler Problem 758 Solution:");
   Put("Result: ");
   Put(Solve);
   New_Line;
end Euler_758;
