# Euler Problem 250 Solution in Ada

This problem asks for the last 16 digits of the sum of all Fibonacci numbers whose index is a perfect square.

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_250 is
   type Big_Integer is array (1..1000) of Integer;
   
   -- Function to multiply two big integers
   function "*" (Left, Right : Big_Integer) return Big_Integer is
      Result : Big_Integer := (others => 0);
      Carry  : Integer;
   begin
      for I in 1..1000 loop
         Carry := 0;
         for J in 1..1000 loop
            if I + J - 1 <= 1000 then
               Result(I + J - 1) := Result(I + J - 1) + Left(I) * Right(J) + Carry;
               Carry := Result(I + J - 1) / 10;
               Result(I + J - 1) := Result(I + J - 1) mod 10;
            end if;
         end loop;
      end loop;
      return Result;
   end "*";
   
   -- Function to add two big integers
   function "+" (Left, Right : Big_Integer) return Big_Integer is
      Result : Big_Integer := (others => 0);
      Carry  : Integer := 0;
   begin
      for I in 1..1000 loop
         Result(I) := Left(I) + Right(I) + Carry;
         Carry := Result(I) / 10;
         Result(I) := Result(I) mod 10;
      end loop;
      return Result;
   end "+";
   
   -- Function to compute power of a big integer
   function Power (Base : Big_Integer; Exp : Integer) return Big_Integer is
      Result : Big_Integer := (others => 0);
      Temp   : Big_Integer := Base;
   begin
      if Exp = 0 then
         Result(1) := 1;
         return Result;
      end if;
      
      Result(1) := 1;
      for I in 1..Exp loop
         Result := Result * Temp;
      end loop;
      return Result;
   end Power;
   
   -- Function to get last 16 digits of a big integer
   function Get_Last_16_Digits (Num : Big_Integer) return String is
      Result : String(1..16) := (others => '0');
      Index  : Integer := 16;
   begin
      for I in reverse 1..1000 loop
         if Num(I) /= 0 or Index < 16 then
            Result(Index) := Character'Val(Num(I) + 48);
            Index := Index - 1;
            if Index = 0 then
               return Result;
            end if;
         end if;
      end loop;
      return Result;
   end Get_Last_16_Digits;
   
   -- Fibonacci function to get the nth Fibonacci number
   function Fibonacci (N : Integer) return Big_Integer is
      F0, F1, Temp : Big_Integer := (others => 0);
   begin
      if N = 0 then
         F0(1) := 1;
         return F0;
      elsif N = 1 then
         F1(1) := 1;
         return F1;
      end if;
      
      F0(1) := 0; F1(1) := 1;
      for I in 2..N loop
         Temp := F0 + F1;
         F0 := F1;
         F1 := Temp;
      end loop;
      return F1;
   end Fibonacci;
   
   -- Main computation
   Sum : Big_Integer := (others => 0);
   N   : Integer := 1;
   Sqr : Integer := 1;
   Max_Index : constant := 10^6;  -- Reasonable upper bound
   
begin
   -- Loop through perfect squares and compute corresponding Fibonacci numbers
   while Sqr <= Max_Index loop
      declare
         F : Big_Integer;
      begin
         F := Fibonacci(Sqr);
         Sum := Sum + F;
      end;
      
      N := N + 1;
      Sqr := N * N;
   end loop;
   
   -- Output the last 16 digits
   Put_Line("Last 16 digits: " & Get_Last_16_Digits(Sum));
end Euler_250;
```

However, since implementing a full big integer arithmetic library is quite complex and not the main focus of this exercise, here's a more practical approach using modular arithmetic to solve this efficiently:

```ada
with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_250 is
   -- We need to compute sum of F(n) where n is a perfect square
   -- All Fibonacci numbers are computed modulo 10^16 for efficiency
   
   MODULUS : constant := 10**16;
   
   function Modulo (N : Long_Long_Integer) return Long_Long_Integer is
   begin
      return N mod MODULUS;
   end Modulo;
   
   -- Fibonacci computation with modular arithmetic
   function Fibonacci_Mod (Index : Integer) return Long_Long_Integer is
      F0, F1, Temp : Long_Long_Integer := 0;
      I : Integer;
   begin
      if Index = 0 then
         return 0;
      elsif Index = 1 then
         return 1;
      end if;
      
      F0 := 0; F1 := 1;
      for I in 2..Index loop
         Temp := Modulo(F0 + F1);
         F0 := F1;
         F1 := Temp;
      end loop;
      return F1;
   end Fibonacci_Mod;
   
   Sum : Long_Long_Integer := 0;
   N   : Integer := 1;
   Sqr : Integer := 1;
   
begin
   -- Compute sum of Fibonacci numbers at perfect square indices
   while Sqr <= 10**6 loop  -- Reasonable upper limit
      Sum := Modulo(Sum + Fibonacci_Mod(Sqr));
      N := N + 1;
      Sqr := N * N;
   end loop;
   
   Put_Line("Result: " & Long_Long_Integer'Image(Sum));
end Euler_250;
```

**Note**: This problem is computationally intensive and requires careful handling of large numbers. The exact solution would require implementing a full big integer library or using an existing one, as the Fibonacci numbers grow very quickly and exceed standard integer ranges even for moderate values of n.

The key insight is to:
1. Generate perfect squares (1², 2², 3², ...)
2. Compute corresponding Fibonacci numbers
3. Sum them up with modular arithmetic
4. Extract the last 16 digits
