# Euler Problem 262 Solution in Ada

I'll solve Project Euler Problem 262, which asks for the number of lattice points inside a circle with radius r and center at (0,0).

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_262 is
   type Point is record
      X, Y : Integer;
   end record;
   
   -- Function to check if a point is inside the circle
   function Is_Inside_Circle(P : Point; Radius : Integer) return Boolean is
   begin
      return (P.X * P.X + P.Y * P.Y) <= (Radius * Radius);
   end Is_Inside_Circle;
   
   -- Function to count lattice points in a circle
   function Count_Lattice_Points(Radius : Integer) return Integer is
      Count : Integer := 0;
      X, Y  : Integer;
   begin
      -- Check all possible lattice points within the bounding square
      for X in -Radius .. Radius loop
         for Y in -Radius .. Radius loop
            if Is_Inside_Circle((X, Y), Radius) then
               Count := Count + 1;
            end if;
         end loop;
      end loop;
      
      return Count;
   end Count_Lattice_Points;
   
   -- Function to calculate the exact formula for lattice points
   function Lattice_Count(Radius : Integer) return Integer is
      Count : Integer := 0;
      X, Y  : Integer;
      R2    : Integer := Radius * Radius;
   begin
      -- For each x coordinate in range
      for X in -Radius .. Radius loop
         -- Find maximum y such that x^2 + y^2 <= r^2
         -- This means y^2 <= r^2 - x^2, so |y| <= sqrt(r^2 - x^2)
         declare
            Y_Max : Integer := Integer(Real'(R2 - X * X))**0.5;
         begin
            for Y in -Y_Max .. Y_Max loop
               if (X * X + Y * Y) <= R2 then
                  Count := Count + 1;
               end if;
            end loop;
         end;
      end loop;
      
      return Count;
   end Lattice_Count;
   
begin
   -- Calculate for radius 100
   declare
      Radius : constant Integer := 100;
      Result : Integer;
   begin
      Result := Lattice_Count(Radius);
      Put("Number of lattice points inside circle with radius ");
      Put(Radius);
      Put(" is: ");
      Put(Result);
      New_Line;
   end;
   
   -- Also test with smaller values for verification
   Put_Line("Verification with smaller radii:");
   Put("Radius 1: ");
   Put(Lattice_Count(1));
   New_Line;
   
   Put("Radius 2: ");
   Put(Lattice_Count(2));
   New_Line;
   
   Put("Radius 3: ");
   Put(Lattice_Count(3));
   New_Line;
   
end Euler_262;
```

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Float_Text_IO; use Ada.Float_Text_IO;

procedure Euler_262 is
   
   -- Function to count lattice points inside a circle of radius r
   function Count_Lattice_Points(Radius : Integer) return Integer is
      Count : Integer := 0;
   begin
      for X in -Radius .. Radius loop
         for Y in -Radius .. Radius loop
            if X * X + Y * Y <= Radius * Radius then
               Count := Count + 1;
            end if;
         end loop;
      end loop;
      
      return Count;
   end Count_Lattice_Points;
   
   -- More efficient version using mathematical approach
   function Efficient_Count(Radius : Integer) return Integer is
      Count : Integer := 0;
   begin
      for X in -Radius .. Radius loop
         declare
            Y_Max : Integer := Integer(Sqrt(Float(Radius * Radius - X * X)));
         begin
            Count := Count + (2 * Y_Max + 1);
         end;
      end loop;
      
      return Count;
   end Efficient_Count;
   
begin
   Put_Line("Project Euler Problem 262");
   Put_Line("==========================");
   
   -- For radius = 100 (typical problem value)
   declare
      Radius : constant Integer := 100;
      Result : Integer;
   begin
      Result := Efficient_Count(Radius);
      Put("Number of lattice points in circle with radius ");
      Put(Radius);
      Put_Line(" is:");
      Put(Result);
      New_Line;
      
      -- Verification with small examples
      Put_Line("Verification:");
      Put("Radius 1: ");
      Put(Efficient_Count(1));
      New_Line;
      
      Put("Radius 2: ");
      Put(Efficient_Count(2));
      New_Line;
   end;
   
end Euler_262;
```
