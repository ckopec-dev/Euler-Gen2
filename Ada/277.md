# Euler Problem 277 Solution in Ada

This problem asks us to find the number of steps required for a number to reach 1 using the "Recursed" sequence where:
- If n is even: n → n/2
- If n is odd: n → (3n+1)/2

Let me solve this step by step.

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_277 is
   
   -- Function to calculate the number of steps to reach 1
   function Steps_To_One(n : Integer) return Integer is
      current : Integer := n;
      steps   : Integer := 0;
   begin
      while current /= 1 loop
         if current mod 2 = 0 then
            -- If even: n → n/2
            current := current / 2;
         else
            -- If odd: n → (3n+1)/2
            current := (3 * current + 1) / 2;
         end if;
         steps := steps + 1;
      end loop;
      
      return steps;
   end Steps_To_One;
   
   -- Function to check if a number is a "Recursed" number
   function Is_Recursed(n : Integer) return Boolean is
      steps : Integer := Steps_To_One(n);
   begin
      return steps = 15;
   end Is_Recursed;
   
   -- Main solution
   procedure Solve is
      count : Integer := 0;
      limit : constant Integer := 1000000;
   begin
      for i in 1..limit loop
         if Steps_To_One(i) = 15 then
            count := count + 1;
         end if;
      end loop;
      
      Put("Numbers with exactly 15 steps: ");
      Put(count);
      New_Line;
   end Solve;
   
begin
   Solve;
end Euler_277;
```

```ada
with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_277 is
   
   -- Maximum value to check
   MAX_VAL : constant Integer := 1000000;
   
   -- Memoization array to store step counts
   Steps_Count : array(1..MAX_VAL) of Integer := (others => -1);
   
   -- Function to calculate number of steps with memoization
   function Recursed_Steps(n : Integer) return Integer is
      current : Integer := n;
      steps   : Integer := 0;
      temp    : Integer;
   begin
      if n <= MAX_VAL and then Steps_Count(n) /= -1 then
         return Steps_Count(n);
      end if;
      
      temp := n;
      while temp /= 1 loop
         if temp mod 2 = 0 then
            -- Even case: n → n/2
            temp := temp / 2;
         else
            -- Odd case: n → (3n+1)/2  
            temp := (3 * temp + 1) / 2;
         end if;
         steps := steps + 1;
      end loop;
      
      -- Store in memoization table if within bounds
      if n <= MAX_VAL then
         Steps_Count(n) := steps;
      end if;
      
      return steps;
   end Recursed_Steps;
   
   -- Main program
   procedure Solve is
      count : Integer := 0;
   begin
      for i in 1..MAX_VAL loop
         if Recursed_Steps(i) = 15 then
            count := count + 1;
         end if;
      end loop;
      
      Put_Line("Count of numbers under 1,000,000 with exactly 15 steps:");
      Put(count);
      New_Line;
   end Solve;
   
begin
   Solve;
end Euler_277;
```

The key insight of this problem is understanding the "Recursed" sequence where:
- Even numbers are halved (n → n/2)
- Odd numbers follow the rule (n → (3n+1)/2)

This creates a different behavior from the classic Collatz conjecture, and we're looking for numbers that take exactly 15 steps to reach 1.
