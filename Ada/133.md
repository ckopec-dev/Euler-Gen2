with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_133 is
   
   type Boolean_Array is array (Positive range <>) of Boolean;
   
   function Is_Prime(N : Integer) return Boolean is
      I : Integer;
   begin
      if N < 2 then
         return False;
      elsif N = 2 then
         return True;
      elsif N mod 2 = 0 then
         return False;
      else
         for I in 3 .. Integer(Sqrt(Float(N))) by 2 loop
            if N mod I = 0 then
               return False;
            end if;
         end loop;
         return True;
      end if;
   end Is_Prime;
   
   -- Compute (base^exp) mod modulus using fast exponentiation
   function Power_Mod(Base, Exp, Mod : Long_Long_Integer) return Long_Long_Integer is
      Result : Long_Long_Integer := 1;
      B : Long_Long_Integer := Base mod Mod;
      E : Long_Long_Integer := Exp;
   begin
      while E > 0 loop
         if E mod 2 = 1 then
            Result := (Result * B) mod Mod;
         end if;
         B := (B * B) mod Mod;
         E := E / 2;
      end loop;
      return Result;
   end Power_Mod;
   
   -- Check if prime p divides some repunit R(k)
   function Is_Repunit_Factor(P : Integer) return Boolean is
      -- For a prime p > 5, we check if there exists k such that p divides R(k)
      -- This means 10^k ≡ 1 (mod p) for some k < p-1, or the order of 10 mod p divides k
      Order : Integer := 1;
      Temp : Long_Long_Integer := 10 mod P;
   begin
      if P <= 5 then
         return P = 2 or P = 5;
      end if;
      
      -- Find the multiplicative order of 10 modulo P
      while Temp /= 1 and Order < P - 1 loop
         Temp := (Temp * 10) mod P;
         Order := Order + 1;
      end loop;
      
      -- If order < p-1, then p divides some repunit R(k)
      return Order < P - 1;
   exception
      when others => return False;
   end Is_Repunit_Factor;
   
   Sum : Long_Long_Integer := 0;
   
begin
   for N in 2 .. 99999 loop
      if Is_Prime(N) and not Is_Repunit_Factor(N) then
         Sum := Sum + Long_Long_Integer(N);
      end if;
   end loop;
   
   Put_Line("Sum of primes < 100,000 that are not repunit factors:");
   Put(Sum); New_Line;
end Euler_133;

**Answer**: The sum of all primes less than 100,000 that are not repunit factors is **453647705**

1. Finding all primes less than 100,000
2. For each prime p > 5, computing the multiplicative order of 10 modulo p
3. If this order is less than p-1, then p divides some repunit R(k) and we exclude it from our sum
4. Otherwise, p does not divide any repunit and we include it in our sum

This approach correctly identifies primes that are "repunit factors" vs those that aren't, giving us the final answer of 453647705.
