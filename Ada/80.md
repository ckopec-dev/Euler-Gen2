# Euler Problem 80 Solution in Ada

Euler Problem 80 asks us to find the digital sum of the first 100 decimal digits of the square root of all natural numbers that are not perfect squares.

with Ada.Numerics.Big_Numbers.Big_Rationals; use Ada.Numerics.Big_Numbers.Big_Rationals;
with Ada.Numerics.Big_Numbers.Big_Integers; use Ada.Numerics.Big_Numbers.Big_Integers;

procedure Euler_80 is
   type Decimal_Array is array (Positive range <>) of Natural;
   
   -- Function to compute square root with high precision
   function Sqrt_Rational(N : Big_Integer) return Big_Rational is
      Result : Big_Rational := To_Big_Rational(1);
      Prev   : Big_Rational;
      Iter   : Positive := 1;
      Max_Iter : constant Positive := 1000; -- Enough iterations for precision
   begin
      loop
         Prev := Result;
         Result := (Prev + N / Prev) / 2;
         exit when Iter >= Max_Iter or else 
                  (Abs(Prev - Result) < To_Big_Rational(1, 10**50));
         Iter := Iter + 1;
      end loop;
      
      return Result;
   end Sqrt_Rational;
   
   -- Function to get first N decimal digits of a rational number
   function Get_Decimal_Digits(R : Big_Rational; Num_Digits : Positive) return Decimal_Array is
      Temp : Big_Rational := R;
      Digits : Decimal_Array(1..Num_Digits);
      Digit : Natural;
   begin
      -- Extract integer part first
      for I in 1..Num_Digits loop
         if Temp >= 1 then
            Digit := To_Natural(Temp);
            Temp := (Temp - To_Big_Rational(Digit)) * 10;
         else
            Temp := Temp * 10;
            Digit := 0;
         end if;
         Digits(I) := Digit;
      end loop;
      
      return Digits;
   end Get_Decimal_Digits;
   
   -- Function to calculate digital sum of an array
   function Digital_Sum(Digits : Decimal_Array) return Natural is
      Sum : Natural := 0;
   begin
      for I in Digits'Range loop
         Sum := Sum + Digits(I);
      end loop;
      return Sum;
   end Digital_Sum;
   
   -- Check if number is a perfect square
   function Is_Perfect_Square(N : Positive) return Boolean is
      Root : constant Natural := Integer(Sqrt(Real(N)));
   begin
      return Root * Root = N;
   end Is_Perfect_Square;
   
   Total_Sum : Natural := 0;
   Num_Digits : constant Positive := 100;
   
begin
   for I in 1..100 loop
      if not Is_Perfect_Square(I) then
         -- Compute square root with high precision
         declare
            N_Big : constant Big_Integer := To_Big_Integer(I);
            Sqrt_Val : constant Big_Rational := Sqrt_Rational(N_Big);
            Digits : constant Decimal_Array := Get_Decimal_Digits(Sqrt_Val, Num_Digits);
         begin
            Total_Sum := Total_Sum + Digital_Sum(Digits);
         end;
      end if;
   end loop;
   
   Put_Line("The digital sum is: " & Natural'Image(Total_Sum));
end Euler_80;
