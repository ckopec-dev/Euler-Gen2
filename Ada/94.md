# Euler Problem 94 Solution in Ada

This problem asks for the perimeter of almost equilateral triangles with integer sides and area, where the difference between two sides is at most 1.

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_94 is
   type Triangle is record
      a, b, c : Integer;
   end record;
   
   function Is_Integer(S : Float) return Boolean is
   begin
      return S = Float(Integer(S));
   end Is_Integer;
   
   function Area(T : Triangle) return Float is
      s : Float := Float(T.a + T.b + T.c) / 2.0;
   begin
      return sqrt(s * (s - Float(T.a)) * (s - Float(T.b)) * (s - Float(T.c)));
   end Area;
   
   function Is_Almost_Equilateral(T : Triangle) return Boolean is
   begin
      return (abs(T.a - T.b) <= 1) and (abs(T.b - T.c) <= 1) and (abs(T.a - T.c) <= 1);
   end Is_Almost_Equilateral;
   
   function Perimeter(T : Triangle) return Integer is
   begin
      return T.a + T.b + T.c;
   end Perimeter;
   
   Max_Perimeter : constant := 1_000_000_000;
   Total : Integer := 0;
   
begin
   -- We need to find triangles where:
   -- 1. The difference between any two sides is at most 1 (almost equilateral)
   -- 2. All sides are integers
   -- 3. The area is also an integer
   
   -- For almost equilateral triangles, there are only a few cases:
   -- Case 1: Three equal sides (equilateral) - not "almost" but valid
   -- Case 2: Two sides equal, third differs by 1
   -- Case 3: All three sides differ by at most 1
   
   -- Let's focus on the case where two sides are equal and the third differs by 1
   -- This gives us triangles of form (a, a, a+1) or (a, a, a-1)
   
   for a in 1..Max_Perimeter loop
      -- Check triangle (a, a, a+1)
      declare
         T : Triangle := (a, a, a + 1);
         Area_Val : Float;
      begin
         if Perimeter(T) > Max_Perimeter then
            exit;
         end if;
         
         Area_Val := Area(T);
         if Is_Integer(Area_Val) and Is_Almost_Equilateral(T) then
            Total := Total + Perimeter(T);
         end if;
      end;
      
      -- Check triangle (a, a, a-1)
      if a > 1 then
         declare
            T : Triangle := (a, a, a - 1);
            Area_Val : Float;
         begin
            if Perimeter(T) > Max_Perimeter then
               exit;
            end if;
            
            Area_Val := Area(T);
            if Is_Integer(Area_Val) and Is_Almost_Equilateral(T) then
               Total := Total + Perimeter(T);
            end if;
         end;
      end if;
   end loop;
   
   Put_Line("Result: " & Integer'Image(Total));
end Euler_94;
```

```ada
with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_94 is
   Max_Perimeter : constant := 1_000_000_000;
   Total : Long_Long_Integer := 0;
   
   function Sqrt(X : Long_Long_Float) return Long_Long_Float is
   begin
      return Long_Long_Float'Sqrt(X);
   end Sqrt;
   
   function Area(A, B, C : Long_Long_Integer) return Long_Long_Float is
      S : Long_Long_Float := Long_Long_Float(A + B + C) / 2.0;
   begin
      return Sqrt(S * (S - Long_Long_Float(A)) * (S - Long_Long_Float(B)) * (S - Long_Long_Float(C)));
   end Area;
   
   function Is_Integer(X : Long_Long_Float) return Boolean is
   begin
      return X = Long_Long_Float(Long_Long_Integer(X));
   end Is_Integer;
   
begin
   -- For almost equilateral triangles with integer sides and area:
   -- The key is to recognize that these are related to Pell's equation
   
   -- We'll look for triangles of the form (a, a, a+1) or (a, a, a-1)
   -- where a > 0 and the area is an integer
   
   -- Let's use a mathematical approach:
   -- For triangle with sides a, a, a±1, using Heron's formula
   -- Area = sqrt(3a² ± 2a - 1) / 4
   
   for a in 1..Max_Perimeter loop
      declare
         perimeter : Long_Long_Integer;
         area : Long_Long_Float;
      begin
         -- Case (a, a, a+1)
         if 3 * a + 1 <= Max_Perimeter then
            perimeter := 3 * a + 1;
            area := Area(a, a, a + 1);
            if Is_Integer(area) then
               Total := Total + perimeter;
            end if;
         end if;
         
         -- Case (a, a, a-1)
         if a > 1 and then 3 * a - 1 <= Max_Perimeter then
            perimeter := 3 * a - 1;
            area := Area(a, a, a - 1);
            if Is_Integer(area) then
               Total := Total + perimeter;
            end if;
         end if;
      end;
   end loop;
   
   Put_Line("Answer: " & Long_Long_Integer'Image(Total));
end Euler_94;
```
