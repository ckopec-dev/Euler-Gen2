with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_324 is
   type Digit is range 1..5;
   type Mod_5 is range 0..4;
   
   -- We'll use dynamic programming: dp[length][sum_mod_5]
   type DP_Table is array (0..9, Mod_5) of Long_Long_Integer;
   
   -- Initialize DP table
   DP : DP_Table := (others => (others => 0));
   
   -- For a 10-digit number, we need to count valid combinations
   -- We'll build up from 1 digit to 10 digits
   
   -- Base case: 1-digit numbers
   -- Each of {1,2,3,4,5} gives a remainder when divided by 5
   for I in 1..5 loop
      DP(0, Mod_5(I mod 5)) := DP(0, Mod_5(I mod 5)) + 1;
   end loop;
   
   -- Fill the DP table
   for Length in 1..9 loop
      for Remainder in Mod_5 loop
         -- For each digit from 1 to 5
         for D in Digit loop
            -- New remainder = (old_remainder + new_digit) mod 5
            declare
               New_Remainder : constant Mod_5 := Mod_5((Remainder + D) mod 5);
            begin
               DP(Length, New_Remainder) := 
                  DP(Length, New_Remainder) + DP(Length-1, Remainder);
            end;
         end loop;
      end loop;
   end loop;
   
   -- The answer is the number of 10-digit numbers with sum divisible by 5
   -- Length = 9 (since we start from 0), remainder = 0
   Result : Long_Long_Integer := DP(9, 0);
   
begin
   Put("The answer to Euler Problem 324 is: ");

   New_Line;
