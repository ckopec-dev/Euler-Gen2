# Euler Problem 209 Solution in Ada

This problem asks for the number of ways to color a 3x3 grid with 3 colors such that no two adjacent cells (sharing an edge) have the same color.

procedure Euler_209 is
   type Color is (Red, Green, Blue);
   type Grid is array (1..3, 1..3) of Color;
   
   -- Function to check if a coloring is valid
   function Is_Valid_Coloring(G : Grid) return Boolean is
      Adjacent : array (1..3, 1..3) of Boolean := 
        ((True, True, True),
         (True, True, True),
         (True, True, True));
   begin
      -- Check all adjacent pairs (horizontal and vertical)
      for i in 1..3 loop
         for j in 1..3 loop
            -- Check right neighbor
            if j < 3 and G(i,j) = G(i,j+1) then
               return False;
            end if;
            
            -- Check bottom neighbor
            if i < 3 and G(i,j) = G(i+1,j) then
               return False;
            end if;
         end loop;
      end loop;
      
      return True;
   end Is_Valid_Coloring;
   
   -- Count valid colorings
   Count : Natural := 0;
   
   -- Recursive procedure to try all colorings
   procedure Try_Coloring(G : in out Grid; Row, Col : Integer) is
   begin
      if Row > 3 then
         -- Check if this is a valid coloring
         if Is_Valid_Coloring(G) then
            Count := Count + 1;
         end if;
         return;
      end if;
      
      -- Move to next cell
      declare
         Next_Row, Next_Col : Integer;
      begin
         if Col = 3 then
            Next_Row := Row + 1;
            Next_Col := 1;
         else
            Next_Row := Row;
            Next_Col := Col + 1;
         end if;
         
         -- Try each color for current cell
         for C in Color loop
            G(Row, Col) := C;
            Try_Coloring(G, Next_Row, Next_Col);
         end loop;
      end;
   end Try_Coloring;
   
begin
   declare
      Grid : Grid;
   begin
      Try_Coloring(Grid, 1, 1);
      Put("Number of valid colorings: ");

   end;
end Euler_209;
