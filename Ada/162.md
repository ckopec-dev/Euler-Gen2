# Euler Problem 162 Solution in Ada

This problem asks for the number of hexadecimal numbers with at most 16 digits that contain all three digits 0, A, and F at least once.

with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_162 is
   type Hex_Count is array (0..15) of array (0..7) of Long_Long_Integer;
   -- State: 0=none, 1=0 only, 2=A only, 3=0&A, 4=F only, 5=0&F, 6=A&F, 7=0&A&F
   
   function Count_Hex (Digits : Positive; State : Natural) return Long_Long_Integer is
      Result : Long_Long_Integer := 0;
   begin
      if Digits = 0 then
         if State = 7 then
            return 1;
         else
            return 0;
         end if;
      end if;
      
      -- For the first digit, we can't use 0 (as it would make leading zeros)
      if Digits = 16 then
         -- First digit: A-F (15 choices), but not 0
         Result := Count_Hex(Digits - 1, 2) + Count_Hex(Digits - 1, 4); -- A or F
         -- Add counts for 0 (but only if we're counting a valid number)
         Result := Result + 15 * Count_Hex(Digits - 1, 0);
      else
         -- For subsequent digits: 0-F (16 choices)
         case State is
            when 0 =>
               Result := 14 * Count_Hex(Digits - 1, 0) + 
                         Count_Hex(Digits - 1, 1) + 
                         Count_Hex(Digits - 1, 2) +
                         Count_Hex(Digits - 1, 4);
            when 1 =>
               Result := 15 * Count_Hex(Digits - 1, 1) + 
                         Count_Hex(Digits - 1, 3) +
                         Count_Hex(Digits - 1, 4);
            when 2 =>
               Result := 15 * Count_Hex(Digits - 1, 2) + 
                         Count_Hex(Digits - 1, 3) +
                         Count_Hex(Digits - 1, 6);
            when 3 =>
               Result := 14 * Count_Hex(Digits - 1, 3) + 
                         Count_Hex(Digits - 1, 5) +
                         Count_Hex(Digits - 1, 6);

               Result := 15 * Count_Hex(Digits - 1, 4) +

                         Count_Hex(Digits - 1, 6);

               Result := 14 * Count_Hex(Digits - 1, 5) +

            when 6 =>
               Result := 14 * Count_Hex(Digits - 1, 6) + 
                         Count_Hex(Digits - 1, 7);
            when 7 =>
               Result := 16 * Count_Hex(Digits - 1, 7);
         end case;
      end if;
      
      return Result;
   end Count_Hex;

               when 7 =>
                  DP(Digits, State) := 16 * DP(Digits - 1, State) + DP(Digits - 1, 3) + DP(Digits - 1, 5) + DP(Digits - 1, 6);

      Result : Long_Long_Integer := 0;

      -- Total valid strings = All strings - Strings missing at least one of {0,A,F}

      -- But for valid numbers, first digit can't be 0, so it's 15*16^(N-1)

      begin
         if Exp = 0 then
            return 1;
         elsif Exp = 1 then
            return Base;
         else
            return Base * Power(Base, Exp - 1);
         end if;
      end Power;

      -- dp[i][s] = number of i-digit hex strings with state s
      -- where state s encodes which digits we've seen:

                     D(i, 7) := 16 * D(i-1, 7) + D(i-1, 3) + D(i-1, 5) + D(i-1, 6);
               end case;
            end loop;
         end loop;
         
         return D(Max_Length, 7);
      end;
   end Count_Valid_Hex_Strings;
