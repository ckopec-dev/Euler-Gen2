# Euler Problem 201 Solution in Ada

This problem asks for the number of ways to select exactly 50 elements from the set {1², 2², ..., 100²} such that their sum equals 201.

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_201 is
   -- Precompute squares
   type Square_Array is array (1..100) of Integer;
   Squares : Square_Array;
   
   -- Memoization table for dynamic programming
   type Memo_Table is array (0..100, 0..201, 0..50) of Long_Long_Integer;
   Memo : Memo_Table := (others => (others => (others => -1)));
   
   -- Compute number of ways to select exactly 'count' elements from first 'n' squares
   -- such that their sum equals 'target'
   function Count_Ways(n, target, count : Integer) return Long_Long_Integer is
      Result : Long_Long_Integer;
   begin
      -- Base cases
      if count = 0 then
         if target = 0 then
            return 1;
         else
            return 0;
         end if;
      end if;
      
      if n = 0 or target < 0 or count > n then
         return 0;
      end if;
      
      -- Check memo table
      if Memo(n, target, count) /= -1 then
         return Memo(n, target, count);
      end if;
      
      -- Two choices: include the nth square or don't include it
      Result := Count_Ways(n-1, target, count);  -- Don't include n²
      Result := Result + Count_Ways(n-1, target - Squares(n), count-1);  -- Include n²
      
      Memo(n, target, count) := Result;
      return Result;
   end Count_Ways;
   
begin
   -- Initialize squares
   for I in 1..100 loop
      Squares(I) := I * I;
   end loop;
   
   -- Calculate result: ways to select exactly 50 elements from first 100 squares
   -- such that their sum equals 201
   Put_Line("Result: ");
   Put(Count_Ways(100, 201, 50));
   New_Line;
end Euler_201;
```
