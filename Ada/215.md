# Euler Problem 215 Solution in Ada

This problem asks us to find the number of ways to build a wall of height 32 and width 10 using bricks of sizes 1×1, 1×2, 1×3, and 1×4.

Let me break this down into steps:

1. First, we need to find all possible brick arrangements for a single row (width 10)
2. Then, we need to determine which rows can be stacked on top of each other
3. Finally, we use dynamic programming to count the total arrangements

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_215 is
   
   -- Constants
   Width : constant Integer := 10;
   Height : constant Integer := 32;
   
   -- Maximum number of possible row arrangements
   Max_Row_Arrangements : constant Integer := 1000;
   
   -- Type definitions
   type Brick_Array is array (Integer range <>) of Integer;
   type Row_Arrangement is array (Integer range 1..Width) of Boolean; -- True where brick ends
   
   -- Function to generate all possible row arrangements
   function Generate_Row_Arrangements return Integer is
      Arrangements : array (1..Max_Row_Arrangements) of Row_Arrangement;
      Count : Integer := 0;
      
      procedure Try_Row(Position : Integer; Current_Row : in out Row_Arrangement) is
      begin
         if Position > Width then
            Count := Count + 1;
            Arrangements(Count) := Current_Row;
            return;
         end if;
         
         -- Try placing a brick of size 1, 2, 3, or 4
         for Brick_Size in 1..4 loop
            if Position + Brick_Size - 1 <= Width then
               -- Mark the positions where this brick ends
               for I in Position..Position + Brick_Size - 1 loop
                  Current_Row(I) := True;
               end loop;
               Try_Row(Position + Brick_Size, Current_Row);
               -- Backtrack: unmark positions
               for I in Position..Position + Brick_Size - 1 loop
                  Current_Row(I) := False;
               end loop;
            end if;
         end loop;
      end Try_Row;
      
   begin
      declare
         Empty_Row : Row_Arrangement := (others => False);
      begin
         Try_Row(1, Empty_Row);
      end;
      return Count;
   end Generate_Row_Arrangements;
   
   -- Function to check if two arrangements can be stacked
   function Can_Stack(Arr1, Arr2 : Row_Arrangement) return Boolean is
      -- Two rows can be stacked if they don't have joints in the same position
      -- (except at the very edges)
   begin
      for I in 2..Width-1 loop
         if Arr1(I) and Arr2(I) then
            return False;
         end if;
      end loop;
      return True;
   end Can_Stack;
   
   -- Main solution
   function Solve return Long_Long_Integer is
      Row_Count : constant Integer := Generate_Row_Arrangements;
      
      -- Store all valid row arrangements
      All_Rows : array (1..Max_Row_Arrangements) of Row_Arrangement;
      
      -- Precompute which rows can stack on each other
      Can_Stack_Matrix : array (1..Max_Row_Arrangements, 1..Max_Row_Arrangements) of Boolean;
      
      -- DP table: dp[row][height] = number of ways to build wall ending with that row at that height
      DP : array (1..Max_Row_Arrangements, 0..Height) of Long_Long_Integer;
      
      procedure Generate_Rows is
         Count : Integer := 0;
         
         procedure Try_Row(Position : Integer; Current_Row : in out Row_Arrangement) is
         begin
            if Position > Width then
               Count := Count + 1;
               All_Rows(Count) := Current_Row;
               return;
            end if;
            
            for Brick_Size in 1..4 loop
               if Position + Brick_Size - 1 <= Width then
                  -- Mark the positions where this brick ends
                  for I in Position..Position + Brick_Size - 1 loop
                     Current_Row(I) := True;
                  end loop;
                  Try_Row(Position + Brick_Size, Current_Row);
                  -- Backtrack
                  for I in Position..Position + Brick_Size - 1 loop
                     Current_Row(I) := False;
                  end loop;
               end if;
            end loop;
         end Try_Row;
         
      begin
         declare
            Empty_Row : Row_Arrangement := (others => False);
         begin
            Try_Row(1, Empty_Row);
         end;
      end Generate_Rows;
      
   begin
      -- Generate all row arrangements
      Generate_Rows;
      
      -- Build the stacking matrix
      for I in 1..Row_Count loop
         for J in 1..Row_Count loop
            Can_Stack_Matrix(I,J) := Can_Stack(All_Rows(I), All_Rows(J));
         end loop;
      end loop;
      
      -- Initialize DP: for height = 0 (base case), each arrangement contributes 1 way
      for I in 1..Row_Count loop
         DP(I, 0) := 1;
      end loop;
      
      -- Fill DP table
      for H in 1..Height loop
         for I in 1..Row_Count loop
            DP(I, H) := 0;
            for J in 1..Row_Count loop
               if Can_Stack_Matrix(J, I) then
                  DP(I, H) := DP(I, H) + DP(J, H-1);
               end if;
            end loop;
         end loop;
      end loop;
      
      -- Sum up all ways to reach height 32
      declare
         Result : Long_Long_Integer := 0;
      begin
         for I in 1..Row_Count loop
            Result := Result + DP(I, Height);
         end loop;
         return Result;
      end;
   end Solve;
   
begin
   Put_Line("Euler Problem 215 Solution:");
   Put(Solve);
   New_Line;
end Euler_215;
```
