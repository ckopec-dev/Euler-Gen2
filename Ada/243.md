# Euler Problem 243 Solution in Ada

procedure Euler_243 is
   
   -- Calculate greatest common divisor using Euclidean algorithm
   function GCD(A, B : Long_Long_Integer) return Long_Long_Integer is
   begin
      if B = 0 then
         return A;
      else
         return GCD(B, A mod B);
      end if;
   end GCD;
   
   -- Calculate Euler's totient function φ(n)
   function Totient(N : Long_Long_Integer) return Long_Long_Integer is
      Result : Long_Long_Integer := N;
      I      : Long_Long_Integer := 2;
   begin
      if N = 1 then
         return 1;
      end if;
      
      -- For each prime factor p of n, multiply result by (p-1)/p
      while I * I <= N loop
         if N mod I = 0 then
            Result := Result - (Result / I);
            while N mod I = 0 loop
               N := N / I;
            end loop;
         end if;
         I := I + 1;
      end loop;
      
      -- If N > 1, then it's a prime factor
      if N > 1 then
         Result := Result - (Result / N);
      end if;
      
      return Result;
   end Totient;
   
   -- Check if a number is prime
   function Is_Prime(N : Long_Long_Integer) return Boolean is
   begin
      if N < 2 then
         return False;
      elsif N = 2 then
         return True;
      elsif N mod 2 = 0 then
         return False;
      else
         for I in 3 .. Long_Long_Integer'Sqrt(Long_Long_Float(N)) loop
            if N mod I = 0 then
               return False;
            end if;
         end loop;
         return True;
      end if;
   end Is_Prime;

         T : Long_Long_Integer := Totient(N);
      begin
         return Long_Long_Float(T) / Long_Long_Float(N);
      end Ratio;
