# Euler Problem 61 Solution in Ada

Euler Problem 61 asks us to find the sum of a set of six-digit numbers that form a cyclic set, where each number is a polygonal number (triangle, square, pentagonal, hexagonal, heptagonal, octagonal) and each number links to the next by sharing the last two digits of the previous number.

Here's my solution in Ada:

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_61 is
   
   -- Maximum number of polygonal numbers to consider
   Max_Numbers : constant := 10000;
   
   -- Function to check if a number is a triangle number
   function Is_Triangle(N : Integer) return Boolean is
      K : Integer := Integer(2.0 * Float(N)) ** 0.5;
   begin
      return K * (K + 1) / 2 = N;
   end Is_Triangle;
   
   -- Function to check if a number is a square number
   function Is_Square(N : Integer) return Boolean is
      K : Integer := Integer(Float(N) ** 0.5);
   begin
      return K * K = N;
   end Is_Square;
   
   -- Function to check if a number is a pentagonal number
   function Is_Pentagonal(N : Integer) return Boolean is
      K : Integer := Integer((1.0 + (1.0 + 24.0 * Float(N)) ** 0.5) / 6.0);
   begin
      return K * (3 * K - 1) / 2 = N;
   end Is_Pentagonal;
   
   -- Function to check if a number is a hexagonal number
   function Is_Hexagonal(N : Integer) return Boolean is
      K : Integer := Integer((1.0 + (1.0 + 8.0 * Float(N)) ** 0.5) / 4.0);
   begin
      return K * (2 * K - 1) = N;
   end Is_Hexagonal;
   
   -- Function to check if a number is a heptagonal number
   function Is_Heptagonal(N : Integer) return Boolean is
      K : Integer := Integer((3.0 + (9.0 + 24.0 * Float(N)) ** 0.5) / 6.0);
   begin
      return K * (5 * K - 3) / 2 = N;
   end Is_Heptagonal;
   
   -- Function to check if a number is an octagonal number
   function Is_Octagonal(N : Integer) return Boolean is
      K : Integer := Integer((2.0 + (4.0 + 8.0 * Float(N)) ** 0.5) / 4.0);
   begin
      return K * (3 * K - 2) = N;
   end Is_Octagonal;
   
   -- Get last two digits of a number
   function Last_Two_Digits(N : Integer) return Integer is
   begin
      return N mod 100;
   end Last_Two_Digits;
   
   -- Get first two digits of a number
   function First_Two_Digits(N : Integer) return Integer is
   begin
      return N / 100;
   end First_Two_Digits;
   
   -- Array to store polygonal numbers for each type
   Triangle_Numbers : array(1..Max_Numbers) of Integer := (others => 0);
   Square_Numbers : array(1..Max_Numbers) of Integer := (others => 0);
   Pentagonal_Numbers : array(1..Max_Numbers) of Integer := (others => 0);
   Hexagonal_Numbers : array(1..Max_Numbers) of Integer := (others => 0);
   Heptagonal_Numbers : array(1..Max_Numbers) of Integer := (others => 0);
   Octagonal_Numbers : array(1..Max_Numbers) of Integer := (others => 0);
   
   -- Counters for each type
   Triangle_Count : Integer := 0;
   Square_Count : Integer := 0;
   Pentagonal_Count : Integer := 0;
   Hexagonal_Count : Integer := 0;
   Heptagonal_Count : Integer := 0;
   Octagonal_Count : Integer := 0;
   
   -- Generate polygonal numbers
   procedure Generate_Polygonal_Numbers is
      I : Integer;
      T, S, P, H, He, O : Integer;
   begin
      I := 1;
      loop
         T := I * (I + 1) / 2;     -- Triangle
         S := I * I;               -- Square  
         P := I * (3 * I - 1) / 2; -- Pentagonal
         H := I * (2 * I - 1);     -- Hexagonal
         He := I * (5 * I - 3) / 2;-- Heptagonal
         O := I * (3 * I - 2);     -- Octagonal
         
         exit when T > 9999 or S > 9999 or P > 9999 or H > 9999 or He > 9999 or O > 9999;
         
         if T >= 1000 and T <= 9999 then
            Triangle_Count := Triangle_Count + 1;
            Triangle_Numbers(Triangle_Count) := T;
         end if;
         
         if S >= 1000 and S <= 9999 then
            Square_Count := Square_Count + 1;
            Square_Numbers(Square_Count) := S;
         end if;
         
         if P >= 1000 and P <= 9999 then
            Pentagonal_Count := Pentagonal_Count + 1;
            Pentagonal_Numbers(Pentagonal_Count) := P;
         end if;
         
         if H >= 1000 and H <= 9999 then
            Hexagonal_Count := Hexagonal_Count + 1;
            Hexagonal_Numbers(Hexagonal_Count) := H;
         end if;
         
         if He >= 1000 and He <= 9999 then
            Heptagonal_Count := Heptagonal_Count + 1;
            Heptagonal_Numbers(Heptagonal_Count) := He;
         end if;
         
         if O >= 1000 and O <= 9999 then
            Octagonal_Count := Octagonal_Count + 1;
            Octagonal_Numbers(Octagonal_Count) := O;
         end if;
         
         I := I + 1;
      end loop;
   end Generate_Polygonal_Numbers;
   
   -- Function to check if numbers form a cyclic set
   function Is_Cyclic_Set(Numbers : array(1..6) of Integer) return Boolean is
      First : Integer := Numbers(1);
      Last : Integer := Numbers(6);
   begin
      return 
         (First / 100 = Last mod 100) and
         (Numbers(2) / 100 = Numbers(1) mod 100) and
         (Numbers(3) / 100 = Numbers(2) mod 100) and
         (Numbers(4) / 100 = Numbers(3) mod 100) and
         (Numbers(5) / 100 = Numbers(4) mod 100) and
         (Numbers(6) / 100 = Numbers(5) mod 100);
   end Is_Cyclic_Set;
   
   -- Main solution
   procedure Solve is
      Result : Integer := 0;
      Found : Boolean := False;
      
      -- Arrays to hold the six numbers
      Numbers : array(1..6) of Integer;
      
      -- Loop through all combinations
      procedure Try_Combinations(Index : Integer) is
         I, J, K, L, M, N : Integer;
         Temp_Numbers : array(1..6) of Integer;
      begin
         if Index = 7 then
            -- Check if this set forms a valid cycle
            if Is_Cyclic_Set(Numbers) then
               -- Verify that each number is from a different polygonal type
               if Numbers(1) /= Numbers(2) and Numbers(1) /= Numbers(3) and 
                  Numbers(1) /= Numbers(4) and Numbers(1) /= Numbers(5) and 
                  Numbers(1) /= Numbers(6) and Numbers(2) /= Numbers(3) and 
                  Numbers(2) /= Numbers(4) and Numbers(2) /= Numbers(5) and 
                  Numbers(2) /= Numbers(6) and Numbers(3) /= Numbers(4) and 
                  Numbers(3) /= Numbers(5) and Numbers(3) /= Numbers(6) and 
                  Numbers(4) /= Numbers(5) and Numbers(4) /= Numbers(6) and 
                  Numbers(5) /= Numbers(6) then
                  Result := Numbers(1) + Numbers(2) + Numbers(3) + Numbers(4) + Numbers(5) + Numbers(6);
                  Found := True;
               end if;
            end if;
         else
            case Index is
               when 1 =>
                  for I in 1..Triangle_Count loop
                     Numbers(Index) := Triangle_Numbers(I);
                     Try_Combinations(Index + 1);
                     exit when Found;
                  end loop;
               when 2 =>
                  for J in 1..Square_Count loop
                     Numbers(Index) := Square_Numbers(J);
                     Try_Combinations(Index + 1);
                     exit when Found;
                  end loop;
               when 3 =>
                  for K in 1..Pentagonal_Count loop
                     Numbers(Index) := Pentagonal_Numbers(K);
                     Try_Combinations(Index + 1);
                     exit when Found;
                  end loop;
               when 4 =>
                  for L in 1..Hexagonal_Count loop
                     Numbers(Index) := Hexagonal_Numbers(L);
                     Try_Combinations(Index + 1);
                     exit when Found;
                  end loop;
               when 5 =>
                  for M in 1..Heptagonal_Count loop
                     Numbers(Index) := Heptagonal_Numbers(M);
                     Try_Combinations(Index + 1);
                     exit when Found;
                  end loop;
               when 6 =>
                  for N in 1..Octagonal_Count loop
                     Numbers(Index) := Octagonal_Numbers(N);
                     Try_Combinations(Index + 1);
                     exit when Found;
                  end loop;
            end case;
         end if;
      end Try_Combinations;
      
   begin
      Generate_Polygonal_Numbers;
      Try_Combinations(1);
      Put_Line("The sum is: " & Integer'Image(Result));
   end Solve;

begin
   Solve;
end Euler_61;
```

This solution:

1. **Generates all 4-digit polygonal numbers** for each type (triangle, square, pentagonal, hexagonal, heptagonal, octagonal)
2. **Uses a backtracking approach** to try different combinations of these numbers
3. **Checks for the cyclic property** where the last two digits of one number match the first two digits of the next
4. **Ensures each number is from a different polygonal type**
5. **Returns the sum** of the six numbers that satisfy all conditions

The algorithm works by:
- Generating all 4-digit polygonal numbers for each category
- Trying all combinations systematically
- Validating both the cyclic property and that each number comes from a different polygonal type
- Outputting the sum when the solution is found

This approach ensures we find the unique set of six numbers that form the required cyclic pattern, where each number is a different polygonal number and the sequence wraps around properly.
