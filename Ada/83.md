# Euler Problem 83 Solution in Ada

This problem requires finding the minimal path sum from the top-left to bottom-right of a grid, where movement is allowed in all four directions (up, down, left, right).

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;
with Ada.Containers.Vectors;
with Ada.Priority_Queues;

procedure Euler_83 is
   
   -- Grid dimensions
   Width  : constant := 80;
   Height : constant := 80;
   
   -- Define grid type
   type Grid_Type is array (1..Height, 1..Width) of Integer;
   
   -- Read the grid from input
   procedure Read_Grid(Grid : out Grid_Type);
   
   -- Find minimum path sum using Dijkstra's algorithm
   function Min_Path_Sum(Grid : Grid_Type) return Integer;
   
   -- Implementation
   Grid : Grid_Type;
   
begin
   Read_Grid(Grid);
   Put_Line("Minimum path sum: " & Integer'Image(Min_Path_Sum(Grid)));
end Euler_83;

procedure Read_Grid(Grid : out Grid_Type) is
   File : File_Type;
   Line : String(1..1000);
   Last : Natural;
begin
   Open(File, In_File, "matrix.txt");
   
   for I in 1..Height loop
      Get_Line(File, Line, Last);
      declare
         Pos : Natural := 1;
         Col : Natural := 1;
      begin
         while Pos <= Last and then Col <= Width loop
            -- Parse integer from line
            declare
               Num_Start : Natural := Pos;
               Num_End   : Natural := Pos;
            begin
               while Num_End <= Last and then Line(Num_End) /= ',' loop
                  Num_End := Num_End + 1;
               end loop;
               
               Grid(I, Col) := Integer'Value(Line(Pos..Num_End-1));
               Col := Col + 1;
               Pos := Num_End + 1;
            end;
         end loop;
      end;
   end loop;
   
   Close(File);
end Read_Grid;

function Min_Path_Sum(Grid : Grid_Type) return Integer is
   type Position is record
      Row, Col : Natural;
   end record;
   
   type Distance is new Integer;
   
   -- Priority queue for Dijkstra's algorithm
   package PQ is new Ada.Priority_Queues(
      Element_Type => Position,
      Priority_Type => Distance,
      "<" => (function (Left, Right : Distance) return Boolean is
                 begin return Left < Right; end)
   );
   
   type Visited_Array is array (1..Height, 1..Width) of Boolean;
   type Distance_Array is array (1..Height, 1..Width) of Integer;
   
   Visited : Visited_Array := (others => (others => False));
   Distances : Distance_Array := (others => (others => Integer'Last));
   PQ : PQ.Queue;
   Current : Position;
   
begin
   -- Initialize distances and priority queue
   Distances(1, 1) := Grid(1, 1);
   PQ.Enqueue((1, 1), Grid(1, 1));
   
   while not PQ.Is_Empty loop
      Current := PQ.Dequeue;
      
      if Visited(Current.Row, Current.Col) then
         continue;
      end if;
      
      Visited(Current.Row, Current.Col) := True;
      
      -- Check all four neighbors
      -- Up
      if Current.Row > 1 and not Visited(Current.Row - 1, Current.Col) then
         declare
            New_Dist : constant Integer := Distances(Current.Row, Current.Col) + Grid(Current.Row - 1, Current.Col);
         begin
            if New_Dist < Distances(Current.Row - 1, Current.Col) then
               Distances(Current.Row - 1, Current.Col) := New_Dist;
               PQ.Enqueue((Current.Row - 1, Current.Col), New_Dist);
            end if;
         end;
      end if;
      
      -- Down
      if Current.Row < Height and not Visited(Current.Row + 1, Current.Col) then
         declare
            New_Dist : constant Integer := Distances(Current.Row, Current.Col) + Grid(Current.Row + 1, Current.Col);
         begin
            if New_Dist < Distances(Current.Row + 1, Current.Col) then
               Distances(Current.Row + 1, Current.Col) := New_Dist;
               PQ.Enqueue((Current.Row + 1, Current.Col), New_Dist);
            end if;
         end;
      end if;
      
      -- Left
      if Current.Col > 1 and not Visited(Current.Row, Current.Col - 1) then
         declare
            New_Dist : constant Integer := Distances(Current.Row, Current.Col) + Grid(Current.Row, Current.Col - 1);
         begin
            if New_Dist < Distances(Current.Row, Current.Col - 1) then
               Distances(Current.Row, Current.Col - 1) := New_Dist;
               PQ.Enqueue((Current.Row, Current.Col - 1), New_Dist);
            end if;
         end;
      end if;
      
      -- Right
      if Current.Col < Width and not Visited(Current.Row, Current.Col + 1) then
         declare
            New_Dist : constant Integer := Distances(Current.Row, Current.Col) + Grid(Current.Row, Current.Col + 1);
         begin
            if New_Dist < Distances(Current.Row, Current.Col + 1) then
               Distances(Current.Row, Current.Col + 1) := New_Dist;
               PQ.Enqueue((Current.Row, Current.Col + 1), New_Dist);
            end if;
         end;
      end if;
   end loop;
   
   return Distances(Height, Width);
end Min_Path_Sum;
```

## Explanation

This solution uses **Dijkstra's algorithm** to find the minimum path sum in a grid where movement is allowed in all four directions:

1. **Data Structures**: 
   - Grid stored as 2D array
   - Priority queue to manage nodes by distance
   - Distance tracking arrays

2. **Algorithm Steps**:
   - Initialize the starting position (1,1) with its value
   - Use priority queue to always process the node with minimum distance
   - For each node, examine all four neighbors (up, down, left, right)
   - Update distances if a shorter path is found
   - Continue until reaching the bottom-right corner

3. **Key Features**:
   - Handles input parsing from file
   - Uses priority queue for efficient shortest path selection
   - Implements proper Dijkstra algorithm with visited tracking
   - Returns the minimum sum to reach the destination

The time complexity is O(Wh log(Wh)) where W and H are grid dimensions, and space complexity is O(Wh).
