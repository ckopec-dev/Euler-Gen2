# Euler Problem 141 Solution in Ada

This problem asks us to find the sum of all positive integers n ≤ 10¹⁶ for which the equation n = q³ + r² holds where q and r are positive integers, and q divides r.

Let me break down the approach:
- We need n = q³ + r² where q|r
- This means r = kq for some positive integer k
- So n = q³ + (kq)² = q³ + k²q² = q²(q + k²)
- Since q divides r and r = kq, we have k ≥ 1

Let's implement this in Ada:

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_141 is
   type Large_Integer is range 0 .. 10**16;
   
   function Is_Perfect_Square(n : Large_Integer) return Boolean is
      Root : Large_Integer;
   begin
      if n = 0 then
         return True;
      end if;
      
      Root := Large_Integer(Sqrt(Real(n)));
      
      -- Check nearby values to handle floating point precision issues
      for i in -1 .. 1 loop
         declare
            r : constant Large_Integer := Root + i;
         begin
            if r * r = n then
               return True;
            end if;
         end;
      end loop;
      
      return False;
   end Is_Perfect_Square;
   
   function GCD(a, b : Large_Integer) return Large_Integer is
      t : Large_Integer;
   begin
      while b /= 0 loop
         t := b;
         b := a mod b;
         a := t;
      end loop;
      return a;
   end GCD;
   
   Total_Sum : Large_Integer := 0;
   Limit     : constant Large_Integer := 10**16;
   
begin
   -- We'll iterate over possible values of q and k
   -- For each valid combination, compute n = q²(q + k²)
   -- Check if this n is a perfect square (as required by the problem)
   
   declare
      Q_Limit : constant Large_Integer := Large_Integer(10**(16.0/3.0)) + 1;
   begin
      for q in 1 .. Q_Limit loop
         exit when q * q * q > Limit;
         
         -- For each q, iterate over possible k values
         -- We need n = q²(q + k²) ≤ Limit
         declare
            Max_K : constant Large_Integer := 
               Large_Integer(Sqrt(Real(Limit / (q * q)) - Real(q))) + 1;
         begin
            for k in 1 .. Max_K loop
               declare
                  n : constant Large_Integer := q * q * (q + k * k);
               begin
                  exit when n > Limit;
                  
                  -- Check if n is a perfect square
                  if Is_Perfect_Square(n) then
                     Total_Sum := Total_Sum + n;
                  end if;
               end;
            end loop;
         end;
      end loop;
   end;
   
   Put_Line("Sum of all valid n: " & Large_Integer'Image(Total_Sum));
end Euler_141;
```
