## Problem Statement

For example, if a = 7 and n = 3, then r = 42: 6³ + 8³ = 216 + 512 = 728 ≡ 42 (mod 49).

      -- Calculate (a-1)^n + (a+1)^n mod a^2
      Power_A_Minus_1 : Long_Long_Integer := 1;
      Power_A_Plus_1 : Long_Long_Integer := 1;
      Result : Long_Long_Integer := 0;
   begin
      for i in 1..n loop
         Power_A_Minus_1 := (Power_A_Minus_1 * (a - 1)) mod (a * a);
         Power_A_Plus_1 := (Power_A_Plus_1 * (a + 1)) mod (a * a);
      end loop;
      
      Result := (Power_A_Minus_1 + Power_A_Plus_1) mod (a * a);
      return Integer(Result);

with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_120 is
   
   function Power_Mod(base, exp, mod : Long_Long_Integer) return Long_Long_Integer is
      Result : Long_Long_Integer := 1;
      Base   : Long_Long_Integer := base mod mod;
   begin
      while exp > 0 loop
         if exp mod 2 = 1 then
            Result := (Result * Base) mod mod;
         end if;
         Base := (Base * Base) mod mod;
         exp := exp / 2;
      end loop;
      return Result;
   end Power_Mod;
   
   function Remainder(a, n : Integer) return Integer is
      A_Squared : Long_Long_Integer := Long_Long_Integer(a) * Long_Long_Integer(a);
      Term1 : Long_Long_Integer;
      Term2 : Long_Long_Integer;
      Result : Long_Long_Integer;
   begin
      Term1 := Power_Mod(Long_Long_Integer(a - 1), Long_Long_Integer(n), A_Squared);
      Term2 := Power_Mod(Long_Long_Integer(a + 1), Long_Long_Integer(n), A_Squared);
      Result := (Term1 + Term2) mod A_Squared;
      return Integer(Result);
   end Remainder;
   
   function Max_Remainder(a : Integer) return Integer is
      Max_Rem : Integer := 0;
      Rem : Integer;
   begin
      -- For each a, find the maximum remainder for n from 1 to 2*a
      -- This is sufficient due to periodicity properties
      for n in 1..2*a loop
         Rem := Remainder(a, n);
         if Rem > Max_Rem then
            Max_Rem := Rem;
         end if;
      end loop;
      return Max_Rem;
   end Max_Remainder;
