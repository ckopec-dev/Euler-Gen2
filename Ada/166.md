# Euler Problem 166 Solution in Ada

This problem asks for the number of 4×4 grids where each row, column, and both diagonals sum to the same value (called the magic sum).

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_166 is
   type Grid is array (1..4, 1..4) of Integer;
   
   -- Function to check if grid is a magic square
   function Is_Magic(g : Grid) return Boolean is
      magic_sum : Integer := g(1,1) + g(1,2) + g(1,3) + g(1,4);
      row_sum   : Integer;
      col_sum   : Integer;
      diag1_sum : Integer := 0;
      diag2_sum : Integer := 0;
   begin
      -- Check rows
      for i in 1..4 loop
         row_sum := 0;
         for j in 1..4 loop
            row_sum := row_sum + g(i,j);
         end loop;
         if row_sum /= magic_sum then
            return False;
         end if;
      end loop;
      
      -- Check columns
      for j in 1..4 loop
         col_sum := 0;
         for i in 1..4 loop
            col_sum := col_sum + g(i,j);
         end loop;
         if col_sum /= magic_sum then
            return False;
         end if;
      end loop;
      
      -- Check diagonals
      for i in 1..4 loop
         diag1_sum := diag1_sum + g(i,i);
         diag2_sum := diag2_sum + g(i,5-i);
      end loop;
      
      return diag1_sum = magic_sum and diag2_sum = magic_sum;
   end Is_Magic;
   
   -- Count valid grids using backtracking
   count : Integer := 0;
   
   procedure Backtrack(g : in out Grid; pos : Integer) is
      row : Integer := (pos - 1) / 4 + 1;
      col : Integer := (pos - 1) mod 4 + 1;
   begin
      if pos > 16 then
         -- Check if we have a valid magic square
         if Is_Magic(g) then
            count := count + 1;
         end if;
         return;
      end if;
      
      -- Try values from 0 to 9 for this position
      for val in 0..9 loop
         g(row,col) := val;
         Backtrack(g, pos + 1);
      end loop;
   end Backtrack;
   
begin
   declare
      grid : Grid := (others => (others => 0));
   begin
      Backtrack(grid, 1);
      Put("Number of 4x4 grids: ");
      Put(count);
      New_Line;
   end;
end Euler_166;
```
