   function Is_Pandigital(S : String) return Boolean is
      Found : array (Character range '1'..'9') of Boolean := (others => False);
      Count : Natural := 0;
   begin
      for I in S'Range loop
         if S(I) >= '1' and S(I) <= '9' then
            if not Found(S(I)) then
               Found(S(I)) := True;
               Count := Count + 1;
            else
               return False; -- Duplicate digit
            end if;
         else
            return False; -- Contains non-digit or zero
         end if;
      end loop;
      
      return Count = 9;
   end Is_Pandigital;
