# Euler Problem 155 Solution in Ada

This problem asks us to find the number of distinct capacitance values that can be obtained by connecting up to 18 capacitors, each with a capacitance of 1 μF, using series and parallel connections.

Let me solve this step by step:

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Containers.Indefinite_Vectors;
with Ada.Containers.Vectors;

procedure Euler_155 is
   type Capacitance is new Float;
   
   package Capacitance_Vectors is new Ada.Containers.Indefinite_Vectors
     (Index_Type => Natural, Element_Type => Capacitance);
   
   package Capacitance_Sets is new Ada.Containers.Vectors
     (Index_Type => Natural, Element_Type => Capacitance);
   
   -- Set of capacitances we can achieve with n capacitors
   type Capacitance_Set is array (0 .. 18) of Capacitance_Sets.Vector;
   
   -- Function to compute parallel combination: C1 + C2
   function Parallel(C1, C2 : Capacitance) return Capacitance is
   begin
      return C1 + C2;
   end Parallel;
   
   -- Function to compute series combination: (C1 * C2) / (C1 + C2)
   function Series(C1, C2 : Capacitance) return Capacitance is
   begin
      if C1 + C2 = 0.0 then
         return 0.0;
      else
         return (C1 * C2) / (C1 + C2);
      end if;
   end Series;
   
   -- Function to add a capacitance value to a set, avoiding duplicates
   procedure Add_Value(S : in out Capacitance_Sets.Vector; Value : Capacitance) is
      Found : Boolean := False;
      Index : Natural;
   begin
      for I in 0 .. Capacitance_Sets.Length(S) - 1 loop
         if abs (Capacitance_Sets.Element(S, I) - Value) < 1.0e-10 then
            Found := True;
            exit;
         end if;
      end loop;
      
      if not Found then
         Capacitance_Sets.Append(S, Value);
      end if;
   end Add_Value;
   
   -- Main computation function
   procedure Compute_Capacitances(Results : in out Capacitance_Set) is
      Temp_Set : Capacitance_Sets.Vector;
   begin
      -- Base case: 1 capacitor gives us 1 μF
      Capacitance_Sets.Append(Results(1), 1.0);
      
      -- For each number of capacitors from 2 to 18
      for N in 2 .. 18 loop
         -- Clear the set for this number of capacitors
         Capacitance_Sets.Clear(Temp_Set);
         
         -- Add single capacitor case (1 μF)
         Add_Value(Temp_Set, 1.0);
         
         -- Try all possible ways to split N capacitors into two groups
         for I in 1 .. N/2 loop
            declare
               J : constant Natural := N - I;
            begin
               -- For each pair of capacitance values from the two groups
               for C1 of Results(I) loop
                  for C2 of Results(J) loop
                     -- Add parallel combination
                     Add_Value(Temp_Set, Parallel(C1, C2));
                     -- Add series combination (only if both are non-zero)
                     if C1 /= 0.0 and C2 /= 0.0 then
                        Add_Value(Temp_Set, Series(C1, C2));
                     end if;
                  end loop;
               end loop;
            end;
         end loop;
         
         -- Copy result back to our main array
         Results(N) := Temp_Set;
      end loop;
   end Compute_Capacitances;
   
   -- Main program
begin
   declare
      Results : Capacitance_Set;
      Count   : Natural := 0;
   begin
      Compute_Capacitances(Results);
      
      -- Count all distinct capacitance values
      for I in 1 .. 18 loop
         Count := Count + Natural(Capacitance_Sets.Length(Results(I)));
      end loop;
      
      Put_Line("Number of distinct capacitance values: " & Integer'Image(Count));
   end;
end Euler_155;
```

```ada
with Ada.Text_IO; use Ada.Text_IO;

procedure Euler_155 is
   -- We'll solve this using dynamic programming approach
   
   -- Define a simple fraction type for exact arithmetic
   type Fraction is record
      Num : Integer;
      Den : Integer;
   end record;
   
   -- Function to get GCD (for normalization)
   function GCD(A, B : Integer) return Integer is
   begin
      if B = 0 then
         return abs A;
      else
         return GCD(B, A mod B);
      end if;
   end GCD;
   
   -- Normalize a fraction
   function Normalize(F : Fraction) return Fraction is
      G : constant Integer := GCD(abs(F.Num), abs(F.Den));
   begin
      if F.Den < 0 then
         return (Num => -F.Num / G, Den => -F.Den / G);
      else
         return (Num => F.Num / G, Den => F.Den / G);
      end if;
   end Normalize;
   
   -- Parallel combination: C1 + C2
   function Parallel(C1, C2 : Fraction) return Fraction is
   begin
      return Normalize((Num => C1.Num * C2.Den + C2.Num * C1.Den,
                        Den => C1.Den * C2.Den));
   end Parallel;
   
   -- Series combination: (C1 * C2) / (C1 + C2)
   function Series(C1, C2 : Fraction) return Fraction is
   begin
      if C1.Num = 0 or C2.Num = 0 then
         return Normalize((Num => 0, Den => 1));
      end if;
      
      return Normalize((Num => C1.Num * C2.Num,
                        Den => C1.Num * C2.Den + C2.Num * C1.Den));
   end Series;
   
   -- Set to track unique capacitance values
   type Capacitance_Set is array (0 .. 18) of 
     Ada.Containers.Indefinite_Vectors.Vector(Character, Fraction);
   
   -- Store results
   Results : Capacitance_Set;
   
   -- Add value to set if not already present
   procedure Add_Value(S : in out Ada.Containers.Indefinite_Vectors.Vector(Character, Fraction);
                       Value : Fraction) is
      Found : Boolean := False;
   begin
      for I in 0 .. Ada.Containers.Indefinite_Vectors.Length(S) - 1 loop
         if S.Element(I).Num = Value.Num and S.Element(I).Den = Value.Den then
            Found := True;
            exit;
         end if;
      end loop;
      
      if not Found then
         Ada.Containers.Indefinite_Vectors.Append(S, Value);
      end if;
   end Add_Value;
   
begin
   -- Initialize base case: 1 capacitor = 1 μF
   Ada.Containers.Indefinite_Vectors.Append(Results(1), Normalize((Num => 1, Den => 1)));
   
   -- For each number of capacitors from 2 to 18
   for N in 2 .. 18 loop
      -- Add single capacitor value (1 μF)
      Add_Value(Results(N), Normalize((Num => 1, Den => 1)));
      
      -- Try all possible splits
      for I in 1 .. N/2 loop
         declare
            J : constant Integer := N - I;
         begin
            for C1 of Results(I) loop
               for C2 of Results(J) loop
                  Add_Value(Results(N), Parallel(C1, C2));
                  if C1.Num /= 0 and C2.Num /= 0 then
                     Add_Value(Results(N), Series(C1, C2));
                  end if;
               end loop;
            end loop;
         end;
      end loop;
   end loop;
   
   -- Count total distinct values
   declare
      Total : Natural := 0;
   begin
      for I in 1 .. 18 loop
         Total := Total + Natural(Ada.Containers.Indefinite_Vectors.Length(Results(I)));
      end loop;
      
      Put_Line("Number of distinct capacitance values: " & Integer'Image(Total));
   end;
end Euler_155;
```
