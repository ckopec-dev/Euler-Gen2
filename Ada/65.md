   function To_Big_Integer(N : Integer) return Big_Integer is
      Result : Big_Integer;

      Result(1) := N;

         Result(I) := 0;
      end loop;
      return Result;
   end To_Big_Integer;
   
   function "+"(Left, Right : Big_Integer) return Big_Integer is
      Result : Big_Integer;
      Carry  : Integer := 0;
   begin
      for I in 1..Result'Last loop
         Result(I) := Left(I) + Right(I) + Carry;
         if Result(I) >= 10 then
            Carry := Result(I) / 10;
            Result(I) := Result(I) mod 10;
         else
            Carry := 0;
         end if;
      end loop;
      return Result;
   end "+";
   
   function "*" (Left : Big_Integer; Right : Integer) return Big_Integer is
      Result : Big_Integer;
      Carry  : Integer := 0;
   begin
      for I in 1..Result'Last loop
         Result(I) := Left(I) * Right + Carry;
         if Result(I) >= 10 then
            Carry := Result(I) / 10;
            Result(I) := Result(I) mod 10;
         else
            Carry := 0;
         end if;
      end loop;
      return Result;
   end "*";

      Result : Big_Integer;
   begin
      for I in 1..Result'Last loop
         Result(I) := 0;
      end loop;
      
      for I in 1..Left'Last loop
         if Left(I) /= 0 then
            declare
               Temp : Big_Integer;
               Carry : Integer := 0;
            begin
               for J in 1..Right'Last loop
                  Temp(J) := Result(J) + Left(I) * Right(J) + Carry;
                  if Temp(J) >= 10 then
                     Carry := Temp(J) / 10;
                     Temp(J) := Temp(J) mod 10;
                  else
                     Carry := 0;
                  end if;
               end loop;
               Result := Temp;
            end;
         end if;
      end loop;
      
      return Result;

                  Temp_A : Big_Integer := Result_A;
                  Temp_B : Big_Integer := Result_B;

                  Result_A := (Result_A * Coeff) + A0;
                  Result_B := (Result_B * Coeff) + B0;
                  
                  A0 := Temp_A;
                  B0 := Temp_B;
               end;
            end;
         end loop;
         
         return Result_A;
      end if;
   end Get_Numerator;

      Temp : Big_Integer := N;

            Total := Total + Temp(I);
         end if;
      end loop;
      return Total;
   end Sum_Of_Digits;
   
   function Sum_Of_Digits_Big(N : Big_Integer) return Integer is

      for I in 1..Temp'Last loop

      return Total;

2. Uses the standard recurrence relation for convergents: 
   - A₀ = 2, A₁ = 3

3. Calculates the 100th convergent's numerator
4. Computes the sum of digits in that numerator

The pattern for coefficients is:
- a₀ = 2
- a₁ = 1, a₂ = 2, a₃ = 1, a₄ = 1, a₅ = 4, a₆ = 1, a₇ = 1, a₈ = 6, ...
- For n ≥ 1: if (n-1) mod 3 = 0, then aₙ = 2 × ((n-1)/3 + 1), else aₙ = 1
