# Euler Problem 143 Solution in Ada

This problem asks us to find the sum of all distinct values of a(n) for n ≤ 100000 where a(n) is the number of ways to write n as a sum of consecutive positive integers.

Let me solve this step by step:

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_143 is
   type Count_Array is array (Positive range <>) of Natural;
   
   -- Function to count number of ways to write n as sum of consecutive positive integers
   function Count_Consecutive_Sums(n : Positive) return Natural is
      count : Natural := 0;
      i : Positive := 1;
      
      -- For a sequence of k consecutive integers starting with i,
      -- the sum is: k*i + k*(k-1)/2 = n
      -- So: i = (n - k*(k-1)/2) / k
      
   begin
      while i * (i + 1) / 2 <= n loop
         declare
            numerator : constant Integer := 2 * n - i * (i - 1);
            denominator : constant Integer := 2 * i;
         begin
            if numerator > 0 and then numerator mod denominator = 0 then
               count := count + 1;
            end if;
         end;
         i := i + 1;
      end loop;
      
      return count;
   end Count_Consecutive_Sums;
   
   -- More efficient approach using divisors
   function Count_Consecutive_Sums_Efficient(n : Positive) return Natural is
      count : Natural := 0;
      k : Positive := 1;
   begin
      while k * (k + 1) / 2 <= n loop
         if (2 * n) mod k = 0 then
            -- If k divides 2*n, check if the resulting starting number is positive
            declare
               start_num : constant Integer := (2 * n) / k - k + 1;
            begin
               if start_num > 0 and then start_num mod 2 = 0 then
                  count := count + 1;
               end if;
            end;
         end if;
         k := k + 1;
      end loop;
      
      return count;
   end Count_Consecutive_Sums_Efficient;
   
   -- Even more efficient approach based on odd divisors
   function a(n : Positive) return Natural is
      count : Natural := 0;
      d : Positive := 1;
   begin
      while d * d <= n loop
         if n mod d = 0 then
            -- Check if d is odd (since we're looking for sequences of odd length)
            if d mod 2 = 1 then
               count := count + 1;
            end if;
            
            -- Check if n/d is odd and not equal to d
            if (n / d) /= d and then (n / d) mod 2 = 1 then
               count := count + 1;
            end if;
         end if;
         d := d + 1;
      end loop;
      
      return count;
   end a;
   
   limit : constant Positive := 100000;
   total_sum : Natural := 0;
   result : array (1..limit) of Natural;
begin
   -- Compute all values up to limit
   for i in 1..limit loop
      result(i) := a(i);
      total_sum := total_sum + result(i);
   end loop;
   
   Put_Line("Sum of all a(n) for n ≤ " & Positive'Image(limit) & ":");
   Put_Line(Positive'Image(total_sum));
end Euler_143;
```
