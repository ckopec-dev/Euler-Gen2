# Euler Problem 115 Solution in Ada

This problem asks us to find the least value of n for which the fill count function exceeds one million, where we can fill a row of length n with blocks of minimum length m.

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_115 is
   
   -- Problem parameters
   M : constant Integer := 50;  -- Minimum block length
   Target : constant Integer := 1_000_000;
   
   -- Array to store computed values (using a reasonable upper bound)
   F : array(0..1000) of Integer := (others => 0);
   
   function Fill_Count(N : Integer) return Integer is
      -- Calculate the number of ways to fill a row of length N
      Result : Integer := 1;  -- Empty arrangement
   begin
      if N < M then
         return Result;
      end if;
      
      -- For each possible position where we can place a block of minimum length M
      for I in M..N loop
         -- Place a block of length I at position 0
         -- Remaining space after this block is (N - I) positions
         -- We can fill the remaining positions in F(N-I) ways
         Result := Result + F(N - I);
      end loop;
      
      return Result;
   end Fill_Count;
   
begin
   -- Initialize base cases
   for N in 0..M-1 loop
      F(N) := 1;
   end loop;
   
   -- Calculate values iteratively
   for N in M..1000 loop
      F(N) := Fill_Count(N);
      
      if F(N) > Target then
         Put("The least value of n for which F(n) exceeds ");
         Put(Target, Width => 0);
         Put(" is: ");
         Put(N, Width => 0);
         New_Line;
         exit;
      end if;
   end loop;
   
end Euler_115;
```
