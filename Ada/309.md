# Euler Problem 309 Solution in Ada

Euler Problem 309 asks us to find the number of integer solutions for a right triangle with integer sides where the perimeter equals the area.

Let's break down the problem:
- We have a right triangle with sides a, b, c (where c is the hypotenuse)
- The perimeter equals the area: a + b + c = (1/2) × a × b
- This gives us: 2(a + b + c) = ab
- Since c = √(a² + b²), we have: 2(a + b + √(a² + b²)) = ab

Here's the Ada solution:

```ada
with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_309 is
   type Point is record
      X, Y : Integer;
   end record;
   
   function Is_Integer(Sq : Long_Long_Integer) return Boolean is
      Root : Long_Long_Integer := Long_Long_Integer(Integer(1));
   begin
      if Sq <= 0 then
         return False;
      end if;
      
      while Root * Root < Sq loop
         Root := Root + 1;
      end loop;
      
      return Root * Root = Sq;
   end Is_Integer;
   
   function Sqrt_Int(Sq : Long_Long_Integer) return Integer is
      Root : Long_Long_Integer := Long_Long_Integer(Integer(1));
   begin
      while Root * Root < Sq loop
         Root := Root + 1;
      end loop;
      
      if Root * Root = Sq then
         return Integer(Root);
      else
         raise Constraint_Error;
      end if;
   end Sqrt_Int;
   
   Count : Integer := 0;
   Max_Perimeter : constant Integer := 1000000;
begin
   -- We need to find integer solutions where perimeter = area
   -- For a right triangle: a + b + c = (1/2) * a * b
   -- This gives us: 2(a + b + c) = ab
   -- With c = sqrt(a² + b²)
   
   for A in 1..Max_Perimeter loop
      exit when 2 * A >= Max_Perimeter;
      
      for B in (A + 1)..Max_Perimeter loop
         exit when 2 * (A + B) >= Max_Perimeter;
         
         declare
            C_Squared : Long_Long_Integer := Long_Long_Integer(A*A) + Long_Long_Integer(B*B);
         begin
            if Is_Integer(C_Squared) then
               declare
                  C : Integer := Sqrt_Int(C_Squared);
                  Perimeter : Integer := A + B + C;
                  Area : Integer := (A * B) / 2;
               begin
                  if Perimeter = Area then
                     Count := Count + 1;
                     Put_Line("Solution: a=" & Integer'Image(A) & 
                              ", b=" & Integer'Image(B) & 
                              ", c=" & Integer'Image(C));
                  end if;
               end;
            end if;
         end;
      end loop;
   end loop;
   
   Put_Line("Number of solutions: ");
   Put(Count);
   New_Line;
end Euler_309;
```
