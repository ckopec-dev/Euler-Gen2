I'll solve Project Euler Problem 299, which asks for the number of integer-sided triangles with perimeter 100,000,000 that have an inscribed circle with integer radius.

with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_299 is
   type Long_Integer is range -2**63 .. 2**63-1;
   
   Perimeter : constant Long_Integer := 100_000_000;
   
   -- Function to compute greatest common divisor using Euclidean algorithm
   function GCD(A, B : Long_Integer) return Long_Integer is
   begin
      if B = 0 then
         return A;
      else
         return GCD(B, A mod B);
      end if;
   end GCD;
   
   -- Function to compute number of valid triangles with given perimeter
   function Count_Triangles(P : Long_Integer) return Long_Integer is
      Count : Long_Integer := 0;
      A, B, C : Long_Integer;
      S : Long_Integer; -- Semi-perimeter
      Area : Long_Integer;
      R : Long_Integer; -- Inradius
   begin
      S := P / 2;
      
      -- For a triangle with sides a, b, c and inradius r:
      -- Area = r * s where s is semi-perimeter
      -- Using Heron's formula: Area = sqrt(s*(s-a)*(s-b)*(s-c))
      -- So: r = sqrt((s-a)*(s-b)*(s-c)/s)
      
      -- We need integer solutions, so (s-a)(s-b)(s-c) must be divisible by s
      -- and the result must be a perfect square
      
      for A in 1 .. P/3 loop
         if (P - A) mod 2 /= 0 then
            -- Skip odd values for B
            continue;
         end if;
         
         B := (P - A) / 2;
         C := P - A - B;
         
         -- Check triangle inequality and that we have a valid triangle
         if A + B > C and A + C > B and B + C > A then
            -- Check if inradius is integer
            declare
               S_A, S_B, S_C : Long_Integer;
               Numerator : Long_Integer;
               Denominator : Long_Integer;
            begin
               S_A := S - A;
               S_B := S - B;
               S_C := S - C;
               
               -- For integer inradius, we need (S_A * S_B * S_C) to be divisible by S
               -- and the quotient to be a perfect square
               Numerator := S_A * S_B * S_C;
               Denominator := S;
               
               if Numerator mod Denominator = 0 then
                  declare
                     Quotient : Long_Integer := Numerator / Denominator;
                     Temp : Long_Integer := 1;
                     Found : Boolean := False;
                  begin
                     -- Check if quotient is a perfect square
                     while Temp * Temp <= Quotient loop
                        if Temp * Temp = Quotient then
                           Found := True;
                           exit;
                        end if;
                        Temp := Temp + 1;
                     end loop;
                     
                     if Found then
                        Count := Count + 1;
                     end if;
                  end;
               end if;
            end;
         end if;
      end loop;
      
      return Count;
   end Count_Triangles;
