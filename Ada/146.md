   -- Miller-Rabin primality test
   function Is_Prime(N : Long_Long) return Boolean is
      -- Handle small cases
      if N < 2 then
         return False;
      elsif N = 2 or N = 3 then
         return True;
      elsif N mod 2 = 0 then
         return False;
      end if;

         D : Long_Long := N - 1;
         R : Integer := 0;

            R := R + 1;
         end loop;

            if A >= N then
               exit;

            begin
               if X /= 1 and X /= N - 1 then
                  for I in 1..R-1 loop
                     X := (X * X) mod N;
                     exit when X = N - 1;
                  end loop;
                  
                  if X /= N - 1 then
                     return False;
                  end if;
               end if;
            end;
         end loop;
         
         return True;
      end;
   end Is_Prime;

         B := (B * B) mod Mod;

      end loop;
      return Result;
