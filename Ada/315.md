## Problem Understanding

The problem asks us to find the number of digits that need to be changed when converting numbers from one form to another, considering how digits are displayed on a digital screen (like a seven-segment display).

with Ada.Text_IO; use Ada.Text_IO;
with Ada.Integer_Text_IO; use Ada.Integer_Text_IO;

procedure Euler_315 is
   
   -- Segment patterns for digits 0-9 on a seven-segment display
   -- Each bit represents a segment (from LSB to MSB: a, b, c, d, e, f, g)
   type Segment_Type is array (0..9) of Integer;
   
   -- Seven-segment patterns (bit positions represent segments)
   Segments : constant Segment_Type := 
     (63,  // 0: abcdef  (all segments except g)
      6,   // 1: bc      (segments b and c)
      91,  // 2: abdeg   (segments a, b, d, e, g)
      79,  // 3: abcdg   (segments a, b, c, d, g)
      102, // 4: bcfg    (segments b, c, f, g)
      109, // 5: acfg    (segments a, c, f, g)
      125, // 6: acdefg  (segments a, c, d, e, f, g)
      7,   // 7: abc     (segments a, b, c)
      127, // 8: abcdefg (all segments)
      111); // 9: abcdfg  (segments a, b, c, d, f, g)
   
   -- Function to count how many segments differ between two digits
   function Segment_Diff(Digit1, Digit2 : Integer) return Integer is
      Diff : Integer := Segments(Digit1) xor Segments(Digit2);
      Count : Integer := 0;
   begin
      -- Count number of bits set in the XOR result
      while Diff > 0 loop
         if Diff mod 2 = 1 then
            Count := Count + 1;
         end if;
         Diff := Diff / 2;
      end loop;
      return Count;
   end Segment_Diff;
   
   -- Function to calculate total segment changes for a number
   function Total_Changes(Number : Integer) return Integer is
      Temp_Number : Integer := Number;
      Digit_Count : Integer := 0;
      Total : Integer := 0;
      Last_Digit : Integer := -1;
   begin
      if Number = 0 then
         return Segment_Diff(0, 0);
      end if;
      
      -- Extract digits and calculate changes
      while Temp_Number > 0 loop
         declare
            Current_Digit : Integer := Temp_Number mod 10;
         begin
            if Last_Digit /= -1 then
               Total := Total + Segment_Diff(Current_Digit, Last_Digit);
            end if;
            Last_Digit := Current_Digit;
            Temp_Number := Temp_Number / 10;
            Digit_Count := Digit_Count + 1;
         end;
      end loop;
      
      return Total;
   end Total_Changes;
   
   -- Function to compute digital root and track changes
   function Digital_Root_Changes(Number : Integer) return Integer is
      Current_Number : Integer := Number;
      Total_Changes : Integer := 0;
   begin
      while Current_Number >= 10 loop
         Total_Changes := Total_Changes + Total_Changes(Current_Number);
         -- Compute digital root step
         declare
            Sum : Integer := 0;
            Temp_Num : Integer := Current_Number;
         begin
            while Temp_Num > 0 loop
               Sum := Sum + (Temp_Num mod 10);
               Temp_Num := Temp_Num / 10;
            end loop;
            Current_Number := Sum;
         end;
      end loop;
      
      -- Add final step for the last digit
      Total_Changes := Total_Changes + Total_Changes(Current_Number);
      
      return Total_Changes;
   end Digital_Root_Changes;
   
   -- Main computation function
   function Compute_Sum(Limit : Integer) return Integer is
      Sum : Integer := 0;
   begin
      for I in 1..Limit loop
         Sum := Sum + Digital_Root_Changes(I);
      end loop;
      return Sum;
   end Compute_Sum;
   
begin

      Result : constant Integer := Compute_Sum(10000000);

      Put("Result: ");
      Put(Result, Width => 0);

   end;
   
end Euler_315;

procedure Euler_315 is

   -- Bit positions: a, b, c, d, e, f, g (MSB to LSB)
   type Digit_Pattern is array (0..9) of Integer;
   
   -- Segments represented as bits:
   -- a=1, b=2, c=4, d=8, e=16, f=32, g=64
   Patterns : constant Digit_Pattern := 
     (63,  // 0: abcdef (all except g) = 1+2+4+8+16+32 = 63
      6,   // 1: bc         = 2+4 = 6  
      91,  // 2: abdeg      = 1+2+8+16+64 = 91
      79,  // 3: abcdg      = 1+2+4+8+64 = 79
      102, // 4: bcfg       = 2+4+32+64 = 102
      109, // 5: aCFG       = 1+4+32+64 = 109
      125, // 6: acdefg     = 1+4+8+16+32+64 = 125
      7,   // 7: abc        = 1+2+4 = 7
      127, // 8: abcdefg    = 1+2+4+8+16+32+64 = 127
      111); // 9: abcdfg     = 1+2+4+8+32+64 = 111
   
   -- Count bits set in an integer (Hamming weight)
   function Bit_Count(N : Integer) return Integer is
      Count : Integer := 0;
      Temp : Integer := N;
   begin
      while Temp > 0 loop
         if Temp mod 2 = 1 then
            Count := Count + 1;
         end if;
         Temp := Temp / 2;
      end loop;
      return Count;
   end Bit_Count;
   
   -- Calculate number of segments that need to change between two digits
   function Segment_Changes(Digit1, Digit2 : Integer) return Integer is
      Diff : Integer := Patterns(Digit1) xor Patterns(Digit2);
   begin
      return Bit_Count(Diff);
   end Segment_Changes;
   
   -- Calculate total segment changes for a number sequence
   function Total_Change_Sequence(Number : Integer) return Integer is
      Temp : Integer := Number;
      Digits : array (0..10) of Integer;  -- Assuming max 10 digits
      Count : Integer := 0;
      Total : Integer := 0;
   begin
      if Number = 0 then
         return 0;
      end if;
      
      -- Extract digits in reverse order
      while Temp > 0 loop
         Digits(Count) := Temp mod 10;
         Count := Count + 1;
         Temp := Temp / 10;
      end loop;
      
      -- Calculate changes between consecutive digits
      for I in 1..Count-1 loop
         Total := Total + Segment_Changes(Digits(I), Digits(I-1));
      end loop;
      
      return Total;
   end Total_Change_Sequence;

## Explanation of Solution

This Ada program solves Euler Problem 315 by:

1. **Defining seven-segment patterns**: Each digit 0-9 has a specific pattern of lit segments (a through g)
2. **Calculating segment differences**: For two digits, XOR their patterns and count the set bits to find how many segments change
3. **Computing total changes**: For each number, calculate the total segment changes needed when transitioning between its digits
4. **Summing all values**: Add up all the segment changes for numbers 1 through the specified limit

The key insight is that we're not computing digital roots per se, but rather measuring how many segments need to change when displaying a sequence of digits in a seven-segment display.

## Key Functions

- `Segment_Changes`: Calculates how many segments change between two digits
- `Total_Change_Sequence`: Computes total changes for a number's digit sequence  
- `Solve`: Computes the sum over all numbers up to the limit

The solution uses bit manipulation for efficient calculation of segment differences, which is crucial for performance when dealing with large ranges.
