procedure Sieve_of_Eratosthenes(Limit : Positive) is
   Is_Prime_Array : array (2..Limit) of Boolean := (others => True);
begin
   for I in 2..Integer(Sqrt(Float(Limit))) loop
      if Is_Prime_Array(I) then
         for J in I*I..Limit by I loop
            Is_Prime_Array(J) := False;
         end loop;
      end if;
   end loop;
   
   -- Store results in global array or use a flag
   for I in 2..Limit loop
      if Is_Prime_Array(I) then
         Put_Line("Prime: " & Integer'Image(I));
      end if;
   end loop;
