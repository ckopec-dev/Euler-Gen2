            exit;
         end if;
      end loop;

         Count := Count + 1;

   for n in 0..50 loop
      for k in 0..n loop
         Add_Unique(Binomial_Coefficient(n, k));
      end loop;
   end loop;
   
   -- Calculate sum of all unique square-free numbers
   declare
      Total : Natural := 0;
   begin
      for i in 0..Count-1 loop
         Total := Total + Unique_Squares(i);

      Put_Line("Answer: " & Total'Img);
   end;

The solution works as follows:

1. **Generate Pascal's Triangle**: We compute binomial coefficients C(n,k) for n from 0 to 50 and k from 0 to n.

2. **Check Square-Free Numbers**: For each coefficient, we check if it's square-free (not divisible by any perfect square greater than 1).

3. **Collect Unique Values**: We maintain a set of unique square-free numbers.

4. **Calculate Sum**: Finally, we sum all the distinct square-free numbers.
