# Euler Problem 150 Solution in Perl

## Problem Statement
In a triangular array of positive and negative numbers, we wish to find a sub-triangle such that the sum of the numbers it contains is the smallest possible.

## Solution Approach
I'll solve this by:
1. Generating the triangular array using the given formula
2. Computing all possible sub-triangles
3. Finding the minimum sum among all sub-triangles

## Perl Implementation

#!/usr/bin/perl
use strict;
use warnings;
use List::Util qw(min);

sub solve {
    my $size = 1000;
    
    # Generate the triangular array using the given formula
    my @triangle;
    my $t = 0;
    
    for my $i (1..$size) {
        for my $j (1..$i) {
            $t = (615949 * $t + 797807) % (2**20);
            $triangle[$i][$j] = $t - 2**19;
        }
    }
    
    my $min_sum = 0;
    my $found = 0;
    
    # Check all possible sub-triangles
    for my $i (1..$size) {
        for my $j (1..$i) {
            my $sum = 0;
            for my $k ($i..$size) {
                $sum += $triangle[$k][$j];
                # Add elements to form the triangle
                my $current_sum = $sum;
                for my $l (1..($k-$i+1)) {
                    if (!$found || $current_sum < $min_sum) {
                        $min_sum = $current_sum;
                        $found = 1;
                    }
                    # Move to next row, shift column
                    if ($j + $l <= $k) {
                        $current_sum += $triangle[$k][$j + $l];
                    }
                }
            }
        }
    }
    
    return $min_sum;
}
