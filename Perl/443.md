# Euler Problem 443 Solution in Perl

I'll solve Euler Problem 443, which asks for the number of ways to write a number as a sum of powers of 2, where each power can be used at most twice.

Let me break down the problem:
- We need to count the number of ways to express a number n as a sum of powers of 2
- Each power of 2 can be used at most twice
- This is equivalent to finding the number of partitions of n using the restricted set {1, 2, 4, 8, 16, ...} where each element appears at most twice

```perl
#!/usr/bin/perl
use strict;
use warnings;

sub count_ways {
    my ($n) = @_;
    
    # Memoization hash
    my %memo;
    
    # Recursive function with memoization
    sub ways {
        my ($target, $max_power) = @_;
        
        # Base cases
        return 1 if $target == 0;
        return 0 if $target < 0 || $max_power < 0;
        
        # Check memo
        my $key = "$target,$max_power";
        return $memo{$key} if exists $memo{$key};
        
        my $count = 0;
        
        # We can use 0, 1, or 2 of the current power
        my $power = 1 << $max_power;  # 2^max_power
        
        # Try using 0, 1, or 2 of this power
        for my $uses (0..2) {
            my $remaining = $target - $uses * $power;
            if ($remaining >= 0) {
                $count += ways($remaining, $max_power - 1);
            }
        }
        
        $memo{$key} = $count;
        return $count;
    }
    
    # Find the highest power needed
    my $max_power = 0;
    my $temp = $n;
    while ($temp > 0) {
        $temp >>= 1;
        $max_power++;
    }
    $max_power--;
    
    return ways($n, $max_power);
}

# Alternative approach using dynamic programming
sub count_ways_dp {
    my ($n) = @_;
    
    # dp[i][j] = number of ways to make sum i using powers up to 2^j
    my @dp = (0) x ($n + 1);
    $dp[0] = 1;  # One way to make 0 (use nothing)
    
    # For each power of 2
    my $power = 1;
    while ($power <= $n) {
        # Process in reverse to avoid counting the same power multiple times
        for my $i ($n .. $power) {
            # Try using 1 or 2 copies of current power
            $dp[$i] += $dp[$i - $power];  # Use 1 copy
            if ($i >= 2 * $power) {
                $dp[$i] += $dp[$i - 2 * $power];  # Use 2 copies
            }
        }
        $power <<= 1;  # Next power of 2
    }
    
    return $dp[$n];
}
```
