sub prime_factorization {
    my $n = shift;

    while ($n % 2 == 0) {
        $factors{2}++;
        $n /= 2;
    }
    
    # Handle odd factors
    my $i = 3;
    while ($i * $i <= $n) {
        while ($n % $i == 0) {
            $factors{$i}++;
            $n /= $i;
        }
        $i += 2;
    }
    
    # If n is still greater than 1, then it's a prime
    if ($n > 1) {
        $factors{$n} = 1;
    }
    
    return %factors;
}

sub smallest_n_for_factorization {

    # For each prime factor and its required power
    for my $prime (keys %factors) {
        my $required_power = $factors{$prime};
        
        # Find the smallest n such that n! contains at least required_power copies of prime
        my $n = $prime;
        while (1) {
            my $power_in_factorial = 0;
            my $temp = $n;
            
            # Count how many times prime divides n!
            while ($temp >= $prime) {

                $temp /= $prime;

                $result = $result > $n ? $result : $n;

        }
    }
    
    return $result;
}
