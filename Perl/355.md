I need to solve Euler Problem 355, which asks for the sum of all distinct safe primes less than 10^8.

Let me first understand what a safe prime is:
- A safe prime is a prime p such that (p-1)/2 is also prime
- The problem asks for the sum of all distinct safe primes less than 10^8

Here's my approach:
1. Generate all primes up to 10^8 using Sieve of Eratosthenes
2. For each prime p, check if (p-1)/2 is also prime

```perl
#!/usr/bin/perl
use strict;
use warnings;

sub sieve_of_eratosthenes {
    my ($limit) = @_;
    my @is_prime = (1) x ($limit + 1);
    $is_prime[0] = $is_prime[1] = 0;
    
    for my $i (2 .. int(sqrt($limit))) {
        if ($is_prime[$i]) {
            for my $j ($i * $i .. $limit step $i) {
                $is_prime[$j] = 0;
            }
        }
    }
    
    my @primes;
    for my $i (2 .. $limit) {
        push @primes, $i if $is_prime[$i];
    }
    
    return @primes;
}

sub is_prime {
    my ($n) = @_;
    return 0 if $n < 2;
    return 1 if $n == 2;
    return 0 if $n % 2 == 0;
    
    for my $i (3 .. int(sqrt($n)) step 2) {
        return 0 if $n % $i == 0;
    }
    return 1;
}

# Generate all primes up to 10^8
my $limit = 100000000;
my @primes = sieve_of_eratosthenes($limit);

my $sum = 0;
my %safe_primes;

    # Skip if p is too large for (p-1)/2 to be meaningful
    last if $p > $limit;

    if ($half > 1 && $half == int($half) && is_prime($half)) {
        $safe_primes{$p} = 1;
        $sum += $p;
    }
}
```

```perl
use warnings;

sub sieve_of_eratosthenes {
    my ($limit) = @_;
    my @is_prime = (1) x ($limit + 1);
    $is_prime[0] = $is_prime[1] = 0;

    }
    
    return @primes;
}

# Generate all primes up to 10^8
my $limit = 100000000;
my @primes = sieve_of_eratosthenes($limit);

# Create a hash for fast prime lookup
my %prime_set;
for my $p (@primes) {
    $prime_set{$p} = 1;
}

my $sum = 0;

# Check each prime to see if it's a safe prime
for my $p (@primes) {
```
