use strict;
use warnings;
use Math::BigInt;

sub count_divisors {
    my $n = shift;
    my $count = 1;
    my $temp = Math::BigInt->new($n);
    
    # Factorize n and count divisors
    my $i = 2;
    while ($i * $i <= $n) {
        my $power = 0;
        while ($temp % $i == 0) {
            $temp /= $i;
            $power++;
        }
        if ($power > 0) {
            $count *= ($power + 1);
        }
        $i++;
    }
    
    if ($temp > 1) {
        $count *= 2;
    }
    
    return $count;
}
