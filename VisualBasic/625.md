    Function SumOfDivisors(n As Long) As Long
        Dim sum As Long = 0
        Dim i As Long = 1
        
        While i * i <= n
            If n Mod i = 0 Then
                sum += i
                If i <> n \ i Then
                    sum += n \ i
                End If
            End If
            i += 1
        End While
        
        Return sum
    End Function
