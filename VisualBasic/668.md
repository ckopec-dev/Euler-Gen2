# Euler Problem 668 Solution in Visual Basic

Euler Problem 668 asks us to find the sum of all integers k from 1 to 10^12 such that k^2 + 1 is divisible by an odd prime p where p ≤ 1000.

Let me break this down:
- We need k^2 + 1 ≡ 0 (mod p) for some odd prime p ≤ 1000
- This means k^2 ≡ -1 (mod p)
- This is equivalent to k^2 ≡ p-1 (mod p)
- For this to have solutions, -1 must be a quadratic residue modulo p

## Visual Basic Implementation

```vb
Module Euler668
    Sub Main()
        Dim limit As Long = 1000000000000L ' 10^12
        Dim maxPrime As Integer = 1000
        Dim primes As List(Of Integer) = GetPrimes(maxPrime)
        Dim result As Long = 0
        
        ' For each odd prime p <= 1000
        For Each p As Integer In primes
            If p = 2 Then Continue For ' Skip even prime
            
            ' Check if -1 is a quadratic residue modulo p
            ' This happens when p ≡ 1 (mod 4)
            If p Mod 4 = 1 Then
                ' Find the smallest positive k such that k^2 ≡ -1 (mod p)
                Dim k As Long = FindSmallestK(p)
                
                ' Add all multiples of k that are <= limit
                Dim count As Long = limit \ k
                result += count * k
                
                ' Also add the other solution (p-k) if it's valid
                Dim otherK As Long = p - k
                If otherK <> k AndAlso otherK <= limit Then
                    count = limit \ otherK
                    result += count * otherK
                End If
            End If
        Next
        
        Console.WriteLine("Result: " & result)
        Console.ReadKey()
    End Sub
    
    Function GetPrimes(maxValue As Integer) As List(Of Integer)
        Dim primes As New List(Of Integer)
        Dim isPrime(maxValue) As Boolean
        
        ' Initialize all numbers as prime
        For i As Integer = 2 To maxValue
            isPrime(i) = True
        Next
        
        ' Sieve of Eratosthenes
        For i As Integer = 2 To Math.Sqrt(maxValue)
            If isPrime(i) Then
                For j As Integer = i * i To maxValue Step i
                    isPrime(j) = False
                Next
            End If
        Next
        
        ' Collect primes
        For i As Integer = 2 To maxValue
            If isPrime(i) Then
                primes.Add(i)
            End If
        Next
        
        Return primes
    End Function
    
    Function FindSmallestK(p As Integer) As Long
        ' Find smallest k such that k^2 ≡ -1 (mod p)
        ' We can use the fact that if p ≡ 1 (mod 4), then k = sqrt(p-1) (mod p)
        ' But since we're looking for the smallest positive k, we need to be more careful
        
        For k As Long = 1 To p - 1
            If (k * k) Mod p = (p - 1) Then
                Return k
            End If
        Next
        
        Return -1 ' Should not happen for valid primes
    End Function
End Module
```

```vb
        Dim maxPrime As Integer = 1000

    Function GetPrimes(maxValue As Integer) As List(Of Integer)
```

```vb
    Function FindK(p As Integer) As Long

        For k As Long = 1 To p - 1
            If (k * k) Mod p = (p - 1) Then
                Return k
            End If
        Next
        
        Return 1 ' Default case
    End Function
```
