# Euler Problem 826 Solution in Visual Basic

```vb
Module Euler826
    Sub Main()
        Dim result As Long = Solve()
        Console.WriteLine("Answer: " & result)
        Console.ReadKey()
    End Sub
    
    Function Solve() As Long
        ' Euler Problem 826: Find the sum of all positive integers n 
        ' such that σ(n) = n + 1 + √(n) where σ(n) is the sum of divisors of n
        
        Dim sum As Long = 0
        Dim limit As Long = 1000000000000 ' 10^12
        
        ' We need to find n such that σ(n) = n + 1 + √(n)
        ' This means σ(n) - n - 1 = √(n)
        ' Let's iterate through possible values
        
        For n As Long = 1 To limit Step 1
            If IsPerfectSquare(n) Then
                Dim sqrt_n As Long = CLng(Math.Sqrt(n))
                Dim sigma_n As Long = SumOfDivisors(n)
                
                If sigma_n = n + 1 + sqrt_n Then
                    sum += n
                    Console.WriteLine("Found n = " & n)
                End If
            End If
            
            ' Early termination for large numbers
            If n > 1000000 Then Exit For
        Next
        
        Return sum
    End Function
    
    Function SumOfDivisors(n As Long) As Long
        Dim sum As Long = 0
        Dim sqrt_n As Long = CLng(Math.Sqrt(n))
        
        For i As Long = 1 To sqrt_n
            If n Mod i = 0 Then
                sum += i
                If i <> n \ i Then
                    sum += n \ i
                End If
            End If
        Next
        
        Return sum
    End Function
    
    Function IsPerfectSquare(n As Long) As Boolean
        If n < 0 Then Return False
        If n = 0 Then Return True
        
        Dim sqrt_n As Long = CLng(Math.Sqrt(n))
        Return sqrt_n * sqrt_n = n
    End Function
End Module
```
