Module Problem459
    Sub Main()
        Dim n As Long = 100000000
        Dim result As Long = Solve(n)
        Console.WriteLine("Result: " & result)
        Console.ReadKey()
    End Sub

    Function Solve(n As Long) As Long
        ' Calculate the sum of all proper divisors for numbers up to n
        Dim divSum(n) As Long
        Dim i As Long, j As Long
        
        ' Initialize divSum array
        For i = 1 To n
            divSum(i) = 1 ' 1 is always a proper divisor for numbers > 1
        Next
        
        ' Calculate proper divisors sum
        For i = 2 To Math.Sqrt(n)
            For j = i * i To n Step i
                If j Mod i = 0 Then
                    If i * i = j Then
                        divSum(j) += i
                    Else
                        divSum(j) += i + (j \ i)
                    End If
                End If
            Next
        Next
        
        ' Count numbers where sigma(n) = sigma(n-1) + 1
        Dim count As Long = 0
        Dim sigma(n) As Long
        
        ' Calculate sigma(n) for all n
        For i = 1 To n
            sigma(i) = divSum(i) + i ' sigma(n) = sum of all divisors including n
        Next
        
        ' Check condition sigma(n) = sigma(n-1) + 1
        For i = 2 To n
            If sigma(i) = sigma(i - 1) + 1 Then
                count += 1
            End If
        Next
        
        Return count
    End Function
End Module
