# Euler Problem 142 Solution in Visual Basic

This problem asks us to find the smallest x > 0 such that x, x + y, and x + z are perfect squares, where y and z are positive integers, and x + y + z is also a perfect square.

    End Sub

    Function FindSmallestX() As Long
        Dim x As Long = 1
        
        While True
            ' Check if x is a perfect square (this is the first condition)
            If IsPerfectSquare(x) Then
                ' For each possible y (where x + y is a perfect square)
                Dim y As Long = 1
                While y * y < x * 2  ' Reasonable upper bound
                    Dim xPlusY As Long = x + y * y
                    If IsPerfectSquare(xPlusY) Then
                        ' For each possible z (where x + z is a perfect square)
                        Dim z As Long = 1
                        While z * z < x * 2
                            Dim xPlusZ As Long = x + z * z
                            If IsPerfectSquare(xPlusZ) Then
                                ' Check if x + y + z is also a perfect square
                                Dim sum As Long = x + y * y + z * z
                                If IsPerfectSquare(sum) Then
                                    Return x
                                End If
                            End If
                            z += 1
                        End While
                    End If
                    y += 1
                End While
            End If
            x += 1
        End While
    End Function

        If num < 0 Then Return False
        If num = 0 Then Return True
