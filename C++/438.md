#include <iostream>
#include <cmath>

using namespace std;

int main() {
    long long limit = 100000000000000000LL;  // 10^17
    
    // All even numbers from 2 to limit
    // Count = limit / 2

}

- Common difference d = 2

Number of terms = (10¹⁷ - 2) / 2 + 1 = 10¹⁷ / 2
