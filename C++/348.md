# Euler Problem 348 Solution in C++

## Problem Statement
A number is called a "digital root" if it equals the sum of the cubes of its digits. For example, 153 = 1³ + 5³ + 3³ = 1 + 125 + 27 = 153.

Find the sum of all numbers that are equal to the sum of the cubes of their digits.

## Solution

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

using namespace std;

// Function to calculate sum of cubes of digits
long long sumOfCubesOfDigits(long long n) {
    long long sum = 0;
    while (n > 0) {
        long long digit = n % 10;
        sum += digit * digit * digit;
        n /= 10;
    }
    return sum;
}

// Function to check if a number equals sum of cubes of its digits
bool isEqualSumOfCubes(long long n) {
    return n == sumOfCubesOfDigits(n);
}

int main() {
    vector<long long> results;
    
    // We need to find numbers that are equal to sum of cubes of their digits
    // Let's search up to a reasonable limit
    for (long long i = 1; i < 1000000; i++) {
        if (isEqualSumOfCubes(i)) {
            results.push_back(i);
        }
    }
    
    // Print all results
    cout << "Numbers equal to sum of cubes of their digits:" << endl;
    for (long long num : results) {
        cout << num << endl;
    }
    
    // Calculate and print the sum
    long long totalSum = 0;
    for (long long num : results) {
        totalSum += num;
    }
    
    cout << "Sum of all such numbers: " << totalSum << endl;
    
    return 0;
}
```
