// Function to calculate sum of digits of a number
long long sumOfDigits(long long n) {

    while (n > 0) {
        sum += n % 10;
        n /= 10;
    }
    return sum;
}

int main() {
    set<long long> distinct_values;
    
    // Fibonacci numbers: F(0)=0, F(1)=1, F(n)=F(n-1)+F(n-2)
    long long prev = 0;
    long long curr = 1;
    
    // Add F(0) = 0
    distinct_values.insert(0);
    
    // Add F(1) = 1
    distinct_values.insert(1);
    
    // Generate Fibonacci numbers and their digit sums
    for (int i = 2; i <= 1000000; i++) {
        long long next = prev + curr;
        
        // Calculate sum of digits of Fibonacci number
        long long digit_sum = sumOfDigits(next);
        distinct_values.insert(digit_sum);
        
        prev = curr;
        curr = next;
        
        // Stop when Fibonacci number becomes too large
        if (next > 1000000000000000LL) break;

    long long result = 0;

## Explanation

This solution:

1. **Generates Fibonacci numbers** using the standard recurrence relation
2. **Calculates sum of digits** for each Fibonacci number
3. **Maintains distinct values** in a set to avoid duplicates
4. **Computes the final sum** of all distinct digit sums

The approach:
- Initialize F(0) = 0 and F(1) = 1
- Generate subsequent Fibonacci numbers using F(n) = F(n-1) + F(n-2)
- For each Fibonacci number, compute the sum of its digits
- Store only distinct digit sums in a set
- Sum all distinct values to get the final answer
