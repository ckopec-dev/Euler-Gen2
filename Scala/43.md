# Euler Problem 43 Solution in Scala

Euler Problem 43 asks us to find the sum of all 0 to 9 pandigital numbers with a specific sub-string divisibility property.

## Problem Statement
The number 1406357289 is a 0 to 9 pandigital number because it uses all digits 0-9 exactly once. The substrings of this number have the following divisibility properties:
- d₂d₃d₄ = 406 is divisible by 2
- d₃d₄d₅ = 063 is divisible by 3
- d₄d₅d₆ = 635 is divisible by 5
- d₅d₆d₇ = 357 is divisible by 7
- d₆d₇d₈ = 572 is divisible by 11
- d₇d₈d₉ = 728 is divisible by 13
- d₈d₉d₁₀ = 289 is divisible by 17

Find the sum of all 0 to 9 pandigital numbers with this property.

## Solution

```scala
import scala.util.Sorting

object Euler43 {
  def main(args: Array[String]): Unit = {
    val result = solve()
    println(s"Sum of all pandigital numbers with the property: $result")
  }
  
  def solve(): Long = {
    // Generate all permutations of digits 0-9
    val digits = (0 to 9).toList
    val permutations = permute(digits)
    
    // Filter permutations that satisfy the divisibility property
    val validNumbers = permutations.filter(isValidDivisibility)
    
    // Convert to numbers and sum them
    validNumbers.map(perm => convertToNumber(perm)).sum
  }
  
  def permute[T](list: List[T]): List[List[T]] = {
    if (list.isEmpty) List(List())
    else {
      for {
        head <- list
        tail <- permute(list diff List(head))
      } yield head :: tail
    }
  }
  
  def convertToNumber(digits: List[Int]): Long = {
    digits.mkString.toLong
  }
  
  def isValidDivisibility(digits: List[Int]): Boolean = {
    // Check divisibility conditions
    // d₂d₃d₄ divisible by 2 (index 1,2,3)
    if ((digits(1) * 100 + digits(2) * 10 + digits(3)) % 2 != 0) return false
    
    // d₃d₄d₅ divisible by 3 (index 2,3,4)
    if ((digits(2) * 100 + digits(3) * 10 + digits(4)) % 3 != 0) return false
    
    // d₄d₅d₆ divisible by 5 (index 3,4,5)
    if ((digits(3) * 100 + digits(4) * 10 + digits(5)) % 5 != 0) return false
    
    // d₅d₆d₇ divisible by 7 (index 4,5,6)
    if ((digits(4) * 100 + digits(5) * 10 + digits(6)) % 7 != 0) return false
    
    // d₆d₇d₈ divisible by 11 (index 5,6,7)
    if ((digits(5) * 100 + digits(6) * 10 + digits(7)) % 11 != 0) return false
    
    // d₇d₈d₉ divisible by 13 (index 6,7,8)
    if ((digits(6) * 100 + digits(7) * 10 + digits(8)) % 13 != 0) return false
    
    // d₈d₉d₁₀ divisible by 17 (index 7,8,9)
    if ((digits(7) * 100 + digits(8) * 10 + digits(9)) % 17 != 0) return false
    
    true
  }
}
```

## Optimized Version

Here's a more efficient approach using the fact that we can generate only valid combinations:

```scala
object Euler43Optimized {
  def main(args: Array[String]): Unit = {
    val result = solve()
    println(s"Sum of all pandigital numbers with the property: $result")
  }
  
  def solve(): Long = {
    // Precompute divisors
    val divisors = List(2, 3, 5, 7, 11, 13, 17)
    
    // Generate all valid permutations using backtracking
    val validNumbers = generateValidNumbers(List(), Set(), divisors, 0)
    
    validNumbers.map(_.mkString.toLong).sum
  }
  
  def generateValidNumbers(current: List[Int], used: Set[Int], divisors: List[Int], index: Int): List[List[Int]] = {
    if (index == 10) {
      // Check if we have a valid 0-9 pandigital
      if (current.length == 10 && used.size == 10) List(current)
      else List()
    } else {
      val results = scala.collection.mutable.ListBuffer[List[Int]]()
      
      // For each digit that hasn't been used yet
      for (digit <- 0 to 9 if !used.contains(digit)) {
        val newCurrent = digit :: current
        val newUsed = used + digit
        
        // Check if this partial sequence satisfies the divisibility property
        if (index >= 3) {
          val substring = newCurrent.take(3).reverse
          val number = substring(0) * 100 + substring(1) * 10 + substring(2)
          val divisor = divisors(index - 3)
          
          if (number % divisor == 0) {
            results ++= generateValidNumbers(newCurrent, newUsed, divisors, index + 1)
          }
        } else {
          results ++= generateValidNumbers(newCurrent, newUsed, divisors, index + 1)
        }
      }
      
      results.toList
    }
  }
}
```

## Final Solution (Most Efficient)

```scala
object Euler43 {
  def main(args: Array[String]): Unit = {
    val result = solve()
    println(s"Sum of all pandigital numbers with the property: $result")
  }
  
  def solve(): Long = {
    val primes = List(2, 3, 5, 7, 11, 13, 17)
    val validNumbers = generateValidPandigitals(primes)
    validNumbers.map(_.mkString.toLong).sum
  }
  
  def generateValidPandigitals(primes: List[Int]): List[List[Int]] = {
    // Generate all valid combinations using recursive approach
    def backtrack(current: List[Int], used: Set[Int], index: Int): List[List[Int]] = {
      if (index == 10) {
        if (used.size == 10) List(current)
        else List()
      } else {
        val results = scala.collection.mutable.ListBuffer[List[Int]]()
        
        for (digit <- 0 to 9 if !used.contains(digit)) {
          val newCurrent = digit :: current
          val newUsed = used + digit
          
          // Check divisibility condition for current substring
          if (index >= 3) {
            val substring = newCurrent.take(3).reverse
            val number = substring(0) * 100 + substring(1) * 10 + substring(2)
            val divisor = primes(index - 3)
            
            if (number % divisor == 0) {
              results ++= backtrack(newCurrent, newUsed, index + 1)
            }
          } else {
            results ++= backtrack(newCurrent, newUsed, index + 1)
          }
        }
        results.toList
      }
    }
    
    backtrack(List(), Set(), 0)
  }
}
```
