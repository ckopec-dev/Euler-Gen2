# Euler Problem 100 Solution in C#

## Problem Statement
Finding the number of blue discs for which there is 50% chance of taking two blue discs.

## Solution

using System;
using System.Numerics;

class Program
{
    static void Main()
    {
        // We need to find the smallest solution to the equation:
        // (b/(b+d)) * ((b-1)/(b+d-1)) = 1/2
        // Which simplifies to: b(b-1) = (b+d)(b+d-1)/2
        // Further simplification gives us: 2b² - 2b = (b+d)² - (b+d)
        // This leads to: b² - b = (b+d)² - (b+d) - b² + b
        // Which becomes: b² - b = (b+d)² - (b+d) - b² + b
        // Simplifying: 2b² - 2b = (b+d)² - (b+d)
        // This is equivalent to: 2b² - 2b = (b+d)(b+d-1)
        
        // Let's use the recurrence relation approach
        // From the Pell equation: x² - 2y² = 1
        // Where b = y and d = x - y
        
        // The recurrence relation for solutions:
        // x(n+1) = 3*x(n) + 2*y(n)
        // y(n+1) = 4*x(n) + 3*y(n)
        // Starting with x₀ = 1, y₀ = 1
        
        BigInteger x = 1;
        BigInteger y = 1;
        BigInteger b = 0;
        BigInteger d = 0;
        
        // Find the first solution where b > 10^12
        while (b <= 1000000000000)
        {
            b = y;
            d = x - y;
            
            // Check if this satisfies our probability condition
            // We need: b(b-1) = (b+d)(b+d-1)/2
            BigInteger left = b * (b - 1);
            BigInteger right = (b + d) * (b + d - 1) / 2;
            
            if (left == right)
            {
                if (b > 1000000000000)
                {
                    Console.WriteLine($"Answer: {b}");
                    break;
                }
            }
            
            // Generate next solution using recurrence relation
            BigInteger newX = 3 * x + 2 * y;
            BigInteger newY = 4 * x + 3 * y;
            x = newX;
            y = newY;
        }
        
        Console.WriteLine($"Answer: {b}");
    }
}
