## Solution Approach

I'll use dynamic programming to compute the number of partitions of n into distinct parts, then sum all distinct values.

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

public class Euler729
{
    public static void Main()
    {
        // Calculate the sum of all distinct values of f(n) where f(n) is 
        // the number of partitions of n into distinct parts
        long result = Solve();
        Console.WriteLine($"Result: {result}");
    }
    
    public static long Solve()
    {
        const int LIMIT = 1000000;
        
        // dp[i] will store the number of partitions of i into distinct parts
        long[] dp = new long[LIMIT + 1];
        dp[0] = 1; // One way to partition 0 (empty partition)
        
        // For each number i, we want to count partitions into distinct parts
        // We can add i to existing partitions that don't contain i
        for (int i = 1; i <= LIMIT; i++)
        {
            // Process in reverse order to avoid using updated values
            for (int j = LIMIT; j >= i; j--)
            {
                if (j >= i)
                {
                    dp[j] += dp[j - i];
                }
            }
        }
        
        // Collect all distinct values of f(n) where n <= LIMIT
        HashSet<long> distinctValues = new HashSet<long>();
        
        for (int i = 1; i <= LIMIT; i++)
        {
            if (dp[i] > 0) // Only consider values where f(n) > 0
            {
                distinctValues.Add(dp[i]);
            }
        }
        
        // Sum all distinct values
        return distinctValues.Sum();
    }
}
```
