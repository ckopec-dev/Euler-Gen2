        }
        
        return total;

    private static BigInteger CountWays(int n)
    {
        // Memoization check
        if (memo.ContainsKey((n, 1, 1)))
            return memo[(n, 1, 1)];

        if (n == 0)
            return 1;
        if (n < 0)
            return 0;

        BigInteger count = 0;
        int power = 1;

        while (power <= n)

            // Use 0, 1, or 2 copies of this power
            for (int uses = 0; uses <= 2 && uses * power <= n; uses++)

                count += CountWays(n - uses * power);
            }
            power *= 2;
        }
        
        // Store in memoization
        memo[(n, 1, 1)] = count;
        return count;
    }

1. We build up the number of ways to represent each number from 0 to the limit
2. For each power of 2 (1, 2, 4, 8, ...), we update our dp array

4. Finally, we sum all the ways to represent numbers from 1 to the limit
