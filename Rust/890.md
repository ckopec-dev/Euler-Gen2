I need to solve Euler Problem 890, which is about finding the number of ways to write a number as a sum of distinct powers of 2, where each power can be used at most twice.

fn solve(n: i64) -> i64 {
    // We need to count the number of ways to write n as sum of powers of 2
    // where each power can be used at most twice
    
    // This is equivalent to finding the number of representations of n
    // in the form: n = sum_{i=0}^{k} a_i * 2^i where a_i in {0,1,2}
    
    // We can use dynamic programming approach
    let mut dp = vec![0i64; (n + 1) as usize];
    dp[0] = 1;
    
    // For each power of 2
    let mut power = 1i64;
    while power <= n {
        // For each number from power to n
        for i in (power..=n).rev() {
            // We can use 0, 1, or 2 copies of this power
            // dp[i] += dp[i - power] (1 copy)
            // dp[i] += dp[i - 2 * power] (2 copies)
            if i >= power {
                dp[i as usize] += dp[(i - power) as usize];
            }
            if i >= 2 * power {
                dp[i as usize] += dp[(i - 2 * power) as usize];
            }
        }
        power *= 2;
    }
    
    dp[n as usize]
}

fn solve_recursive(n: i64) -> i64 {
    let mut memo = std::collections::HashMap::new();
    count_ways(n, 1, &mut memo)
}

fn count_ways(n: i64, power: i64, memo: &mut std::collections::HashMap<(i64, i64), i64>) -> i64 {
    if n == 0 {
        return 1;
    }
    if n < 0 || power > n {
        return 0;
    }
    
    if let Some(&result) = memo.get(&(n, power)) {
        return result;
    }
    
    // We can use 0, 1, or 2 copies of current power
    let result = count_ways(n, power * 2, memo) + 
                 count_ways(n - power, power * 2, memo) + 
                 count_ways(n - 2 * power, power * 2, memo);
    
    memo.insert((n, power), result);
    result
}

fn count_ways(n: i64) -> i64 {
    // dp[i] represents the number of ways to write i as sum of powers of 2
    // where each power can be used at most twice
    let mut dp = vec![0i64; (n + 1) as usize];
    dp[0] = 1;
    
    let mut power = 1i64;
    while power <= n {
        // Process from right to left to avoid double counting
        for i in (power..=n).rev() {
            // Add ways using 1 copy of current power
            if i >= power {
                dp[i as usize] += dp[(i - power) as usize];
            }
            // Add ways using 2 copies of current power  
            if i >= 2 * power {
                dp[i as usize] += dp[(i - 2 * power) as usize];
            }
        }
        power *= 2;
    }
    
    dp[n as usize]
}
