```rust
struct Grid {
    cells: std::collections::HashMap<(i64, i64), bool>, // (x, y) -> is_black
    min_x: i64,
    max_x: i64,
    min_y: i64,
    max_y: i64,
}

impl Grid {
    fn new() -> Self {
        Grid {
            cells: std::collections::HashMap::new(),
            min_x: 0,
            max_x: 0,
            min_y: 0,
            max_y: 0,
        }
    }
    
    fn is_black(&self, x: i64, y: i64) -> bool {
        *self.cells.get(&(x, y)).unwrap_or(&false)
    }
    
    fn toggle_cell(&mut self, x: i64, y: i64) {
        let current = self.cells.get(&(x, y)).unwrap_or(&false);
        self.cells.insert((x, y), !current);
        
        // Update boundaries
        self.min_x = self.min_x.min(x);
        self.max_x = self.max_x.max(x);
        self.min_y = self.min_y.min(y);
        self.max_y = self.max_y.max(y);
    }
    
    fn black_cells_count(&self) -> i64 {
        self.cells.values().filter(|&&is_black| is_black).count() as i64
    }
}

struct Ant {
    x: i64,
    y: i64,
    direction: Direction, // 0: North, 1: East, 2: South, 3: West
}

#[derive(Clone, Copy)]
enum Direction {
    North = 0,
    East = 1,
    South = 2,
    West = 3,
}

impl Ant {
    fn new() -> Self {
        Ant {
            x: 0,
            y: 0,
            direction: Direction::North,
        }
    }
    
    fn move_ant(&mut self, grid: &mut Grid) {
        let is_black = grid.is_black(self.x, self.y);
        
        // Turn right if black, left if white
        if is_black {
            self.direction = self.direction.right();
        } else {
            self.direction = self.direction.left();
        }
        
        // Toggle the cell
        grid.toggle_cell(self.x, self.y);
        
        // Move forward
        match self.direction {
            Direction::North => self.y += 1,
            Direction::East => self.x += 1,
            Direction::South => self.y -= 1,
            Direction::West => self.x -= 1,
        }
    }
}

impl Direction {
    fn right(&self) -> Self {
        match self {
            Direction::North => Direction::East,
            Direction::East => Direction::South,
            Direction::South => Direction::West,
            Direction::West => Direction::North,
        }
    }
    
    fn left(&self) -> Self {
        match self {
            Direction::North => Direction::West,
            Direction::West => Direction::South,
            Direction::South => Direction::East,
            Direction::East => Direction::North,
        }
    }
}
```

```rust
use std::collections::HashMap;

fn main() {

fn solve() -> i64 {
    let mut grid: HashMap<(i64, i64), bool> = HashMap::new();
    let mut ant_x = 0i64;
    let mut ant_y = 0i64;
    let mut direction = 0i32; // 0: North, 1: East, 2: South, 3: West
    
    // Run 10^6 steps
    for _ in 0..1_000_000 {
        let key = (ant_x, ant_y);
        
        // Check if current cell is black
        let is_black = grid.contains_key(&key);
        
        // Turn right if black, left if white
        if is_black {
            direction = (direction + 1) % 4;  // Right turn
        } else {
            direction = (direction + 3) % 4;  // Left turn
        }
        
        // Toggle the cell state
        if is_black {
            grid.remove(&key);
        } else {
            grid.insert(key, true);
        }
        
        // Move forward one step
        match direction {
            0 => ant_y += 1,  // North
            1 => ant_x += 1,  // East
            2 => ant_y -= 1,  // South
            3 => ant_x -= 1,  // West
            _ => panic!("Invalid direction"),
        }
    }
    
    grid.len() as i64
}
```

This solution simulates Langton's Ant for exactly 1,000,000 steps and returns the number of black cells. The key insights are:

1. **Langton's Ant**: The ant follows simple rules - turn right on black cells, left on white cells, then flip the color and move forward
2. **Efficient Storage**: Using a HashMap to store only black cells (white cells are implicitly assumed)
3. **Direction Handling**: Using modular arithmetic to handle direction changes
4. **Pattern Recognition**: After many steps, the ant enters a periodic "highway" pattern, but we need to simulate all steps to get the exact count
