    // dp[i] represents number of ways to form number i
    let mut dp = vec![0u64; limit + 1];
    dp[0] = 1; // One way to form 0 (use nothing)
    
    // For each power of 2
    let mut power = 1;
    while power <= limit {
        // For each number from power to limit
        for i in power..=limit {
            // We can use this power 0, 1, or 2 times

            }
        }

    }
