# Euler Problem 287 Solution in Assembly

Euler Problem 287 asks us to find the number of black pixels in a quadtree representation of a square grid where each pixel is black if the distance from the center is less than or equal to 1/2.

Let me solve this step by step using Assembly language.

## Problem Understanding

We need to:
1. Create a quadtree representation of a 2^20 × 2^20 grid
2. Each pixel is black if distance from center ≤ 1/2
3. Count total black pixels in the quadtree

## Assembly Solution

```assembly
.section .data
    # Constants
    .equ GRID_SIZE, 1048576    # 2^20
    .equ CENTER, 524288        # 2^19
    .equ MAX_DEPTH, 20         # Since 2^20 grid
    
    # Quadtree node structure
    .equ NODE_SIZE, 16         # 4*4 bytes for 4 pointers + 1 byte flag + 3 padding
    
    # Global variables
    .global total_black_pixels
    .global root_node
    
    total_black_pixels: .long 0
    root_node: .long 0

.section .text
    .global _start

# Function to create a quadtree node
create_quadtree_node:
    # Parameters: rdi = grid_size, rsi = x, rdx = y, rcx = depth
    # Returns: rax = node pointer
    
    push rbp
    mov rbp, rsp
    
    # Check if we've reached maximum depth
    cmp ecx, MAX_DEPTH
    jge create_leaf_node
    
    # Calculate half size
    mov r8, rdi
    shr r8, 1                   # r8 = half_size
    
    # Check if this region is entirely black or white
    # For simplicity, we'll implement a recursive approach
    
    # Allocate memory for node
    mov rax, 16                 # Size of node
    call malloc
    
    # Initialize node structure
    mov [rax], rdi              # Store grid_size
    mov [rax+4], rsi            # Store x
    mov [rax+8], rdx            # Store y
    mov [rax+12], ecx           # Store depth
    
    pop rbp
    ret

# Function to determine if a region is entirely black or white
is_region_uniform:
    # Parameters: rdi = grid_size, rsi = x, rdx = y
    # Returns: rax = 0 (white), 1 (black), 2 (mixed)
    
    push rbp
    mov rbp, rsp
    
    # Calculate center of region
    mov r8, rdi
    shr r8, 1                   # half_size
    add r8, rsi                 # center_x = x + half_size
    mov r9, r8
    add r9, rdx                 # center_y = y + half_size
    
    # Calculate distance from center of entire grid
    mov r10, CENTER
    sub r10, r8                 # distance_x
    mov r11, CENTER
    sub r11, r9                 # distance_y
    
    # Calculate squared distance
    imul r10, r10               # distance_x^2
    imul r11, r11               # distance_y^2
    add r10, r11                # squared_distance
    
    # Compare with threshold (1/2)^2 = 1/4
    # We'll use 1/4 * 2^40 = 2^38 as integer comparison
    mov r11, 268435456          # 2^28 (approximation of 1/4 * 2^30)
    cmp r10, r11
    jg is_white                 # If distance^2 > 1/4, it's white
    
    mov rax, 1                  # Black
    jmp is_uniform_end
    
is_white:
    mov rax, 0                  # White
    jmp is_uniform_end
    
is_uniform_end:
    pop rbp
    ret

# Main quadtree traversal function
traverse_quadtree:
    # Parameters: rdi = node pointer
    # Returns: rax = black pixel count
    
    push rbp
    mov rbp, rsp
    
    # Check if node is leaf
    mov rax, [rdi]              # Get grid_size
    cmp rax, 0
    je leaf_node
    
    # If not leaf, traverse children
    mov rax, 0                  # Initialize count
    
    # For each child (recursive call)
    # This is a simplified approach - in practice we'd need proper
    # child node handling
    
    jmp traverse_end
    
leaf_node:
    # Calculate if this leaf is black
    mov rax, [rdi+4]            # x coordinate
    mov rdx, [rdi+8]            # y coordinate
    mov rsi, [rdi]              # grid_size
    
    call is_region_uniform
    cmp rax, 1
    jne traverse_end            # If not black, skip
    
    # Add 1 to black pixel count
    mov rax, [total_black_pixels]
    inc rax
    mov [total_black_pixels], rax
    
traverse_end:
    pop rbp
    ret

# Main function
_start:
    # Initialize
    xor rax, rax
    mov [total_black_pixels], rax
    
    # Create root node for 2^20 grid
    mov rdi, GRID_SIZE          # grid_size
    mov rsi, 0                  # x = 0
    mov rdx, 0                  # y = 0
    mov rcx, 0                  # depth = 0
    
    # Call quadtree creation
    call create_quadtree_node
    
    # Store root node
    mov [root_node], rax
    
    # Traverse and count
    mov rdi, [root_node]
    call traverse_quadtree
    
    # Exit program
    mov rax, 60                 # sys_exit
    mov rdi, 0                  # exit status
    syscall

# Helper function to calculate distance
calculate_distance:
    # Parameters: rdi = x1, rsi = y1, rdx = x2, rcx = y2
    # Returns: rax = squared distance
    
    push rbp
    mov rbp, rsp
    
    # Calculate differences
    mov r8, rdi
    sub r8, rdx                 # x_diff
    mov r9, rsi
    sub r9, rcx                 # y_diff
    
    # Square differences
    imul r8, r8
    imul r9, r9
    add r8, r9                  # squared_distance
    
    mov rax, r8
    pop rbp
    ret

# Alternative implementation using bit manipulation for efficiency
# This is more efficient for large grids
efficient_quadtree_count:
    # Since we're dealing with a 2^20 grid, we can use bit operations
    
    push rbp
    mov rbp, rsp
    
    # For a 2^20 grid, we can calculate the exact number of black pixels
    # using mathematical approach instead of building the tree
    
    # The problem is essentially counting pixels inside a circle
    # with radius 1/2 centered at the grid center
    
    # For 2^20 grid, the center is at (2^19, 2^19)
    # We need to count pixels within distance 1/2 from center
    
    # This can be computed mathematically as:
    # For each row, we compute how many pixels are within the circle
    # and sum them up
    
    xor rax, rax                # total_count
    xor r8, r8                  # row counter
    
    # Loop through rows
count_rows:
    cmp r8, GRID_SIZE
    jge count_end
    
    # Calculate row y coordinate
    mov r9, r8
    sub r9, CENTER              # y_offset from center
    
    # Calculate how many pixels in this row are black
    mov r10, 0                  # pixel_count for this row
    
    # For each pixel in row, check if it's within circle
    xor r11, r11                # x counter
    
check_pixels:
    cmp r11, GRID_SIZE
    jge row_end
    
    # Calculate x_offset from center
    mov r12, r11
    sub r12, CENTER             # x_offset from center
    
    # Calculate squared distance
    imul r9, r9                 # y_offset^2
    imul r12, r12               # x_offset^2
    add r9, r12                 # squared_distance
    
    # Compare with threshold (1/2)^2 = 1/4
    # For integer comparison, we use 1/4 * 2^40
    mov r12, 268435456          # 2^28 (approximation)
    cmp r9, r12
    jg pixel_white
    
    # This pixel is black
    inc r10
    
pixel_white:
    inc r11
    jmp check_pixels
    
row_end:
    add rax, r10                # Add row count to total
    inc r8
    jmp count_rows
    
count_end:
    # Store result
    mov [total_black_pixels], rax
    
    pop rbp
    ret
```
