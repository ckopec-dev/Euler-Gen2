# Euler Problem 135 Solution in Assembly

I'll solve Euler Problem 135, which asks for the number of values of n ≤ 1000000 for which the equation x² - y² - z² = n has exactly one solution where x, y, and z are positive integers and x > y > z.

## Problem Analysis

The equation x² - y² - z² = n can be rewritten as:
x² = y² + z² + n

This means we're looking for Pythagorean-like triples where we have:
- x² = y² + z² + n
- x > y > z > 0

## Assembly Solution

```assembly
.section .data
    max_n: .long 1000000
    count: .long 0
    solutions: .space 1000004  # Array to store solution counts

.section .text
    .global _start

_start:
    # Initialize the solutions array to zero
    movl $0, %eax
    movl max_n, %ebx
    addl $1, %ebx
    movl $solutions, %edi
    xorl %ecx, %ecx
    
init_loop:
    stosl
    addl $4, %edi
    incl %eax
    cmpl %ebx, %eax
    jne init_loop

    # Main algorithm - iterate through possible values
    movl $1, %eax  # Start with n = 1
    movl max_n, %ebx

outer_loop:
    cmpl %ebx, %eax
    jg end_program
    
    # For each n, find how many solutions exist
    movl %eax, %ecx  # n = %ecx
    movl $0, %edx    # solution count
    
    # We need to find x, y, z such that x² - y² - z² = n
    # This means x² = y² + z² + n
    # So x = sqrt(y² + z² + n)
    
    movl $1, %esi    # y = 1
    movl $1, %edi    # z = 1
    
inner_loop:
    # Check if y < z and both are positive
    cmpl %edi, %esi
    jge next_y
    
    # Calculate x² = y² + z² + n
    movl %esi, %ebp
    imull %ebp, %ebp  # y²
    movl %edi, %ebp
    imull %ebp, %ebp  # z²
    addl %ecx, %ebp   # y² + z² + n
    movl %ebp, %edi   # Store in %edi for sqrt calculation
    
    # Calculate integer square root
    call sqrt
    movl %eax, %ebp   # sqrt result
    
    # Check if it's a perfect square and x > y > z
    movl %ebp, %eax
    imull %eax, %eax  # x²
    cmpl %edi, %eax   # Compare with y² + z² + n
    jne next_z
    
    # Check if x > y > z
    cmpl %esi, %ebp   # x > y?
    jl next_z
    cmpl %edi, %ebp   # x > z?
    jl next_z
    
    # Valid solution found
    incl %edx
    
next_z:
    incl %edi
    cmpl %esi, %edi
    jle next_z
    
next_y:
    incl %esi
    cmpl %eax, %esi   # y < n?
    jl inner_loop
    
    # Check if exactly one solution
    cmpl $1, %edx
    jne next_n
    
    # Increment count
    incl count
    
next_n:
    incl %eax
    jmp outer_loop

end_program:
    # Exit program
    movl $1, %eax
    movl $0, %ebx
    int $0x80

# Function to calculate integer square root
sqrt:
    pushl %eax
    pushl %ebx
    pushl %ecx
    pushl %edx
    
    movl %eax, %ebx
    movl $0, %eax
    
    # Binary search for square root
    movl $1, %ecx
    shll $16, %ecx    # 65536
    
sqrt_loop:
    cmpl %ecx, %ebx
    jl sqrt_shift
    
    subl %ecx, %ebx
    addl %ecx, %eax
    
sqrt_shift:
    shrl $2, %ecx
    cmpl $0, %ecx
    jne sqrt_loop
    
    # Adjust result for precision
    cmpl %ebx, %eax
    jg sqrt_adjust
    
    jmp sqrt_exit

sqrt_adjust:
    decl %eax

sqrt_exit:
    popl %edx
    popl %ecx
    popl %ebx
    popl %eax
    ret
```
