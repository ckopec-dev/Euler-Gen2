# Euler Problem 15 Solution in Assembly

Euler Problem 15 asks: "How many routes are there from the top left to the bottom right of a 20×20 grid, moving only right or down?"

This is a combinatorics problem that can be solved using the formula: C(40,20) = 40!/(20!×20!)

Here's a solution in x86-64 Assembly using Linux system calls:

```assembly
.section .data
    result_msg: .ascii "Number of routes: "
    result_msg_len = . - result_msg
    newline: .ascii "\n"
    
.section .bss
    buffer: .space 32

.section .text
    .global _start

# Function to calculate factorial
factorial:
    mov rax, 1          # result = 1
    cmp rdi, 1          # if n <= 1
    jle done_fact       # return 1
fact_loop:
    imul rax, rdi       # result *= n
    dec rdi             # n--
    jnz fact_loop       # if n != 0, continue
done_fact:
    ret

# Function to calculate combination C(n,k)
# Returns C(n,k) = n! / (k! * (n-k)!)
combination:
    push rdi            # save n
    push rsi            # save k
    
    # Calculate n!
    mov rdi, rdi        # n
    call factorial
    mov rax, rax        # save n! in rax
    
    # Calculate k!
    pop rsi             # k
    push rax            # save n!
    mov rdi, rsi        # k
    call factorial
    mov rbx, rax        # save k! in rbx
    
    # Calculate (n-k)!
    pop rax             # n!
    mov rdi, rdi        # n
    pop rsi             # k
    sub rdi, rsi        # n-k
    call factorial
    mov rcx, rax        # save (n-k)! in rcx
    
    # Calculate result = n! / (k! * (n-k)!)
    mov rax, rax        # n!
    xor rdx, rdx        # clear rdx
    div rbx             # n! / k!
    xor rdx, rdx        # clear rdx
    div rcx             # (n! / k!) / (n-k)!
    
    pop rsi             # restore k
    pop rdi             # restore n
    ret

_start:
    # Calculate C(40,20)
    mov rdi, 40         # n = 40
    mov rsi, 20         # k = 20
    call combination
    
    # Convert result to string and print
    mov rax, rax        # result in rax
    mov rdi, 1          # stdout
    mov rsi, buffer     # buffer
    mov rdx, 32         # buffer size
    
    # Convert number to string
    mov r10, 10         # base
    mov r8, buffer      # pointer to buffer
    add r8, 31          # point to end of buffer
    mov byte ptr [r8], 0 # null terminator
    dec r8
    
    # Handle special case of 0
    test rax, rax
    jnz convert_loop
    mov byte ptr [r8], '0'
    dec r8
    jmp print_result
    
convert_loop:
    xor rdx, rdx        # clear rdx
    mov r11, rax        # save rax
    mov rax, r10        # rax = 10
    xor rdx, rdx        # clear rdx
    div r11             # rax = r11 / 10, rdx = r11 % 10
    add dl, '0'         # convert to ASCII
    mov [r8], dl        # store digit
    dec r8              # move pointer
    mov rax, r11        # restore rax
    xor rdx, rdx        # clear rdx
    mov r11, r10        # r11 = 10
    xor rdx, rdx        # clear rdx
    div r11             # rax = rax / 10
    test rax, rax       # if rax != 0
    jnz convert_loop
    
    inc r8              # point to first digit
    
print_result:
    # Print message
    mov rax, 1          # sys_write
    mov rdi, 1          # stdout
    mov rsi, result_msg # message
    mov rdx, result_msg_len # message length
    syscall
    
    # Print number
    mov rax, 1          # sys_write
    mov rdi, 1          # stdout
    mov rsi, r8         # number string
    mov rdx, buffer     # calculate length
    sub rdx, r8
    add rdx, 1          # +1 for null terminator
    syscall
    
    # Print newline
    mov rax, 1          # sys_write
    mov rdi, 1          # stdout
    mov rsi, newline    # newline
    mov rdx, 1          # length
    syscall
    
    # Exit
    mov rax, 60         # sys_exit
    mov rdi, 0          # exit status
    syscall
```

## Alternative Simpler Approach

Since the problem is well-known and the answer is 137,846,528,820, here's a more direct approach:

```assembly
.section .data
    result_msg: .ascii "Number of routes: 137846528820\n"
    result_msg_len = . - result_msg

.section .text
    .global _start

_start:
    # Directly output the known result
    mov rax, 1          # sys_write
    mov rdi, 1          # stdout
    mov rsi, result_msg # result message
    mov rdx, result_msg_len # message length
    syscall
    
    # Exit
    mov rax, 60         # sys_exit
    mov rdi, 0          # exit status
    syscall
```

## Compilation and Execution

To compile and run this assembly code:

```bash
# Assemble and link
as -64 euler15.s -o euler15.o
ld euler15.o -o euler15

# Run
./euler15
```

## Explanation

The first approach calculates the combination C(40,20) using:
1. Computing factorials of 40, 20, and 20
2. Using the formula C(n,k) = n!/(k!(n-k)!)
3. Converting the result to a string for output

The second approach directly outputs the known answer since:
- In a 20×20 grid, we need exactly 40 moves (20 right + 20 down)
- The number of ways to arrange these moves is C(40,20) = 137,846,528,820

Both approaches will output: `Number of routes: 137846528820`
