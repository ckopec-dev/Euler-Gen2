# Euler Problem 272 Solution in Assembly

I'll solve Euler Problem 272 using assembly language. First, let me understand what the problem asks for, then implement it in assembly.

## Problem Understanding

Euler Problem 272 asks for the sum of all positive integers n ≤ 10¹⁶ for which φ(n) divides n, where φ(n) is Euler's totient function.

## Assembly Implementation

```assembly
.section .data
    .align 8
    limit: .quad 10000000000000000    # 10^16
    result: .quad 0
    mod_result: .quad 0

.section .text
    .global _start

_start:
    # Initialize registers
    movq limit(%rip), %rax          # Load limit into rax
    movq $1, %rbx                   # n = 1
    movq $0, %rcx                   # result = 0
    
loop_start:
    # Check if n > limit
    cmpq %rax, %rbx
    jg loop_end
    
    # Calculate phi(n) and check if phi(n) divides n
    movq %rbx, %rdi                 # n in rdi
    call euler_totient
    movq %rax, %r8                  # phi(n) in r8
    
    # Check if n % phi(n) == 0
    movq %rbx, %rax                 # n
    movq %r8, %rbx                  # phi(n)
    xorq %rdx, %rdx                 # Clear rdx for division
    divq %rbx                       # rax = n / phi(n), rdx = remainder
    
    # If remainder is 0, add n to result
    testq %rdx, %rdx
    jnz skip_add
    
    # Add n to result
    movq %rbx, %rax
    addq %rax, %rcx
    
skip_add:
    # Increment n
    incq %rbx
    jmp loop_start

loop_end:
    # Store result
    movq %rcx, result(%rip)
    
    # Exit program
    movq $60, %rax                  # sys_exit
    movq $0, %rdi                   # exit status
    syscall

# Function to calculate Euler's totient function φ(n)
euler_totient:
    # Input: n in %rdi
    # Output: φ(n) in %rax
    movq %rdi, %rax                 # Copy n to result
    movq %rdi, %rbx                 # Copy n to rbx for processing
    
    # Handle special case n = 1
    cmpq $1, %rbx
    je totient_done
    
    # For each prime factor p of n:
    # φ(n) = n * ∏(1 - 1/p) for all prime factors p of n
    # This is equivalent to: φ(n) = n * ∏(p-1)/p for all prime factors p of n
    
    movq $2, %rcx                   # Start with factor 2
    movq $1, %r8                    # result = 1
    
factor_loop:
    # Check if rcx divides rbx
    movq %rbx, %rax
    xorq %rdx, %rdx
    divq %rcx
    
    # If remainder is 0, rcx is a factor
    testq %rdx, %rdx
    jnz next_factor
    
    # rcx is a factor, so multiply result by (rcx-1) and divide by rcx
    movq %r8, %rax
    imulq %rcx, %rax
    decq %rcx
    movq %rax, %r9
    movq %r9, %rax
    xorq %rdx, %rdx
    divq %rcx
    movq %rax, %r8
    
    # Remove all factors of rcx from rbx
    factor_reduction:
        movq %rbx, %rax
        xorq %rdx, %rdx
        divq %rcx
        testq %rdx, %rdx
        jnz factor_loop
        movq %rax, %rbx
        jmp factor_reduction
    
next_factor:
    # Increment factor
    incq %rcx
    # If factor squared > rbx, we're done
    movq %rcx, %rax
    imulq %rax, %rax
    cmpq %rbx, %rax
    jg totient_done
    
    # For odd numbers only (except 2)
    cmpq $2, %rcx
    je factor_loop
    movq %rcx, %rax
    andq $1, %rax
    jz factor_loop
    
    incq %rcx
    jmp factor_loop

totient_done:
    movq %r8, %rax
    ret
```

        xorq %rdx, %rdx
        divq %rdx
        testq %rdx, %rdx
        jnz factor_loop

        jmp remove_factor
