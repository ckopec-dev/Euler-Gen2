# Euler Problem 361 Solution in Assembly

Euler Problem 361 asks us to find the number of ways to tile a 3×n rectangle with 2×1 dominoes.

Let me solve this using dynamic programming in Assembly language.

## Problem Analysis

We need to find the number of ways to tile a 3×n rectangle using 2×1 dominoes. This is a classic dynamic programming problem.

Let's define:
- f(n) = number of ways to tile a 3×n rectangle
- We can use the recurrence relation: f(n) = f(n-1) + 5×f(n-2) + f(n-3) - f(n-4)

## Assembly Implementation

```assembly
.section .data
    # Precomputed values for base cases
    base_values: .long 1, 3, 11, 36
    
.section .text
    .global _start
    
    # Function to calculate number of ways to tile 3×n rectangle
    # Input: n in register r0
    # Output: result in register r0
calculate_tiling:
    # Handle base cases
    cmp $0, r0
    beq base_case_0
    cmp $1, r0
    beq base_case_1
    cmp $2, r0
    beq base_case_2
    cmp $3, r0
    beq base_case_3
    
    # For n >= 4, use recurrence relation
    # f(n) = f(n-1) + 5*f(n-2) + f(n-3) - f(n-4)
    
    # Allocate stack space for our DP array
    sub $16, sp          # Allocate 4 integers (16 bytes)
    
    # Initialize base values in our array
    mov base_values, r1
    str r1, [sp]         # f(0) = 1
    add $4, r1
    str r1, [sp, #4]     # f(1) = 3
    add $4, r1
    str r1, [sp, #8]     # f(2) = 11
    add $4, r1
    str r1, [sp, #12]    # f(3) = 36
    
    # Loop from 4 to n
    mov $4, r2           # i = 4
    mov r0, r3           # n = r0
    
loop:
    cmp r2, r3
    bgt done
    
    # Calculate f(i) = f(i-1) + 5*f(i-2) + f(i-3) - f(i-4)
    # Get f(i-1)
    ldr r4, [sp, #12]    # f(i-1)
    
    # Get f(i-2)
    ldr r5, [sp, #8]     # f(i-2)
    mul r5, r5, #5       # 5*f(i-2)
    
    # Get f(i-3)
    ldr r6, [sp, #4]     # f(i-3)
    
    # Get f(i-4)
    ldr r7, [sp]         # f(i-4)
    
    # Calculate f(i) = f(i-1) + 5*f(i-2) + f(i-3) - f(i-4)
    add r4, r4, r5       # f(i-1) + 5*f(i-2)
    add r4, r4, r6       # f(i-1) + 5*f(i-2) + f(i-3)
    sub r4, r4, r7       # f(i-1) + 5*f(i-2) + f(i-3) - f(i-4)
    
    # Shift values in array
    ldr r5, [sp, #4]     # f(i-3)
    ldr r6, [sp, #8]     # f(i-2)
    ldr r7, [sp, #12]    # f(i-1)
    
    str r5, [sp]         # Shift f(i-3) to f(i-4)
    str r6, [sp, #4]     # Shift f(i-2) to f(i-3)
    str r7, [sp, #8]     # Shift f(i-1) to f(i-2)
    str r4, [sp, #12]    # Put new f(i) in f(i-1)
    
    add r2, r2, #1       # i++
    b loop
    
done:
    ldr r0, [sp, #12]    # Return f(n)
    add $16, sp          # Deallocate stack space
    bx lr

# Main program
_start:
    # Example: Calculate for n = 12 (the actual Euler problem)
    mov $12, r0
    bl calculate_tiling
    
    # Exit program
    mov $1, r7           # sys_exit
    mov $0, r0           # exit status
    svc #0
```

The recurrence relation for tiling a 3×n rectangle with 2×1 dominoes is:
- f(0) = 1
- f(1) = 3  
- f(2) = 11
- f(3) = 36
- f(n) = f(n-1) + 5×f(n-2) + f(n-3) - f(n-4) for n ≥ 4
