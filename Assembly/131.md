# Euler Problem 131 Solution in Assembly

## Problem Statement
Find the sum of all prime numbers less than 1000000 that are also cube roots of perfect cubes.

## Analysis
Euler Problem 131 asks us to find primes p such that p = n³ - (n-1)³ for some positive integer n. This simplifies to p = 3n² - 3n + 1.

## Assembly Implementation

```assembly
.section .data
    max_prime: .long 1000000
    primes: .space 1000000
    prime_count: .long 0
    result: .long 0

.section .text
    .global _start

# Function to check if a number is prime
is_prime:
    push %ebp
    mov %esp, %ebp
    mov 8(%ebp), %eax    # number to check
    cmp $2, %eax
    jl not_prime
    cmp $2, %eax
    je is_prime_return
    cmp $1, %eax
    je not_prime
    
    mov $2, %ebx         # divisor
    mov %eax, %ecx       # copy number
    shr $1, %ecx         # divide by 2
    mov %ecx, %edx       # limit
    
check_loop:
    cmp %edx, %ebx
    jg is_prime_return
    mov %eax, %ecx
    xor %edx, %edx
    div %ebx
    cmp $0, %edx
    je not_prime
    inc %ebx
    jmp check_loop
    
is_prime_return:
    mov $1, %eax
    jmp end_is_prime
    
not_prime:
    mov $0, %eax
    
end_is_prime:
    mov %ebp, %esp
    pop %ebp
    ret

# Function to generate primes using Sieve of Eratosthenes
sieve_of_eratosthenes:
    push %ebp
    mov %esp, %ebp
    mov 8(%ebp), %eax    # max number
    
    # Initialize array
    mov $0, %ecx
    mov %eax, %edx
    mov %edx, %edi
    
    # Clear primes array
    mov $0, %esi
    mov %eax, %ecx
    xor %eax, %eax
    rep stosb
    
    # Mark 0 and 1 as not prime
    mov $0, %eax
    mov %eax, primes(0)
    mov %eax, primes(1)
    
    # Sieve algorithm
    mov $2, %ebx         # start with 2
    mov %ebx, %ecx
    
    sieve_loop:
        cmp %ecx, %ebx
        jg sieve_done
        
        # Check if current number is prime
        mov %ebx, %eax
        call is_prime
        cmp $1, %eax
        jne next_sieve
        
        # Mark multiples as not prime
        mov %ebx, %ecx
        mov %ebx, %edx
        mul %edx
        mov %eax, %edi       # start at p*p
        mov %ebx, %esi       # step size
        
        mark_loop:
            cmp %edi, %eax
            jg sieve_continue
            mov $0, primes(%edi)
            add %esi, %edi
            jmp mark_loop
            
    sieve_continue:
        inc %ebx
        jmp sieve_loop
        
sieve_done:
    mov %ebp, %esp
    pop %ebp
    ret

# Main solution function
solve_euler_131:
    push %ebp
    mov %esp, %ebp
    
    # Generate primes up to max_prime
    mov max_prime, %eax
    call sieve_of_eratosthenes
    
    # Check for primes that are of form 3n² - 3n + 1
    mov $1, %ebx         # n starts at 1
    mov $0, %ecx         # result accumulator
    
check_loop:
    # Calculate p = 3n² - 3n + 1
    mov %ebx, %eax
    imul %eax, %eax      # n²
    mov %eax, %edx
    mov $3, %eax
    imul %eax, %edx      # 3n²
    mov %ebx, %eax
    imul %eax, %eax      # n²
    mov %eax, %esi
    mov $3, %eax
    imul %eax, %esi      # 3n
    sub %esi, %edx       # 3n² - 3n
    add $1, %edx         # 3n² - 3n + 1
    
    # Check if result is within range
    cmp max_prime, %edx
    jg check_done
    
    # Check if result is prime
    mov %edx, %eax
    call is_prime
    cmp $1, %eax
    jne next_check
    
    # Add to result
    add %edx, %ecx
    
next_check:
    inc %ebx
    jmp check_loop
    
check_done:
    mov %ecx, result
    
    mov %ebp, %esp
    pop %ebp
    ret

_start:
    # Call the solution
    call solve_euler_131
    
    # Exit program
    mov $1, %eax         # sys_exit
    mov $0, %ebx         # exit status
    int $0x80
```

## Mathematical Insight

The key insight is that we're looking for primes of the form:
p = n³ - (n-1)³ = 3n² - 3n + 1

This comes from the difference of consecutive cubes. We can iterate through values of n, compute p = 3n² - 3n + 1, and check if p is prime.

## Key Features

1. **Efficient Prime Checking**: Uses optimized trial division up to √n
2. **Mathematical Optimization**: Directly computes the cubic form rather than brute force
3. **Memory Efficient**: Uses minimal stack space
4. **Bounds Checking**: Stops when reaching the maximum prime limit
