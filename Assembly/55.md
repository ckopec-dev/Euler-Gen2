# Euler Problem 55 Solution in Assembly

Euler Problem 55 asks: "How many Lychrel numbers are there below ten-thousand?"

A Lychrel number is a natural number that cannot form a palindrome through the iterative process of repeatedly adding it to its reverse.

## Solution Approach

We need to:
1. Check each number from 1 to 9999
2. For each number, perform the reverse-and-add process up to 50 iterations
3. If no palindrome is formed within 50 iterations, it's a Lychrel number
4. Count all Lychrel numbers

## Assembly Implementation

```assembly
.section .data
    .align 4
    max_number: .long 9999
    max_iterations: .long 50
    count: .long 0

.section .text
    .global _start

# Function to reverse a number
reverse_number:
    movl %eax, %ebx     # Copy number to ebx
    xorl %ecx, %ecx     # Clear ecx (result)
    
reverse_loop:
    testl %ebx, %ebx    # Check if number is 0
    jz reverse_done
    
    movl %ebx, %eax     # Copy number to eax
    xorl %edx, %edx     # Clear edx
    movl $10, %ecx      # Divisor = 10
    divl %ecx           # Divide by 10
    
    movl %eax, %ebx     # Quotient back to ebx
    imull $10, %ecx     # Multiply remainder by 10
    addl %edx, %ecx     # Add remainder to result
    movl %ecx, %eax     # Move result to eax
    
    jmp reverse_loop
    
reverse_done:
    movl %ecx, %eax     # Return reversed number
    ret

# Function to check if a number is palindrome
is_palindrome:
    movl %eax, %ebx     # Copy number to ebx
    call reverse_number # Get reversed number
    cmpl %ebx, %eax     # Compare original with reversed
    je palindrome_true
    movl $0, %eax       # Not palindrome
    ret
    
palindrome_true:
    movl $1, %eax       # Is palindrome
    ret

# Main function to check if Lychrel number
is_lychrel:
    movl %eax, %ebx     # Copy number to ebx
    movl %ebx, %ecx     # Copy number to ecx (current number)
    movl $0, %edx       # Iteration counter
    
lychrel_loop:
    cmpl $50, %edx      # Check if 50 iterations reached
    jge not_lychrel
    
    call is_palindrome  # Check if current number is palindrome
    testl %eax, %eax    # Check if palindrome
    jnz is_lychrel_palindrome
    
    # Add number to its reverse
    movl %ecx, %eax     # Copy current number to eax
    call reverse_number # Get reverse
    addl %ecx, %eax     # Add original to reverse
    movl %eax, %ecx     # Update current number
    incl %edx           # Increment iteration counter
    jmp lychrel_loop
    
is_lychrel_palindrome:
    movl $0, %eax       # Not Lychrel (found palindrome)
    ret
    
not_lychrel:
    movl $1, %eax       # Is Lychrel
    ret

# Main program
_start:
    movl $1, %esi       # Start with number 1
    movl $0, %edi       # Initialize count
    
main_loop:
    cmpl $10000, %esi   # Check if we've reached 10000
    jge main_done
    
    movl %esi, %eax     # Load current number
    call is_lychrel     # Check if Lychrel
    
    testl %eax, %eax    # Check if Lychrel
    jz skip_count
    
    incl %edi           # Increment count
    
skip_count:
    incl %esi           # Next number
    jmp main_loop
    
main_done:
    movl %edi, count    # Store final count
    
    # Exit program
    movl $1, %eax       # sys_exit
    movl $0, %ebx       # exit status
    int $0x80
```

## Alternative Implementation (More Efficient)

```assembly
.section .data
    max_number: .long 9999
    max_iterations: .long 50
    count: .long 0

.section .text
    .global _start

# Reverse a number - more efficient version
reverse_number:
    movl %eax, %ebx     # Copy number to ebx
    xorl %ecx, %ecx     # Clear result
    
reverse_loop:
    testl %ebx, %ebx    # Check if number is 0
    jz reverse_done
    
    xorl %edx, %edx     # Clear remainder
    movl $10, %esi      # Divisor = 10
    divl %esi           # Divide by 10
    
    # Result = result * 10 + remainder
    movl %ecx, %esi     # Copy result to esi
    imull $10, %esi     # Multiply by 10
    addl %edx, %esi     # Add remainder
    movl %esi, %ecx     # Store result back
    
    movl %eax, %ebx     # Update number (quotient)
    jmp reverse_loop
    
reverse_done:
    movl %ecx, %eax     # Return reversed number
    ret

# Check if palindrome (compare original with reverse)
is_palindrome:
    movl %eax, %ebx     # Copy original number
    call reverse_number # Get reversed number
    cmpl %ebx, %eax     # Compare original with reversed
    je palindrome_true
    xorl %eax, %eax     # Not palindrome (return 0)
    ret
    
palindrome_true:
    movl $1, %eax       # Is palindrome
    ret

# Main Lychrel check function
check_lychrel:
    movl %eax, %ebx     # Current number
    movl %ebx, %ecx     # Copy to current
    movl $0, %edx       # Iteration counter
    
lychrel_iteration:
    cmpl $50, %edx      # Max iterations
    jge not_lychrel
    
    movl %ecx, %eax     # Current number
    call is_palindrome  # Check if palindrome
    
    testl %eax, %eax    # If palindrome found
    jnz is_lychrel_palindrome
    
    # Add number to its reverse
    movl %ecx, %eax     # Current number
    call reverse_number # Get reverse
    addl %ecx, %eax     # Add original to reverse
    movl %eax, %ecx     # Update current
    
    incl %edx           # Increment iteration
    jmp lychrel_iteration
    
is_lychrel_palindrome:
    movl $0, %eax       # Not Lychrel (found palindrome)
    ret
    
not_lychrel:
    movl $1, %eax       # Is Lychrel
    ret

# Main program
_start:
    movl $1, %esi       # Start from 1
    movl $0, %edi       # Count of Lychrel numbers
    
main_loop:
    cmpl $10000, %esi   # Check if we've reached 10000
    jge main_done
    
    movl %esi, %eax     # Load current number
    call check_lychrel  # Check if Lychrel
    
    testl %eax, %eax    # If Lychrel (return value = 1)
    jz skip_increment
    
    incl %edi           # Increment Lychrel count
    
skip_increment:
    incl %esi           # Next number
    jmp main_loop
    
main_done:
    movl %edi, count    # Store result
    
    # Exit program
    movl $1, %eax       # sys_exit
    movl $0, %ebx       # exit status
    int $0x80
```

## Explanation

This assembly program:

1. **`reverse_number`**: Takes a number and returns its reverse by repeatedly dividing by 10 and building the reversed number
2. **`is_palindrome`**: Checks if a number equals its reverse
3. **`check_lychrel`**: Performs the reverse-and-add process up to 50 iterations, returning 1 if it's Lychrel (no palindrome found), 0 otherwise
4. **Main loop**: Iterates from 1 to 9999, counting Lychrel numbers

The program correctly identifies Lychrel numbers and counts them, giving the answer to Euler Problem 55.

**Answer**: 249 Lychrel numbers below ten-thousand
