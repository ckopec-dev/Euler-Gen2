# Euler Problem 152 Solution in Assembly

Euler Problem 152 asks us to find the number of ways to write 1/2 as a sum of inverse squares, where each inverse square 1/n² is used at most once.

Let me solve this using Assembly language (x86-64 AT&T syntax):

```assembly
.section .data
    # Precomputed inverse squares as fractions (numerator, denominator)
    # We'll use the fact that we only need to consider n up to 100
    # since 1/100² = 1/10000, and 1/2 = 5000/10000
    inverse_squares:
        .quad 1, 1      # 1/1²
        .quad 1, 4      # 1/2²
        .quad 1, 9      # 1/3²
        .quad 1, 16     # 1/4²
        .quad 1, 25     # 1/5²
        .quad 1, 36     # 1/6²
        .quad 1, 49     # 1/7²
        .quad 1, 64     # 1/8²
        .quad 1, 81     # 1/9²
        .quad 1, 100    # 1/10²
        # ... continue up to 1/100²
        .quad 1, 10000  # 1/100²

    target_numerator: .quad 1
    target_denominator: .quad 2
    count: .quad 0
    max_n: .quad 100

.section .text
    .global _start

# Function to compute GCD
gcd:
    # Input: rdi = a, rsi = b
    # Output: rax = gcd(a,b)
    cmp $0, %rsi
    je gcd_done
    mov %rsi, %rax
    mov %rdi, %rsi
    mov %rax, %rdi
    jmp gcd
gcd_done:
    mov %rdi, %rax
    ret

# Function to add two fractions (a/b + c/d)
add_fractions:
    # Input: rdi = a, rsi = b, rdx = c, rcx = d
    # Output: rax = numerator, rdi = denominator
    # Result: (a*d + b*c) / (b*d)
    mov %rdi, %rax      # a
    imul %rcx, %rax     # a*d
    mov %rsi, %r8       # b
    imul %rdx, %r8      # b*c
    add %r8, %rax       # a*d + b*c (numerator)
    
    mov %rsi, %r8       # b
    imul %rcx, %r8      # b*d (denominator)
    mov %r8, %rdi       # denominator
    
    # Reduce fraction to lowest terms
    mov %rax, %r8       # save numerator
    mov %rdi, %r9       # save denominator
    call gcd
    mov %rax, %r10      # gcd
    mov %r8, %rax       # numerator
    xor %rdx, %rdx
    div %r10            # numerator / gcd
    mov %rax, %r8       # reduced numerator
    mov %r9, %rax       # denominator
    xor %rdx, %rdx
    div %r10            # denominator / gcd
    mov %rax, %rdi      # reduced denominator
    mov %r8, %rax       # reduced numerator
    ret

# Function to subtract two fractions (a/b - c/d)
sub_fractions:
    # Input: rdi = a, rsi = b, rdx = c, rcx = d
    # Output: rax = numerator, rdi = denominator
    # Result: (a*d - b*c) / (b*d)
    mov %rdi, %rax      # a
    imul %rcx, %rax     # a*d
    mov %rsi, %r8       # b
    imul %rdx, %r8      # b*c
    sub %r8, %rax       # a*d - b*c (numerator)
    
    mov %rsi, %r8       # b
    imul %rcx, %r8      # b*d (denominator)
    mov %r8, %rdi       # denominator
    
    # Reduce fraction to lowest terms
    mov %rax, %r8       # save numerator
    mov %rdi, %r9       # save denominator
    call gcd
    mov %rax, %r10      # gcd
    mov %r8, %rax       # numerator
    xor %rdx, %rdx
    div %r10            # numerator / gcd
    mov %rax, %r8       # reduced numerator
    mov %r9, %rax       # denominator
    xor %rdx, %rdx
    div %r10            # denominator / gcd
    mov %rax, %rdi      # reduced denominator
    mov %r8, %rax       # reduced numerator
    ret

# Recursive backtracking function to find solutions
solve:
    # Input: rdi = current_sum_numerator, rsi = current_sum_denominator, rdx = index
    # Returns: number of solutions found
    
    # Base case: if index >= max_n, check if we have 1/2
    cmp %rdx, %max_n
    jge check_solution
    
    # Get current fraction 1/n²
    mov %rdx, %rax
    imul %rax, %rax     # n*n
    mov %rax, %rcx      # denominator of current fraction
    
    # Check if we can add this fraction
    # We want to avoid going over 1/2
    mov %rdi, %r8       # current sum numerator
    mov %rsi, %r9       # current sum denominator
    
    # Add current fraction to sum
    mov %r8, %rax       # sum numerator
    mov %r9, %r10       # sum denominator
    mov $1, %r11        # current fraction numerator
    mov %rcx, %r12      # current fraction denominator
    
    # Add fractions: sum + 1/n²
    call add_fractions
    # Result in rax (numerator), rdi (denominator)
    
    # Check if this sum equals 1/2
    mov %rax, %r8       # sum numerator
    mov %rdi, %r9       # sum denominator
    mov $1, %r10        # target numerator
    mov $2, %r11        # target denominator
    
    # Compare fractions: sum vs 1/2
    mov %r8, %rax
    imul %r11, %rax     # sum_numerator * target_denominator
    mov %r9, %r12
    imul %r10, %r12     # sum_denominator * target_numerator
    cmp %r12, %rax      # compare cross products
    je found_solution
    
    # Continue with backtracking
    # Try without this fraction (subtract it)
    mov %rdi, %r8       # current sum numerator
    mov %rsi, %r9       # current sum denominator
    mov $1, %r10        # current fraction numerator
    mov %rcx, %r11      # current fraction denominator
    
    # Subtract current fraction from sum
    call sub_fractions
    # Result in rax (numerator), rdi (denominator)
    
    # Continue with next index
    inc %rdx
    call solve
    ret

# Check if current sum equals target (1/2)
check_solution:
    # Compare current_sum with 1/2
    # If equal, increment count
    mov %rdi, %rax      # sum numerator
    mov %rsi, %r8       # sum denominator
    mov $1, %r9         # target numerator
    mov $2, %r10        # target denominator
    
    # Cross multiply to compare: sum_numerator * 2 = sum_denominator * 1
    mov %rax, %r11
    imul $2, %r11       # sum_numerator * 2
    mov %r8, %r12
    imul $1, %r12       # sum_denominator * 1
    
    cmp %r12, %r11
    jne not_found
    
    # Found a solution, increment count
    mov %count, %rax
    inc %rax
    mov %rax, %count
    ret

not_found:
    ret

# Main function
_start:
    # Initialize
    mov $0, %rax        # start with sum = 0
    mov $1, %rbx        # start with sum = 0/1
    mov $0, %rcx        # start at index 0
    
    # Call solve function
    call solve
    
    # Exit program
    mov $60, %rax       # sys_exit
    mov $0, %rdi        # exit status
    syscall
```

## Explanation

This is a simplified approach to Euler Problem 152 using assembly language. The problem asks us to find how many ways we can write 1/2 as a sum of distinct inverse squares 1/n².

The key components are:

1. **Fraction arithmetic**: We implement addition and subtraction of fractions with GCD reduction
2. **Backtracking**: We recursively try including or excluding each inverse square
3. **Base case**: When we've considered all numbers, we check if our sum equals 1/2

## Key Concepts

- **Fraction representation**: Each fraction is stored as (numerator, denominator)
- **GCD computation**: Used to reduce fractions to lowest terms
- **Cross multiplication**: Used to compare fractions without floating point operations
- **Recursive backtracking**: Systematically explores all combinations
