export default class Euler458 extends LightningElement {
    result = 0;
    calculationSteps = [];

    connectedCallback() {
        this.solve();
    }

    solve() {
        // Euler Problem 458: Tiling a 3×n rectangle with 2×1 dominoes
        // This is a classic dynamic programming problem
        const n = 1000000;
        const MOD = 1000000007;
        
        // For a 3×n rectangle, we can use the recurrence relation:
        // f(n) = f(n-1) + 5*f(n-2) + f(n-3) - f(n-4)
        // Base cases:
        // f(0) = 1, f(1) = 1, f(2) = 5, f(3) = 11
        
        if (n === 0) {
            this.result = 1;
            return;
        }
        
        // We'll use matrix exponentiation for efficiency
        // But for demonstration, let's compute it step by step
        this.calculateTiling(n, MOD);
    }

    calculateTiling(n, MOD) {
        if (n === 0) {
            this.result = 1;
            return;
        }
        
        if (n === 1) {
            this.result = 1;
            return;
        }
        
        if (n === 2) {
            this.result = 5;
            return;
        }
        
        if (n === 3) {
            this.result = 11;
            return;
        }
        
        // Use the recurrence relation: f(n) = f(n-1) + 5*f(n-2) + f(n-3) - f(n-4)
        let a = 1;  // f(0)
        let b = 1;  // f(1)
        let c = 5;  // f(2)
        let d = 11; // f(3)
        
        for (let i = 4; i <= n; i++) {
            let next = (a + (5 * b) % MOD + c - d + MOD) % MOD;
            a = b;
            b = c;
            c = d;
            d = next;
        }
        
        this.result = d;
    }

    get displayResult() {
        return `The number of ways to tile a 3×${1000000} rectangle with 2×1 dominoes is: ${this.result}`;
