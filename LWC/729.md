# Euler Problem 729 Solution in Lightning Web Component

## Problem Statement
Euler Problem 729 asks us to find the sum of all numbers n ≤ 10^10 such that φ(n) = φ(n+1) where φ is Euler's totient function.

import { LightningElement } from 'lwc';

    result = 0;
    isCalculating = false;
    calculationSteps = [];

    handleCalculate() {
        this.isCalculating = true;
        this.result = 0;
        this.calculationSteps = [];
        
        // For demonstration, we'll calculate a smaller range

                sum += n;
                count++;
                this.logStep(`Found match: n = ${n}, φ(${n}) = φ(${n+1})`);
            }
        }
        
        this.result = sum;
        this.logStep(`Total sum: ${sum}`);
        this.logStep(`Total matches: ${count}`);
        this.isCalculating = false;
    }
    
    logStep(message) {
        this.calculationSteps = [...this.calculationSteps, message];
    }
    
    // Calculate Euler's totient function φ(n)
    eulerTotient(n) {
        if (n === 1) return 1;
        
        let result = n;
        let p = 2;
        
        while (p * p <= n) {
            if (n % p === 0) {
                while (n % p === 0) {
                    n /= p;
                }
                result -= Math.floor(result / p);
            }
            p++;
        }
        
        if (n > 1) {
            result -= Math.floor(result / n);
        }
        
        return result;
    }
    
    // Check if φ(n) = φ(n+1)
    isPhiEqual(n, nPlus1) {
        return this.eulerTotient(n) === this.eulerTotient(nPlus1);
    }
