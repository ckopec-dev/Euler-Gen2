# Euler Problem 51 Solution in Lightning Web Component

I'll solve Euler Problem 51 using Lightning Web Components. This problem asks for the smallest prime which, by replacing part of the number with the same digit, produces 8 different primes.

```javascript
// primeUtils.js
export class PrimeUtils {
    static isPrime(n) {
        if (n <= 1) return false;
        if (n <= 3) return true;
        if (n % 2 === 0 || n % 3 === 0) return false;
        
        for (let i = 5; i * i <= n; i += 6) {
            if (n % i === 0 || n % (i + 2) === 0) {
                return false;
            }
        }
        return true;
    }
    
    static getPrimes(limit) {
        const primes = [];
        for (let i = 2; i < limit; i++) {
            if (this.isPrime(i)) {
                primes.push(i);
            }
        }
        return primes;
    }
    
    static getDigitPositions(number) {
        const positions = [];
        const numStr = number.toString();
        for (let i = 0; i < numStr.length; i++) {
            positions.push(i);
        }
        return positions;
    }
    
    static generateCombinations(arr, k) {
        if (k === 1) return arr.map(item => [item]);
        
        const result = [];
        for (let i = 0; i < arr.length; i++) {
            const smallerCombinations = this.generateCombinations(arr.slice(i + 1), k - 1);
            for (const combo of smallerCombinations) {
                result.push([arr[i], ...combo]);
            }
        }
        return result;
    }
}
```

```javascript
// primeFinder.js
import { PrimeUtils } from './primeUtils.js';

export class PrimeFinder {
    static findSmallestPrimeWithEightFamily() {
        // Start checking from 100000 since we need 6-digit numbers
        for (let num = 100000; num < 1000000; num++) {
            if (PrimeUtils.isPrime(num)) {
                const family = this.getPrimeFamily(num);
                if (family.length >= 8) {
                    return num;
                }
            }
        }
        return -1;
    }
    
    static getPrimeFamily(number) {
        const numStr = number.toString();
        const family = [number];
        const digitPositions = PrimeUtils.getDigitPositions(number);
        
        // Try different patterns of replacing digits
        for (let patternLength = 1; patternLength < numStr.length; patternLength++) {
            const combinations = this.generateCombinations(digitPositions, patternLength);
            
            for (const positions of combinations) {
                const primes = this.getPrimesWithPattern(numStr, positions);
                if (primes.length >= 8) {
                    return primes;
                }
            }
        }
        
        return family;
    }
    
    static generateCombinations(arr, k) {
        if (k === 1) return arr.map(item => [item]);
        
        const result = [];
        for (let i = 0; i < arr.length; i++) {
            const smallerCombinations = this.generateCombinations(arr.slice(i + 1), k - 1);
            for (const combo of smallerCombinations) {
                result.push([arr[i], ...combo]);
            }
        }
        return result;
    }
    
    static getPrimesWithPattern(numStr, positions) {
        const primes = [];
        
        // Try replacing the selected positions with digits 0-9
        for (let digit = 0; digit <= 9; digit++) {
            let newNumStr = numStr;
            let valid = true;
            
            // Replace positions with current digit
            for (const pos of positions) {
                if (pos === 0 && digit === 0) {
                    // First digit can't be 0
                    valid = false;
                    break;
                }
                newNumStr = newNumStr.substring(0, pos) + digit + newNumStr.substring(pos + 1);
            }
            
            if (valid) {
                const newNum = parseInt(newNumStr);
                if (PrimeUtils.isPrime(newNum)) {
                    primes.push(newNum);
                }
            }
        }
        
        return primes;
    }
}
```

## Solution Approach

The solution works by:

1. **Prime Generation**: Using an efficient primality test to identify prime numbers
2. **Pattern Testing**: For each prime number, testing different patterns of digit replacement
3. **Family Counting**: For each replacement pattern, counting how many primes are formed
4. **Optimization**: Starting from 6-digit numbers since we need 8 primes in a family

## Key Concepts

- **Primality Testing**: Efficient algorithm to check if a number is prime
- **Digit Position Combinations**: Generate all possible ways to replace digits
- **Family Generation**: Create families of primes by replacing digits
- **Optimization**: Start from reasonable bounds to reduce computation time
