# Euler Problem 180 Solution in Go

## Problem Statement
Find the sum of all distinct values of the expression (a/b)^(c/d) where a, b, c, d are positive integers with a, b, c, d ≤ 50, and a/b and c/d are in their lowest terms.

## Solution Approach
1. Generate all possible fractions a/b where a, b ≤ 50 and gcd(a,b) = 1
2. Generate all possible fractions c/d where c, d ≤ 50 and gcd(c,d) = 1
3. For each pair of fractions, compute (a/b)^(c/d)
4. Use a map to track distinct values
5. Sum all distinct values

import (

// gcd calculates the greatest common divisor of two integers
func gcd(a, b int) int {
    for b != 0 {
        a, b = b, a%b
    }
    return a
}

// isLowestTerms checks if fraction a/b is in lowest terms
func isLowestTerms(a, b int) bool {
    return gcd(a, b) == 1
}

    return new(big.Rat).Exp(base, exponent)

// power computes (base)^(exponent) using big.Rat for precision
func power(base *big.Rat, exponent *big.Rat) *big.Rat {
    // Convert to float64 for computation
    baseFloat, _ := base.Float64()
    expFloat, _ := exponent.Float64()
    
    // Compute the power
    resultFloat := 1.0
    for i := 0; i < int(expFloat); i++ {
        resultFloat *= baseFloat
    }
    
    // Convert back to big.Rat for precision
    result := new(big.Rat).SetFloat64(resultFloat)
    return result
}
