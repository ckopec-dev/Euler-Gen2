# Euler Problem 60 Solution in Kotlin

I'll solve the problem of finding the lowest sum for a set of five primes that are all prime pairs.

```kotlin
import kotlin.math.sqrt

fun main() {
    val primes = generatePrimes(10000)
    val primeSet = primes.toSet()
    
    // Find all prime pairs that can form valid pairs
    val pairs = mutableListOf<Pair<Int, Int>>()
    
    for (i in primes.indices) {
        for (j in i + 1 until primes.size) {
            val p1 = primes[i]
            val p2 = primes[j]
            
            if (isConcatenationPrime(p1, p2, primeSet)) {
                pairs.add(Pair(p1, p2))
            }
        }
    }
    
    // Find chains of 5 primes where each pair is valid
    val result = findFivePrimeChain(primes, pairs)
    println("Lowest sum: ${result.sum()}")
    println("Primes: ${result.joinToString(", ")}")
}

fun generatePrimes(limit: Int): List<Int> {
    val isPrime = BooleanArray(limit + 1) { true }
    isPrime[0] = false
    isPrime[1] = false
    
    for (i in 2..sqrt(limit.toDouble()).toInt()) {
        if (isPrime[i]) {
            for (j in i * i..limit step i) {
                isPrime[j] = false
            }
        }
    }
    
    return (2..limit).filter { isPrime[it] }
}

fun isConcatenationPrime(p1: Int, p2: Int, primeSet: Set<Int>): Boolean {
    val concat1 = "${p1}${p2}".toIntOrNull()
    val concat2 = "${p2}${p1}".toIntOrNull()
    
    return concat1 != null && concat2 != null && 
           primeSet.contains(concat1) && primeSet.contains(concat2)
}

fun findFivePrimeChain(primes: List<Int>, pairs: List<Pair<Int, Int>>): List<Int> {
    // For each prime, find all pairs it can form
    val primeToPairs = mutableMapOf<Int, MutableList<Int>>()
    
    for ((p1, p2) in pairs) {
        primeToPairs.getOrPut(p1) { mutableListOf() }.add(p2)
        primeToPairs.getOrPut(p2) { mutableListOf() }.add(p1)
    }
    
    // Try to build chains of 5 primes
    for (i in primes.indices) {
        val p1 = primes[i]
        val p1Pairs = primeToPairs[p1] ?: continue
        
        for (j in i + 1 until primes.size) {
            val p2 = primes[j]
            if (!p1Pairs.contains(p2)) continue
            
            val p2Pairs = primeToPairs[p2] ?: continue
            val commonPairs = p1Pairs.intersect(p2Pairs.toSet()).toList()
            
            for (p3 in commonPairs) {
                val p3Pairs = primeToPairs[p3] ?: continue
                val commonPairs2 = p2Pairs.intersect(p3Pairs.toSet()).toList()
                
                for (p4 in commonPairs2) {
                    val p4Pairs = primeToPairs[p4] ?: continue
                    val commonPairs3 = p3Pairs.intersect(p4Pairs.toSet()).toList()
                    
                    for (p5 in commonPairs3) {
                        // Verify all pairs in the chain are valid
                        if (isChainValid(listOf(p1, p2, p3, p4, p5), pairs)) {
                            return listOf(p1, p2, p3, p4, p5)
                        }
                    }
                }
            }
        }
    }
    
    return emptyList()
}

fun isChainValid(chain: List<Int>, pairs: List<Pair<Int, Int>>): Boolean {
    val pairSet = pairs.toSet()
    
    for (i in chain.indices) {
        for (j in i + 1 until chain.size) {
            val p1 = chain[i]
            val p2 = chain[j]
            if (!pairSet.contains(Pair(p1, p2)) && !pairSet.contains(Pair(p2, p1))) {
                return false
            }
        }
    }
    
    return true
}
```

## Problem Analysis

Euler Problem 60 asks us to find the lowest sum for a set of five primes that are all prime pairs. A prime pair means that when we concatenate the two primes in either order, the result is also prime.

## Approach

1. **Generate Primes**: Create a list of primes up to a reasonable limit (10,000)
2. **Find Prime Pairs**: Identify all pairs of primes where concatenating them in both orders produces primes
3. **Build Chains**: Search for chains of 5 primes where every pair within the chain forms a valid prime pair
4. **Return Result**: Output the lowest sum of such a chain

## Key Functions

- `generatePrimes()`: Uses Sieve of Eratosthenes to efficiently generate primes
- `isConcatenationPrime()`: Checks if concatenating two primes in both orders produces primes
- `findFivePrimeChain()`: Builds chains of 5 primes using the valid pairs
- `isChainValid()`: Verifies that all pairs within a potential chain are valid
