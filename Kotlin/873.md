# Euler Problem 873 Solution in Kotlin

I need to solve Euler Problem 873, which asks for the sum of all distinct values of a(n) where a(n) = n + reverse(n) and a(n) is a perfect square.

Let me break this down:
1. For each n, calculate a(n) = n + reverse(n)
2. Check if a(n) is a perfect square
3. Collect all distinct perfect square values
4. Sum them up

```kotlin
import kotlin.math.*

fun reverseNumber(n: Long): Long {
    var num = n
    var reversed = 0L
    while (num > 0) {
        reversed = reversed * 10 + num % 10
        num /= 10
    }
    return reversed
}

fun isPerfectSquare(n: Long): Boolean {
    if (n < 0) return false
    val sqrt = sqrt(n.toDouble()).toLong()
    return sqrt * sqrt == n
}

fun solveEuler873(limit: Long): Long {
    val perfectSquares = mutableSetOf<Long>()
    
    // We need to find all n such that n + reverse(n) is a perfect square
    // Let's think about the range - since we're looking for distinct values,
    // we should consider a reasonable upper bound
    
    for (n in 1..limit) {
        val an = n + reverseNumber(n)
        if (isPerfectSquare(an)) {
            perfectSquares.add(an)
        }
    }
    
    return perfectSquares.sum()
}

fun main() {
    // For a reasonable search limit, let's try 10^7
    val result = solveEuler873(10000000L)
    println("Sum of all distinct perfect square values: $result")
    
    // Let's also verify some examples
    println("Verification:")

        val an = n + reverseNumber(n)
        if (isPerfectSquare(an)) {
            println("$n + reverse($n) = $an = ${sqrt(an.toDouble()).toInt()}²")
        }
    }
}
```
