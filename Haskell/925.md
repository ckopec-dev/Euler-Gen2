Let me break this down:
- For a number like 145, we check if 1! + 4! + 5! = 145
- We need to find all such numbers and sum them up

factorial n = n * factorial (n - 1)

-- Calculate sum of factorials of digits

sumOfFactorialDigits n = sum $ map factorial $ map digitToInt $ show n
  where
    digitToInt = fromIntegral . fromEnum . subtract '0'

digitToInt c = fromEnum c - fromEnum '0'

-- Check if a number equals sum of factorials of its digits
isEqualSumOfFactorialDigits :: Integer -> Bool
isEqualSumOfFactorialDigits n = n == sumOfFactorialDigits n

-- Find all such numbers in a given range
findSpecialNumbers :: Integer -> [Integer]
findSpecialNumbers limit = filter isEqualSumOfFactorialDigits [1..limit]

-- Main solution

main = do
    -- Since we're looking for numbers where n = sum of factorials of digits,
    -- we can limit our search space reasonably
