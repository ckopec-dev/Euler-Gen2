  "Return list of prime factors with their powers"
  (let ((factors '())
        (d 2))
    (loop while (<= (* d d) n)
          do (if (= (mod n d) 0)
                 (let ((count 0) (temp n))
                   (loop while (= (mod temp d) 0)
                         do (incf count)
                         (setf temp (floor temp d)))
                   (push (cons d count) factors)
                   (setf n temp))
                 (incf d)))
    (if (> n 1)
        (push (cons n 1) factors))
    (nreverse factors)))
