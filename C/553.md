// Function to check if a number is prime
bool is_prime(long long n) {
    if (n < 2) return false;
    if (n == 2) return true;
    if (n % 2 == 0) return false;
    
    long long sqrt_n = (long long)sqrt(n);
    for (long long i = 3; i <= sqrt_n; i += 2) {
        if (n % i == 0) return false;
    }
    return true;
}

// Function to find the smallest prime greater than n
long long next_prime(long long n) {
    if (n < 2) return 2;
    if (n == 2) return 3;
    
    long long candidate = n + 1;
    if (candidate % 2 == 0) candidate++;
    
    while (!is_prime(candidate)) {
        candidate += 2;
    }
    return candidate;
}

// Function to compute the pseudo-Fortunate number for prime p
long long pseudo_fortunate(long long p) {
    // Find the smallest positive integer k such that p# + k is prime
    // where p# is the primorial of p (product of all primes <= p)
    
    // Calculate primorial of p
    long long primorial = 1;
    long long temp_p = 2;
    
    while (temp_p <= p) {
        primorial *= temp_p;
        temp_p = next_prime(temp_p);
    }
    
    // Find smallest k such that primorial + k is prime
    long long k = 1;
    while (!is_prime(primorial + k)) {
        k++;
    }
    
    return k;
}

void get_primes(int limit, int *primes, int *count) {
    *count = 0;
    for (int i = 2; i < limit; i++) {
        if (is_prime(i)) {
            primes[(*count)++] = i;
        }
    }
}
