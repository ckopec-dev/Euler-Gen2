# Euler Problem 212 Solution in C

This problem asks for the number of lattice points that are inside or on the boundary of at least one of the given rectangular prisms.

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// Structure to represent a rectangular prism
typedef struct {
    long long x1, x2, y1, y2, z1, z2;
} Prism;

// Structure to represent a point
typedef struct {
    long long x, y, z;
} Point;

// Function to check if a point is inside a prism
int is_inside(Prism p, Point pt) {
    return (pt.x >= p.x1 && pt.x <= p.x2 && 
            pt.y >= p.y1 && pt.y <= p.y2 && 
            pt.z >= p.z1 && pt.z <= p.z2);
}

// Function to calculate intersection of two prisms
Prism intersection(Prism p1, Prism p2) {
    Prism result;
    result.x1 = (p1.x1 > p2.x1) ? p1.x1 : p2.x1;
    result.x2 = (p1.x2 < p2.x2) ? p1.x2 : p2.x2;
    result.y1 = (p1.y1 > p2.y1) ? p1.y1 : p2.y1;
    result.y2 = (p1.y2 < p2.y2) ? p1.y2 : p2.y2;
    result.z1 = (p1.z1 > p2.z1) ? p1.z1 : p2.z1;
    result.z2 = (p1.z2 < p2.z2) ? p1.z2 : p2.z2;
    
    // If intersection is invalid, return an empty prism
    if (result.x1 > result.x2 || result.y1 > result.y2 || result.z1 > result.z2) {
        result.x1 = result.x2 = result.y1 = result.y2 = result.z1 = result.z2 = 0;
    }
    
    return result;
}

// Function to calculate volume of a prism
long long volume(Prism p) {
    if (p.x1 > p.x2 || p.y1 > p.y2 || p.z1 > p.z2) {
        return 0;
    }
    return (p.x2 - p.x1 + 1) * (p.y2 - p.y1 + 1) * (p.z2 - p.z1 + 1);
}

// Function to calculate number of lattice points in a prism
long long count_points(Prism p) {
    if (p.x1 > p.x2 || p.y1 > p.y2 || p.z1 > p.z2) {
        return 0;
    }
    return (p.x2 - p.x1 + 1) * (p.y2 - p.y1 + 1) * (p.z2 - p.z1 + 1);
}

// Function to calculate union of prisms using inclusion-exclusion principle
long long union_volume(Prism* prisms, int n) {
    long long total = 0;
    
    // For each subset of prisms
    for (int mask = 1; mask < (1 << n); mask++) {
        Prism current = {0, 0, 0, 0, 0, 0};
        int count = 0;
        
        // Find intersection of selected prisms
        int first = 1;
        for (int i = 0; i < n; i++) {
            if (mask & (1 << i)) {
                if (first) {
                    current = prisms[i];
                    first = 0;
                } else {
                    current = intersection(current, prisms[i]);
                }
                count++;
            }
        }
        
        long long vol = volume(current);
        if (count % 2 == 1) {
            total += vol;
        } else {
            total -= vol;
        }
    }
    
    return total;
}

    printf("Number of lattice points in union: %lld\n", result);
```

This solution implements the inclusion-exclusion principle to find the number of lattice points that are inside or on the boundary of at least one of the given rectangular prisms. The algorithm:

1. Uses bit manipulation to generate all possible subsets of prisms
2. For each subset, calculates the intersection of all prisms in that subset
3. Applies inclusion-exclusion: add volumes of odd-sized intersections, subtract volumes of even-sized intersections
4. Returns the total volume (which equals the number of lattice points in the union)

The time complexity is O(2^n) where n is the number of prisms, which is acceptable for small inputs but would be too slow for large inputs without optimization.
