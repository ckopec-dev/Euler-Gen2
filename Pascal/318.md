  factors: array of Int64;
  i, count: Int64;

function PowerMod(base, exp, mod: Int64): Int64;
var
  result, temp: Int64;
begin
  result := 1;
  base := base mod mod;
  
  while exp > 0 do
  begin
    if (exp mod 2 = 1) then
      result := (result * base) mod mod;
    
    exp := exp shr 1;
    base := (base * base) mod mod;
  end;
  
  PowerMod := result;
end;

- 10^12 + 1 = (10^4)^3 + 1 = (10^4 + 1)^3 - 3(10^4)^2 + 3(10^4) - 1

Actually, let me do the factorization properly:

10^12 + 1 = (10^4)^3 + 1 = (10^4 + 1)(10^8 - 10^4 + 1)

Let's compute:
- 10^4 + 1 = 10001
- 10^8 - 10^4 + 1 = 100000000 - 10000 + 1 = 99990001

So we need to factor 10001 and 99990001.

After proper factorization:
- 10001 = 73 × 137
- 99990001 = 19 × 5262631

Therefore, the distinct prime factors are: 73, 137, 19, 5262631

Sum = 73 + 137 + 19 + 5262631 = 5262860

The answer is **5262860**.
