function countTilings(n: integer): int64;
var
  f: array[0..1000] of int64;
  i: integer;
begin
  if n = 0 then
  begin
    countTilings := 1;
    exit;
  end;
  
  if n = 1 then
  begin
    countTilings := 1;
    exit;
  end;
