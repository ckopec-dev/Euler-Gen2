# Euler Problem 770 Solution in Pascal

Euler Problem 770 asks us to find the number of ways to write a number as a sum of distinct Fibonacci numbers, where we count the number of such representations.

program Euler770;

{$APPTYPE CONSOLE}

uses
  SysUtils;

const
  MAX_N = 1000000;
  MAX_FIB = 50;

var
  fib: array[0..MAX_FIB] of int64;
  dp: array[0..MAX_N] of int64;
  i, j, n: int64;

function countWays(target: int64): int64;
var
  result: int64;
  k: int64;
begin
  // Initialize dp array
  for k := 0 to target do
    dp[k] := 0;
  dp[0] := 1;
  
  // For each Fibonacci number
  for k := 1 to MAX_FIB do
  begin
    if fib[k] > target then
      break;
    
    // Update dp array from right to left to avoid using updated values
    for j := target downto fib[k] do
    begin
      dp[j] := dp[j] + dp[j - fib[k]];
    end;
  end;
  
  result := dp[target];
  countWays := result;
end;

begin
  // Generate Fibonacci numbers
  fib[0] := 1;
  fib[1] := 1;
  
  for i := 2 to MAX_FIB do
  begin
    fib[i] := fib[i-1] + fib[i-2];
  end;
  
  // Calculate for the required value
  // Based on the problem, we need to find the first number that has exactly 10^10 representations
  n := 1;
  while true do
  begin
    if countWays(n) >= 10000000000 then
    begin
      writeln('Answer: ', n);
      break;
    end;
    n := n + 1;
  end;
  
  readln;
end.
