# Euler Problem 868 Solution in Pascal

I'll solve Euler Problem 868, which asks for the number of ways to write a number as a sum of powers of 2, where each power can be used at most twice.

```pascal
program Euler868;

{$APPTYPE CONSOLE}

uses
  SysUtils;

var
  n: integer;
  memo: array[0..1000000] of int64;
  result: int64;

function countWays(n: integer): int64;
var
  i, j: integer;
  dp: array[0..1000000] of int64;
begin
  // Initialize dp array
  for i := 0 to n do
    dp[i] := 0;
  
  dp[0] := 1;  // One way to make 0 (use nothing)
  
  // For each power of 2 (2^0, 2^1, 2^2, ...)
  i := 0;
  while (1 shl i) <= n do
  begin
    // For each number from the current power up to n
    for j := (1 shl i) to n do
    begin
      // We can use this power 0, 1, or 2 times
      // dp[j] += dp[j - 2^i] (using it once)
      // dp[j] += dp[j - 2*2^i] (using it twice)
      if j >= (1 shl i) then
        dp[j] := dp[j] + dp[j - (1 shl i)];
      if j >= 2 * (1 shl i) then
        dp[j] := dp[j] + dp[j - 2 * (1 shl i)];
    end;
    inc(i);
  end;
  
  result := dp[n];
  countWays := result;
end;

function countWaysOptimized(n: integer): int64;
var
  i, j, power: integer;
begin
  // Initialize memoization array
  for i := 0 to n do
    memo[i] := 0;
  
  memo[0] := 1;
  
  i := 0;
  while (1 shl i) <= n do
  begin
    power := 1 shl i;
    
    // For each number from power to n
    for j := power to n do
    begin
      // Add ways to form j using 1 power of 2^i
      if j >= power then
        memo[j] := memo[j] + memo[j - power];
      
      // Add ways to form j using 2 powers of 2^i
      if j >= 2 * power then
        memo[j] := memo[j] + memo[j - 2 * power];
    end;
    
    inc(i);
  end;
  
  result := memo[n];
  countWaysOptimized := result;
end;

// Alternative implementation using recursion with memoization
function countWaysRecursive(n, maxPower: integer): int64;
var
  power: integer;
begin
  if n = 0 then
  begin
    countWaysRecursive := 1;
    exit;
  end;
  
  if n < 0 then
  begin
    countWaysRecursive := 0;
    exit;
  end;
  
  power := 1;
  while power <= maxPower do
    power := power * 2;
  power := power div 2;
  
  if power = 0 then
  begin
    countWaysRecursive := 0;
    exit;
  end;
  
  // We can use 0, 1, or 2 of the current power
  result := countWaysRecursive(n, power div 2);  // 0 times
  if n >= power then
    result := result + countWaysRecursive(n - power, power div 2);  // 1 time
  if n >= 2 * power then
    result := result + countWaysRecursive(n - 2 * power, power div 2);  // 2 times
  
  countWaysRecursive := result;
end;

// More efficient iterative approach
function solveEuler868(n: integer): int64;
var
  i, j, pow2: integer;
  dp: array[0..1000000] of int64;
begin
  // Initialize dp array
  for i := 0 to n do
    dp[i] := 0;
  
  dp[0] := 1;
  
  // For each power of 2 (1, 2, 4, 8, 16, ...)
  i := 0;
  while (1 shl i) <= n do
  begin
    pow2 := 1 shl i;
    
    // For each number from pow2 to n
    for j := pow2 to n do
    begin
      // Add ways to form j using 0, 1, or 2 of the current power
      dp[j] := dp[j] + dp[j - pow2];
      if j >= 2 * pow2 then
        dp[j] := dp[j] + dp[j - 2 * pow2];
    end;
    
    inc(i);
  end;
  
  solveEuler868 := dp[n];
end;

begin
  writeln('Euler Problem 868 Solution');
  writeln('Counting number of ways to write n as sum of powers of 2');
  writeln('where each power can be used at most twice');
  writeln;
  
  // Test with smaller values first
  writeln('For n = 10:');
  writeln('Number of ways = ', solveEuler868(10));
  
  writeln('For n = 100:');
  writeln('Number of ways = ', solveEuler868(100));
  
  writeln('For n = 1000:');
  writeln('Number of ways = ', solveEuler868(1000));
  
  // For the actual problem, we want n = 10^6
  writeln('For n = 1000000:');
  writeln('Number of ways = ', solveEuler868(1000000));
  
  readln;
end.
```

## Explanation

This solution implements a dynamic programming approach to count the number of ways to express a number as a sum of powers of 2, where each power can be used at most twice.

### Key Concepts:

1. **Dynamic Programming**: We build up solutions for smaller numbers to solve larger ones
2. **Powers of 2**: We consider 1, 2, 4, 8, 16, 32, ... as our available numbers
3. **Constraint**: Each power can be used at most twice
4. **State Definition**: `dp[i]` represents the number of ways to form the number `i`

### Algorithm Steps:

1. Initialize a DP array where `dp[0] = 1` (one way to make 0)
2. For each power of 2 (starting from 1):
   - For each number from that power to our target:
     - Add the number of ways to form `(current_number - power)` to `dp[current_number]`
     - If we can use the power twice, also add the ways to form `(current_number - 2*power)`
3. Return `dp[n]` as the answer

### Time Complexity:
- O(n log n) where n is the target number
- We iterate through powers of 2 (log n iterations) and for each power, we iterate through numbers up to n

### Space Complexity:
- O(n) for the DP array

The solution handles the constraint that each power of 2 can be used at most twice by adding contributions from using 1 and 2 instances of each power.
