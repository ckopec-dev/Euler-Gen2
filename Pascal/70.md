# Euler Problem 70 Solution in Pascal

Euler Problem 70 asks: Find the value of n (where n ≤ 10^7) for which φ(n) is a permutation of n and the ratio n/φ(n) is minimized.

Here's the Pascal solution:

```pascal
program Euler70;
{$APPTYPE CONSOLE}

uses
  SysUtils, Math;

function EulerPhi(n: integer): integer;
var
  i, result: integer;
begin
  result := n;
  i := 2;
  while i * i <= n do
  begin
    if n mod i = 0 then
    begin
      result := result - result div i;
      while n mod i = 0 do
        n := n div i;
    end;
    i := i + 1;
  end;
  if n > 1 then
    result := result - result div n;
  EulerPhi := result;
end;

function IsPermutation(a, b: integer): boolean;
var
  digitsA, digitsB: array[0..9] of integer;
  temp: integer;
  i: integer;
begin
  // Initialize digit arrays
  for i := 0 to 9 do
  begin
    digitsA[i] := 0;
    digitsB[i] := 0;
  end;
  
  // Count digits in a
  temp := a;
  while temp > 0 do
  begin
    digitsA[temp mod 10] := digitsA[temp mod 10] + 1;
    temp := temp div 10;
  end;
  
  // Count digits in b
  temp := b;
  while temp > 0 do
  begin
    digitsB[temp mod 10] := digitsB[temp mod 10] + 1;
    temp := temp div 10;
  end;
  
  // Compare digit counts
  for i := 0 to 9 do
  begin
    if digitsA[i] <> digitsB[i] then
    begin
      IsPermutation := false;
      exit;
    end;
  end;
  
  IsPermutation := true;
end;

var
  n, phi, minRatio, bestN: integer;
  ratio: real;
  startTime: TDateTime;

begin
  startTime := Now;
  minRatio := 1000000000;
  bestN := 0;
  
  // We can optimize by checking only numbers that are products of two primes
  // since those typically give the minimum n/φ(n) ratio
  for n := 2 to 10000000 do
  begin
    phi := EulerPhi(n);
    
    // Check if phi(n) is permutation of n
    if IsPermutation(n, phi) then
    begin
      ratio := n / phi;
      if ratio < minRatio then
      begin
        minRatio := round(ratio * 1000000); // Store as integer for comparison
        bestN := n;
        writeln('Found better: n = ', n, ', phi = ', phi, ', ratio = ', ratio:0:6);
      end;
    end;
  end;
  
  writeln('Answer: n = ', bestN);
  writeln('Time: ', FormatDateTime('hh:nn:ss.zzz', Now - startTime));
end.
```
