# Euler Problem 933 Solution in Fortran

I'll solve Euler Problem 933, which asks for the number of ways to write a number as a sum of powers of 2, where each power can be used at most twice.

```fortran
program euler933
    implicit none
    integer, parameter :: max_n = 1000000
    integer :: n, i, j, k
    integer :: ways(max_n + 1)
    integer :: result
    
    ! Initialize the ways array
    do i = 0, max_n
        ways(i) = 0
    end do
    
    ! Base case: one way to make 0
    ways(0) = 1
    
    ! For each power of 2 (2^0, 2^1, 2^2, ...)
    i = 0
    do while (2**i <= max_n)
        ! For each number from 2^i to max_n
        do j = 2**i, max_n
            ! We can use 0, 1, or 2 copies of 2^i
            ! So we add ways[j - 2^i] + ways[j - 2*2^i]
            if (j >= 2**i) then
                ways(j) = ways(j) + ways(j - 2**i)
            end if
            if (j >= 2*2**i) then
                ways(j) = ways(j) + ways(j - 2*2**i)
            end if
        end do
        i = i + 1
    end do
    
    ! Find the first number n where ways(n) > 10^12
    do n = 1, max_n
        if (ways(n) > 1000000000000) then
            result = n
            exit
        end if
    end do
    
    write(*,*) 'The answer is:', result
    write(*,*) 'ways(', result, ') = ', ways(result)
    
end program euler933
```

```fortran
program euler933
    implicit none
    integer, parameter :: max_n = 1000000
    integer :: i, j, power
    integer :: ways(max_n + 1)
    integer :: result
    
    ! Initialize the ways array
    do i = 0, max_n
        ways(i) = 0
    end do
    
    ! Base case: one way to make 0
    ways(0) = 1
    
    ! For each power of 2 (2^0 = 1, 2^1 = 2, 2^2 = 4, ...)
    power = 1
    do while (power <= max_n)
        ! Process from high to low to avoid double counting
        ! For each number j from max_n down to power
        do j = max_n, power, -1
            ! We can use 0, 1, or 2 copies of the current power
            if (j >= power) then
                ways(j) = ways(j) + ways(j - power)
            end if
            if (j >= 2 * power) then
                ways(j) = ways(j) + ways(j - 2 * power)
            end if
        end do
        power = power * 2
    end do
    
    ! Find the first number n where ways(n) > 10^12
    do i = 1, max_n
        if (ways(i) > 1000000000000) then
            result = i
            exit
        end if
    end do
    
    write(*,*) 'The answer is:', result
    write(*,*) 'ways(', result, ') = ', ways(result)
    
end program euler933
```
