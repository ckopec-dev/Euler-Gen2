# Euler Problem 700 Solution in Fortran

Euler Problem 700 asks us to find the sum of all numbers n ≤ 10^12 such that φ(n) = φ(n+1), where φ is Euler's totient function.

Here's my Fortran solution:

```fortran
program euler700
    implicit none
    integer, parameter :: max_n = 1000000000000
    integer, parameter :: max_prime = 1000000
    integer :: i, j, n, phi_n, phi_n1
    integer, allocatable :: phi(:)
    integer, allocatable :: primes(:)
    integer :: prime_count, sum_result
    logical, allocatable :: is_prime(:)
    
    ! Initialize variables
    sum_result = 0
    
    ! Allocate arrays
    allocate(is_prime(max_prime))
    allocate(primes(max_prime/10))
    allocate(phi(max_n))
    
    ! Initialize is_prime array
    is_prime = .true.
    is_prime(1) = .false.
    
    ! Sieve of Eratosthenes to find primes
    do i = 2, int(sqrt(real(max_prime)))
        if (is_prime(i)) then
            do j = i*i, max_prime, i
                is_prime(j) = .false.
            end do
        end if
    end do
    
    ! Collect primes
    prime_count = 0
    do i = 2, max_prime
        if (is_prime(i)) then
            prime_count = prime_count + 1
            primes(prime_count) = i
        end if
    end do
    
    ! Initialize phi array
    do i = 1, max_n
        phi(i) = i
    end do
    
    ! Compute Euler's totient function using sieve method
    do i = 2, max_n
        if (phi(i) == i) then
            ! i is prime
            do j = i, max_n, i
                phi(j) = phi(j) / i * (i - 1)
            end do
        end if
    end do
    
    ! Find numbers where phi(n) = phi(n+1)
    do n = 1, max_n - 1
        phi_n = phi(n)
        phi_n1 = phi(n+1)
        if (phi_n == phi_n1) then
            sum_result = sum_result + n
        end if
    end do
    
    ! Output result
    print *, 'Sum of all n <= 10^12 such that φ(n) = φ(n+1):'
    print *, sum_result
    
    ! Clean up
    deallocate(is_prime)
    deallocate(primes)
    deallocate(phi)
    
end program euler700
```
