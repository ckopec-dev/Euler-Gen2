# Euler Problem 96 Solution in Fortran

Euler Problem 96 asks us to find the sum of the first nine digits of the 100-digit numbers in the 50 Sudoku puzzles provided in the input file.

Let me solve this step by step:

```fortran
program euler96
    implicit none
    integer, parameter :: n = 9, max_puzzles = 50
    integer :: i, j, k, puzzle_num, sum_first_nine
    integer :: grid(n,n), temp_grid(n,n)
    integer :: solutions(max_puzzles)
    character(len=100) :: line
    character(len=1) :: c
    logical :: solved
    
    ! Initialize sum
    sum_first_nine = 0
    
    ! Read and solve each puzzle
    do puzzle_num = 1, 50
        ! Skip the "Grid XX" line
        read(*,*) line
        
        ! Read the 9x9 grid
        do i = 1, n
            read(*,*) line
            do j = 1, n
                c = line(j:j)
                if (c >= '1' .and. c <= '9') then
                    grid(i,j) = ichar(c) - ichar('0')
                else
                    grid(i,j) = 0
                end if
            end do
        end do
        
        ! Solve the Sudoku puzzle
        call solve_sudoku(grid, solved)
        
        ! If solved successfully, add the first nine digits
        if (solved) then
            temp_grid = grid
            solutions(puzzle_num) = temp_grid(1,1) * 100000000 + &
                                   temp_grid(1,2) * 10000000 + &
                                   temp_grid(1,3) * 1000000 + &
                                   temp_grid(1,4) * 100000 + &
                                   temp_grid(1,5) * 10000 + &
                                   temp_grid(1,6) * 1000 + &
                                   temp_grid(1,7) * 100 + &
                                   temp_grid(1,8) * 10 + &
                                   temp_grid(1,9)
            sum_first_nine = sum_first_nine + solutions(puzzle_num)
        end if
    end do
    
    write(*,*) 'Sum of first nine digits:', sum_first_nine
end program euler96

subroutine solve_sudoku(grid, solved)
    implicit none
    integer, intent(inout) :: grid(9,9)
    logical, intent(out) :: solved
    integer :: i, j, k
    integer :: row, col
    logical :: found
    
    ! Find empty cell
    found = .false.
    do i = 1, 9
        do j = 1, 9
            if (grid(i,j) == 0) then
                row = i
                col = j
                found = .true.
                exit
            end if
        end do
        if (found) exit
    end do
    
    ! If no empty cell, puzzle is solved
    if (.not. found) then
        solved = .true.
        return
    end if
    
    ! Try numbers 1-9
    do k = 1, 9
        if (is_valid(grid, row, col, k)) then
            grid(row,col) = k
            
            ! Recursively solve
            call solve_sudoku(grid, solved)
            
            ! If solution found, return
            if (solved) return
            
            ! Backtrack
            grid(row,col) = 0
        end if
    end do
    
    ! No solution found for this path
    solved = .false.
end subroutine solve_sudoku

logical function is_valid(grid, row, col, num)
    implicit none
    integer, intent(in) :: grid(9,9), row, col, num
    integer :: i, j
    
    ! Check row
    do i = 1, 9
        if (grid(row,i) == num) then
            is_valid = .false.
            return
        end if
    end do
    
    ! Check column
    do i = 1, 9
        if (grid(i,col) == num) then
            is_valid = .false.
            return
        end if
    end do
    
    ! Check 3x3 box
    do i = 1, 3
        do j = 1, 3
            if (grid((row-1)/3*3+i, (col-1)/3*3+j) == num) then
                is_valid = .false.
                return
            end if
        end do
    end do
    
    is_valid = .true.
end function is_valid
```
