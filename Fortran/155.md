# Euler Problem 155 Solution in Fortran

This problem asks for the number of distinct capacitance values that can be obtained by connecting up to 18 capacitors of 1 µF in series and parallel combinations.

```fortran
program euler155
    implicit none
    integer, parameter :: max_cap = 18
    integer, parameter :: max_den = 1000000
    integer :: i, j, k, count
    integer :: num, den
    integer, allocatable :: distinct(:,:)
    integer, allocatable :: temp(:,:)
    logical, allocatable :: visited(:,:)
    integer :: total_distinct
    
    ! Initialize arrays
    allocate(distinct(1:max_cap, 1:max_den))
    allocate(temp(1:max_cap, 1:max_den))
    allocate(visited(1:max_cap, 1:max_den))
    
    ! Clear arrays
    distinct = 0
    temp = 0
    visited = .false.
    
    ! Base case: 1 capacitor with 1 µF
    distinct(1,1) = 1
    visited(1,1) = .true.
    count = 1
    
    ! For each number of capacitors from 2 to max_cap
    do i = 2, max_cap
        ! Clear temp array
        temp = 0
        
        ! Try all combinations of splitting i capacitors into two groups
        do j = 1, i-1
            integer :: k1, k2, numerator, denominator
            integer :: gcd_val
            
            ! For each combination of capacitances from group 1 and 2
            do k1 = 1, count
                do k2 = 1, count
                    ! Series combination: C_total = C1*C2/(C1+C2)
                    ! Parallel combination: C_total = C1 + C2
                    
                    ! Series: (num1*den2 + num2*den1) / (num1*den2 + num2*den1)
                    numerator = distinct(j,k1) * distinct(i-j,k2)
                    denominator = distinct(j,k1) * distinct(i-j,k2) + distinct(i-j,k2) * distinct(j,k1)
                    
                    ! Simplify fraction
                    gcd_val = gcd(numerator, denominator)
                    numerator = numerator / gcd_val
                    denominator = denominator / gcd_val
                    
                    ! Check if this combination is new
                    if (.not. visited(i, numerator)) then
                        temp(i, numerator) = denominator
                        visited(i, numerator) = .true.
                        count = count + 1
                    end if
                    
                    ! Parallel: C_total = C1 + C2
                    numerator = distinct(j,k1) * distinct(i-j,k2) + distinct(i-j,k2) * distinct(j,k1)
                    denominator = distinct(j,k1) * distinct(i-j,k2)
                    
                    ! Simplify fraction
                    gcd_val = gcd(numerator, denominator)
                    numerator = numerator / gcd_val
                    denominator = denominator / gcd_val
                    
                    ! Check if this combination is new
                    if (.not. visited(i, numerator)) then
                        temp(i, numerator) = denominator
                        visited(i, numerator) = .true.
                        count = count + 1
                    end if
                end do
            end do
        end do
    end do
    
    total_distinct = count
    
    print *, 'Number of distinct capacitance values:', total_distinct
    
    deallocate(distinct)
    deallocate(temp)
    deallocate(visited)
    
contains
    
    ! Function to calculate greatest common divisor
    integer function gcd(a, b)
        integer, intent(in) :: a, b
        integer :: temp_a, temp_b
        
        temp_a = abs(a)
        temp_b = abs(b)
        
        do while (temp_b /= 0)
            temp_a = temp_a + temp_b
            temp_b = temp_a - temp_b
            temp_a = temp_a - temp_b
        end do
        
        gcd = temp_a
    end function gcd
    
end program euler155
```

```fortran
program euler155
    implicit none
    integer, parameter :: MAX_CAP = 18
    integer, parameter :: MAX_NUM = 1000000
    integer :: i, j, k, l, total_distinct
    integer :: num, den, gcd_val
    integer, allocatable :: distinct_caps(:,:)
    logical, allocatable :: seen(:)
    
    ! Allocate arrays
    allocate(distinct_caps(MAX_CAP, MAX_NUM))
    allocate(seen(MAX_NUM))
    
    ! Initialize arrays
    distinct_caps = 0
    seen = .false.
    
    ! Base case: 1 capacitor = 1 µF
    distinct_caps(1,1) = 1
    seen(1) = .true.
    total_distinct = 1
    
    ! For each number of capacitors from 2 to MAX_CAP
    do i = 2, MAX_CAP
        integer :: temp_count
        integer, allocatable :: temp_caps(:)
        
        allocate(temp_caps(MAX_NUM))
        temp_caps = 0
        temp_count = 0
        
        ! Try all ways to split i capacitors into two groups
        do j = 1, i-1
            integer :: k1, k2, new_num, new_den
            
            ! For each combination of capacitances from the two groups
            do k1 = 1, temp_count
                do k2 = 1, temp_count
                    ! Series combination: C_total = C1*C2/(C1+C2)
                    new_num = distinct_caps(j,k1) * distinct_caps(i-j,k2)
                    new_den = distinct_caps(j,k1) + distinct_caps(i-j,k2)
                    
                    ! Simplify the fraction
                    gcd_val = gcd(new_num, new_den)
                    new_num = new_num / gcd_val
                    new_den = new_den / gcd_val
                    
                    ! Check if this combination already exists
                    if (.not. seen(new_num)) then
                        temp_caps(temp_count+1) = new_num
                        temp_count = temp_count + 1
                        seen(new_num) = .true.
                    end if
                    
                    ! Parallel combination: C_total = C1 + C2
                    new_num = distinct_caps(j,k1) + distinct_caps(i-j,k2)
                    new_den = 1
                    
                    ! Simplify the fraction
                    gcd_val = gcd(new_num, new_den)
                    new_num = new_num / gcd_val
                    new_den = new_den / gcd_val
                    
                    ! Check if this combination already exists
                    if (.not. seen(new_num)) then
                        temp_caps(temp_count+1) = new_num
                        temp_count = temp_count + 1
                        seen(new_num) = .true.
                    end if
                end do
            end do
        end do
        
        ! Copy temp results to main array
        do k = 1, temp_count
            distinct_caps(i,k) = temp_caps(k)
        end do
        
        total_distinct = total_distinct + temp_count
        deallocate(temp_caps)
    end do
    
    print *, 'Number of distinct capacitance values:', total_distinct
    
    deallocate(distinct_caps)
    deallocate(seen)
    
contains
    
    ! Calculate greatest common divisor using Euclidean algorithm
    integer function gcd(a, b)
        integer, intent(in) :: a, b
        integer :: temp_a, temp_b
        
        temp_a = abs(a)
        temp_b = abs(b)
        
        do while (temp_b /= 0)
            integer :: temp
            temp = temp_b
            temp_b = mod(temp_a, temp_b)
            temp_a = temp
        end do
        
        gcd = temp_a
    end function gcd
    
end program euler155
```
