  # q(n) = number of partitions of n into distinct parts
  
  # We'll use a 1D array for efficiency
  # dp[i] represents number of ways to partition i into distinct parts
  dp = Array.new(n + 1, 0)
  dp[0] = 1  # One way to partition 0 (empty partition)
  
  # For each number from 1 to n
  (1..n).each do |i|
    # We process from right to left to avoid using updated values
    (n).downto(i) do |j|
      dp[j] += dp[j - i]
    end
  end
  
  dp[n]
