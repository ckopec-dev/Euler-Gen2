```ruby
# Euler Problem 525 - The expected length of the altitude to the hypotenuse
# of a unit right triangle

# For a right triangle with legs of length 1, the hypotenuse has length sqrt(2)
# The altitude to the hypotenuse can be calculated as:
# Area = (1/2) * 1 * 1 = 1/2
# Area = (1/2) * hypotenuse * altitude
# So: (1/2) * sqrt(2) * altitude = 1/2
# Therefore: altitude = 1/sqrt(2) = sqrt(2)/2

# But we need to find the expected value when the triangle is randomly oriented
# This requires integration over all possible angles

# The expected length of the altitude from the right angle to the hypotenuse
# for a unit right triangle uniformly distributed over all orientations

# Using the formula for expected value of the altitude:
# E[altitude] = ∫₀^(π/2) (sin θ * cos θ) dθ = ∫₀^(π/2) (1/2 * sin(2θ)) dθ

# Let's compute this using numerical integration

def euler_525_solution
  # The exact mathematical solution is 1/π
  # But let's verify with numerical integration
  
  # For a unit right triangle, if we consider the angle θ between one leg and the horizontal,
  # the altitude to the hypotenuse is sin(θ) * cos(θ)
  
  # The expected value is:
  # E = (1/π) * ∫₀^(π/2) sin(θ) * cos(θ) dθ
  # E = (1/π) * ∫₀^(π/2) (1/2) * sin(2θ) dθ
  # E = (1/2π) * [-cos(2θ)/2]₀^(π/2)
  # E = (1/2π) * [(-cos(π) + cos(0))/2]
  # E = (1/2π) * [(-(-1) + 1)/2]
  # E = (1/2π) * [2/2]
  # E = 1/π
  
  # Calculate 1/π
  result = 1.0 / Math::PI
  
  # Return with sufficient precision
  sprintf("%.10f", result)
end

# Alternative approach using Monte Carlo simulation
def monte_carlo_approach(n = 1000000)
  sum = 0.0
  
  n.times do
    # Generate random angle between 0 and π/2
    theta = rand * Math::PI / 2
    
    # Altitude length for this orientation
    altitude = Math.sin(theta) * Math.cos(theta)
    
    sum += altitude
  end
  
  # Average
  sum / n
end
```
