def fibonacci_representations(n)
  # Generate Fibonacci numbers up to n
  fib = [1, 2]
  while fib.last < n
    fib << fib[-1] + fib[-2]
  end
  
  # Remove any Fibonacci numbers greater than n
  fib = fib.select { |f| f <= n }
  
  # Use dynamic programming to count representations
  # dp[i] = number of ways to represent i
  dp = Array.new(n + 1, 0)
  dp[0] = 1  # One way to represent 0 (using no numbers)
  
  # For each Fibonacci number, update the dp array
  fib.each do |f|
    (n).downto(f) do |i|
      dp[i] += dp[i - f]
    end
  end
  
  dp[n]
end
