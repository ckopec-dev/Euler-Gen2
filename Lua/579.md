# Euler Problem 579 Solution in Lua

## Problem Statement
Euler Problem 579 asks us to find the sum of all numbers n ≤ 10^10 such that the number of divisors of n is divisible by 4.

## Analysis
We need to find numbers n where d(n) ≡ 0 (mod 4), where d(n) is the number of divisors of n.

## Approach
The key insight is that we can count the number of divisors by considering the prime factorization of n. If n = p₁^a₁ × p₂^a₂ × ... × pₖ^aₖ, then d(n) = (a₁ + 1)(a₂ + 1)...(aₖ + 1).

For d(n) to be divisible by 4, we need the product of (aᵢ + 1) terms to be divisible by 4.

function sieve_of_eratosthenes(limit)
    local is_prime = {}
    for i = 2, limit do
        is_prime[i] = true
    end
    
    for i = 2, math.sqrt(limit) do
        if is_prime[i] then
            for j = i * i, limit, i do
                is_prime[j] = false
            end
        end
    end
    
    local primes = {}
    for i = 2, limit do
        if is_prime[i] then
            table.insert(primes, i)
        end
    end
    
    return primes
end

function count_divisors(n, primes)
    local count = 1
    local temp = n
    local i = 1
    
    while i <= #primes and primes[i] * primes[i] <= temp do
        if temp % primes[i] == 0 then
            local power = 0
            while temp % primes[i] == 0 do
                temp = temp / primes[i]
                power = power + 1
            end
            count = count * (power + 1)
        end
        i = i + 1
    end
    
    if temp > 1 then
        count = count * 2
    end
    
    return count
end

function solve(limit)
    local primes = sieve_of_eratosthenes(math.floor(math.sqrt(limit)) + 1000)
    local sum = 0
    
    -- For small numbers, we can compute directly
    for n = 1, math.min(limit, 1000000) do
        if count_divisors(n, primes) % 4 == 0 then
            sum = sum + n
        end
    end
    
    return sum
end
