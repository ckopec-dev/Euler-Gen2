    if n < 1 then
        error("Only positive integers are allowed")
    end
    
    local steps = 0
    local current = n
    
    while current ~= 1 do
        if current % 2 == 0 then
            current = current / 2
        else
            current = 3 * current + 1
        end
        steps = steps + 1
    end
