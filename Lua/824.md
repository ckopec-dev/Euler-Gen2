function count_ways(n)
    -- dp[i] represents number of ways to make sum i
    local dp = {}
    for i = 0, n do
        dp[i] = 0
    end
    dp[0] = 1  -- One way to make 0 (use nothing)
    
    -- For each power of 2 (2^0, 2^1, 2^2, ...)
    local power = 1
    while power <= n do
        -- Process this power at most twice
        for i = 0, n do
            -- Add ways to make i using this power (0, 1, or 2 times)
            if i >= power then
                dp[i] = dp[i] + dp[i - power]
            end
            if i >= 2 * power then
                dp[i] = dp[i] + dp[i - 2 * power]
            end
        end
        power = power * 2
    end
    
    return dp[n]
end
