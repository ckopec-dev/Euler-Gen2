# Euler Problem 60 Solution in Lua

## Problem Statement
The primes 3, 7, 109, and 673 are quite remarkable. By taking any two primes and concatenating them in any order the result will always be prime. For example, taking 7 and 109, both 7109 and 1097 are prime. The sum of these four primes, 792, represents the lowest sum for a set of four primes with this property.

Find the lowest sum for a set of five primes which are remarkable in the same way.

## Solution Approach
This is a brute force approach that:
1. Generates prime numbers
2. Checks if two primes can be concatenated to form other primes
3. Builds sets of primes that satisfy the remarkable property
4. Finds the minimum sum of five such primes

```lua
function is_prime(n)
    if n < 2 then return false end
    if n == 2 then return true end
    if n % 2 == 0 then return false end
    for i = 3, math.sqrt(n), 2 do
        if n % i == 0 then return false end
    end
    return true
end

function concat_primes(a, b)
    local concat1 = tonumber(tostring(a) .. tostring(b))
    local concat2 = tonumber(tostring(b) .. tostring(a))
    return is_prime(concat1) and is_prime(concat2)
end

function can_form_set(primes, candidate)
    for _, p in ipairs(primes) do
        if not concat_primes(p, candidate) then
            return false
        end
    end
    return true
end

function find_five_prime_set()
    local primes = {}
    
    -- Generate primes up to a reasonable limit
    for i = 2, 10000 do
        if is_prime(i) then
            table.insert(primes, i)
        end
    end
    
    -- Find all valid pairs
    local pairs = {}
    for i = 1, #primes do
        for j = i + 1, #primes do
            if concat_primes(primes[i], primes[j]) then
                table.insert(pairs, {primes[i], primes[j]})
            end
        end
    end
    
    -- Find valid triplets
    local triplets = {}
    for i = 1, #pairs do
        for j = i + 1, #pairs do
            local p1, p2 = pairs[i][1], pairs[i][2]
            local p3, p4 = pairs[j][1], pairs[j][2]
            
            -- Check if these pairs share a common prime
            if (p1 == p3 or p1 == p4 or p2 == p3 or p2 == p4) then
                local common = p1 == p3 and p1 or p1 == p4 and p1 or p2 == p3 and p2 or p2
                local others = {}
                if p1 ~= common then table.insert(others, p1) end
                if p2 ~= common then table.insert(others, p2) end
                if p3 ~= common then table.insert(others, p3) end
                if p4 ~= common then table.insert(others, p4) end
                
                if #others == 2 then
                    local triplet = {common, others[1], others[2]}
                    table.sort(triplet)
                    table.insert(triplets, triplet)
                end
            end
        end
    end
    
    -- Find valid quadruplets
    local quadruplets = {}
    for i = 1, #triplets do
        for j = i + 1, #triplets do
            local t1 = triplets[i]
            local t2 = triplets[j]
            
            -- Check if they share 2 common elements
            local common = {}
            for _, v1 in ipairs(t1) do
                for _, v2 in ipairs(t2) do
                    if v1 == v2 then
                        table.insert(common, v1)
                    end
                end
            end
            
            if #common == 2 then
                local remaining = {}
                for _, v in ipairs(t1) do
                    local found = false
                    for _, c in ipairs(common) do
                        if v == c then found = true end
                    end
                    if not found then table.insert(remaining, v) end
                end
                for _, v in ipairs(t2) do
                    local found = false
                    for _, c in ipairs(common) do
                        if v == c then found = true end
                    end
                    if not found then table.insert(remaining, v) end
                end
                
                if #remaining == 2 then
                    local quad = {common[1], common[2], remaining[1], remaining[2]}
                    table.sort(quad)
                    table.insert(quadruplets, quad)
                end
            end
        end
    end
    
    -- Find valid quintuplets
    local min_sum = math.huge
    local best_set = {}
    
    for i = 1, #quadruplets do
        local quad = quadruplets[i]
        local common1 = quad[1]
        local common2 = quad[2]
        local common3 = quad[3]
        local common4 = quad[4]
        
        -- Check all primes that can form a quintuplet with this quad
        for _, prime in ipairs(primes) do
            if prime > common4 then
                local valid = true
                for _, p in ipairs(quad) do
                    if not concat_primes(p, prime) then
                        valid = false
                        break
                    end
                end
                
                if valid then
                    local quint = {quad[1], quad[2], quad[3], quad[4], prime}
                    local sum = 0
                    for _, p in ipairs(quint) do
                        sum = sum + p
                    end
                    if sum < min_sum then
                        min_sum = sum
                        best_set = quint
                    end
                end
            end
        end
    end
    
    return min_sum, best_set
end

-- Simpler approach: direct brute force
function solve_euler_60()
    local primes = {}
    for i = 2, 10000 do
        if is_prime(i) then
            table.insert(primes, i)
        end
    end
    
    -- Build adjacency list - which primes can be paired
    local adj = {}
    for i = 1, #primes do
        adj[i] = {}
        for j = i + 1, #primes do
            if concat_primes(primes[i], primes[j]) then
                table.insert(adj[i], j)
            end
        end
    end
    
    local min_sum = math.huge
    local best_set = {}
    
    -- Check all combinations of 5 primes
    for i = 1, #primes - 4 do
        for j = i + 1, #primes - 3 do
            if concat_primes(primes[i], primes[j]) then
                for k = j + 1, #primes - 2 do
                    if concat_primes(primes[i], primes[k]) and concat_primes(primes[j], primes[k]) then
                        for l = k + 1, #primes - 1 do
                            if concat_primes(primes[i], primes[l]) and concat_primes(primes[j], primes[l]) and concat_primes(primes[k], primes[l]) then
                                for m = l + 1, #primes do
                                    if concat_primes(primes[i], primes[m]) and concat_primes(primes[j], primes[m]) and concat_primes(primes[k], primes[m]) and concat_primes(primes[l], primes[m]) then
                                        local sum = primes[i] + primes[j] + primes[k] + primes[l] + primes[m]
                                        if sum < min_sum then
                                            min_sum = sum
                                            best_set = {primes[i], primes[j], primes[k], primes[l], primes[m]}
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    
    return min_sum, best_set
end

-- Run the solution
local sum, primes = solve_euler_60()
print("Lowest sum for a set of five primes: " .. sum)
print("Primes: " .. table.concat(primes, ", "))
```

## Key Insights

The approach uses:
1. **Prime generation** with efficient primality testing
2. **Concatenation checking** to verify that two primes form valid pairs
3. **Brute force combination** to find sets of 5 primes that satisfy the remarkable property
4. **Early termination** to avoid unnecessary computations
