    -- We'll use dynamic programming approach
    -- dp[i][j] = number of ways to partition i using distinct parts <= j
    local dp = {}
    
    -- Initialize dp table
    for i = 0, n do
        dp[i] = {}
        for j = 0, n do
            dp[i][j] = 0
        end
    end
    
    -- Base case: there's one way to partition 0 (using no parts)
    for j = 0, n do
        dp[0][j] = 1
    end
    
    -- Fill the dp table
    for i = 1, n do
        for j = 1, n do
            if j > i then
                dp[i][j] = dp[i][j-1]
            else
                dp[i][j] = dp[i][j-1] + dp[i-j][j-1]
            end
        end
    end

        -- Two choices: either include max_part or don't include it
