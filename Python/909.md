def euler_totient(n):
    """Calculate Euler's totient function φ(n)"""
    result = n
    p = 2
    while p * p <= n:
        if n % p == 0:
            while n % p == 0:
                n //= p
            result -= result // p
        p += 1
    if n > 1:
        result -= result // n
    return result

def solve_euler_909():
    """Find sum of all n such that φ(n) = φ(n+1) = φ(n+2)"""
    total_sum = 0
    limit = 1000000  # Set a reasonable limit to search
    
    # We'll search for patterns and check consecutive values
    for n in range(1, limit):
        phi_n = euler_totient(n)

        if phi_n == phi_n1 == phi_n2:
            total_sum += n

    return total_sum
