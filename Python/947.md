def sum_of_proper_divisors(n):
    """Calculate the sum of proper divisors of n"""
    if n <= 1:
        return 0
    
    divisors_sum = 1  # 1 is always a proper divisor for n > 1
    
    # Check divisors up to sqrt(n)
    i = 2
    while i * i <= n:
        if n % i == 0:
            divisors_sum += i
            if i != n // i:  # Avoid counting the square root twice
                divisors_sum += n // i
        i += 1
    
    return divisors_sum
